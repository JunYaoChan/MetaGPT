    def _draw_x(self, x: int, y: int) -> None:
        """Draw an X mark at the specified coordinates."""
        size = GRID_SIZE // 3
        pygame.draw.line(
            self.screen, COLORS["x"],
            (x - size, y - size), (x + size, y + size),
            LINE_WIDTH
        )
        pygame.draw.line(
            self.screen, COLORS["x"],
            (x + size, y - size), (x - size, y + size),
            LINE_WIDTH
        )
    
    def _draw_o(self, x: int, y: int) -> None:
        """Draw an O mark at the specified coordinates."""
        radius = GRID_SIZE // 3
        pygame.draw.circle(
            self.screen, COLORS["o"],
            (x, y), radius, LINE_WIDTH
        )
    
    def draw_turn_indicator(self) -> None:
        """Display which player's turn it is."""
        player = self.game.get_current_player()
        text = f"Current Turn: {player}"
        text_surface = self.font.render(text, True, COLORS["text"])
        self.screen.blit(text_surface, (MARGIN, BOARD_SIZE + MARGIN))
    
    def draw_win_message(self, winner: str) -> None:
        """Display the winner message."""
        text = f"Player {winner} wins!"
        text_surface = self.font.render(text, True, COLORS["text"])
        text_rect = text_surface.get_rect(center=(SCREEN_SIZE[0]//2, BOARD_SIZE + MARGIN))
        self.screen.blit(text_surface, text_rect)
    
    def draw_tie_message(self) -> None:
        """Display the tie game message."""
        text = "Game ended in a tie!"
        text_surface = self.font.render(text, True, COLORS["text"])
        text_rect = text_surface.get_rect(center=(SCREEN_SIZE[0]//2, BOARD_SIZE + MARGIN))
        self.screen.blit(text_surface, text_rect)

import unittest
import numpy as np
from tic_tac_toe_6.game import Game


class TestGame(unittest.TestCase):
    """Test suite for the Game class in Tic Tac Toe."""

    def setUp(self) -> None:
        """Set up a fresh Game instance before each test."""
        self.game: Game = Game()

    ## Test Initialization
    def test_initialization(self) -> None:
        """Test that the game initializes with correct default values."""
        self.assertTrue(np.array_equal(self.game.board, np.full((3, 3), None)),
                        "Board should be initialized to 3x3 None array")
        self.assertEqual(self.game.current_player, 'X',
                         "Current player should be 'X' initially")
        self.assertFalse(self.game.game_over,
                         "Game should not be over initially")

    ## Test make_move
    def test_make_valid_move(self) -> None:
        """Test making a valid move updates the board and switches player."""
        result: bool = self.game.make_move(0, 0)
        self.assertTrue(result, "Valid move should return True")
        self.assertEqual(self.game.board[0, 0], 'X',
                         "Board should be updated with 'X'")
        self.assertEqual(self.game.current_player, 'O',
                         "Current player should switch to 'O'")

    def test_make_invalid_move_occupied(self) -> None:
        """Test making a move on an occupied cell returns False."""
        self.game.make_move(0, 0)  # First move by 'X'
        result: bool = self.game.make_move(0, 0)  # Try to make same move by 'O'
        self.assertFalse(result, "Move on occupied cell should return False")
        self.assertEqual(self.game.current_player, 'O',
                         "Current player should remain 'O' after invalid move")

    def test_make_move_after_game_over(self) -> None:
        """Test making a move when game is over returns False."""
        self.game.game_over = True
        result: bool = self.game.make_move(0, 0)
        self.assertFalse(result, "Move after game over should return False")

    ## Test check_winner
    def test_check_winner_row(self) -> None:
        """Test checking winner in a row."""
        self.game.board = np.array([['X', 'X', 'X'],
                                   [None, None, None],
                                   [None, None, None]])
        winner: Optional[str] = self.game.check_winner()
        self.assertEqual(winner, 'X', "Should detect 'X' as winner in row")

    def test_check_winner_column(self) -> None:
        """Test checking winner in a column."""
        self.game.board = np.array([['O', None, None],
                                   ['O', None, None],
                                   ['O', None, None]])
        winner: Optional[str] = self.game.check_winner()
        self.assertEqual(winner, 'O', "Should detect 'O' as winner in column")

    def test_check_winner_diagonal(self) -> None:
        """Test checking winner in main diagonal."""
        self.game.board = np.array([['X', None, None],
                                   [None, 'X', None],
                                   [None, None, 'X']])
        winner: Optional[str] = self.game.check_winner()
        self.assertEqual(winner, 'X', "Should detect 'X' as winner in diagonal")

    def test_check_winner_anti_diagonal(self) -> None:
        """Test checking winner in anti-diagonal."""
        self.game.board = np.array([[None, None, 'O'],
                                   [None, 'O', None],
                                   ['O', None, None]])
        winner: Optional[str] = self.game.check_winner()
        self.assertEqual(winner, 'O', "Should detect 'O' as winner in anti-diagonal")

    def test_check_winner_no_winner(self) -> None:
        """Test checking winner when no winner exists."""
        self.game.board = np.array([['X', 'O', 'X'],
                                   ['X', 'O', 'O'],
                                   ['O', 'X', 'X']])
        winner: Optional[str] = self.game.check_winner()
        self.assertIsNone(winner, "Should return None when no winner")

    ## Test is_board_full
    def test_is_board_full_true(self) -> None:
        """Test board is full when all cells are occupied."""
        self.game.board = np.array([['X', 'O', 'X'],
                                   ['X', 'O', 'O'],
                                   ['O', 'X', 'X']])
        self.assertTrue(self.game.is_board_full(),
                       "Should return True when board is full")

    def test_is_board_full_false(self) -> None:
        """Test board is not full when some cells are empty."""
        self.game.board = np.array([['X', 'O', 'X'],
                                   [None, 'O', 'O'],
                                   ['O', 'X', 'X']])
        self.assertFalse(self.game.is_board_full(),
                        "Should return False when board has empty cells")

    ## Test reset_game
    def test_reset_game(self) -> None:
        """Test resetting the game returns it to initial state."""
        self.game.make_move(0, 0)
        self.game.make_move(1, 1)
        self.game.game_over = True
        self.game.reset_game()
        self.assertTrue(np.array_equal(self.game.board, np.full((3, 3), None)),
        self.assertEqual(self.game.current_player, 'X',
                         "Current player should reset to 'X'")
        self.assertFalse(self.game.game_over,
                         "Game over should reset to False")

    ## Test get_current_player
    def test_get_current_player(self) -> None:
        """Test getting the current player."""
        self.assertEqual(self.game.get_current_player(), 'X',
                         "Should return 'X' initially")
        self.game.make_move(0, 0)
        self.assertEqual(self.game.get_current_player(), 'O',
                         "Should return 'O' after first move")
