import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='solid')
        self.buttons = []
        self.create_display()
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.grid(row=0, column=0, columnspan=4)
    
    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            '0', 'C', '=', '+'
        ]
        
        row_val = 1
        col_val = 0
        
        for text in button_texts:
            button = tk.Button(self.root, text=text, width=5, height=2, font=('Arial', 18),
                               command=lambda t=text: self.on_button_click(t))
            button.grid(row=row_val, column=col_val)
            self.buttons.append(button)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def on_button_click(self, value: str) -> None:
        """Handle button click events."""
        if value == 'C':
            self.clear_display()
        elif value == '=':
            self.calculate()
        else:
            self.add_to_display(value)

    def add_to_display(self, value: str) -> None:
        """Add a value to the display."""
        current_text = self.display.get()
        new_text = current_text + value
        self.display.delete(0, tk.END)
        self.display.insert(0, new_text)

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Calculate the result of the expression in the display."""
        try:
            expression = self.display.get()
            result = eval(expression)  # Use eval to calculate the expression
            self.clear_display()
            self.add_to_display(str(result))
        except Exception as e:
            messagebox.showerror("Error", "Invalid Input")
            self.clear_display()

    def run(self) -> None:
        """Run the main loop of the calculator."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
