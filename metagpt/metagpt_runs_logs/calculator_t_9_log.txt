import unittest
from unittest.mock import patch, MagicMock
import tkinter as tk
from calculator_t_9.main import CalculatorUI
from calculator import Calculator

class TestCalculatorUI(unittest.TestCase):
    """Test cases for CalculatorUI class"""

    def setUp(self) -> None:
        """Set up test environment"""
        self.calculator_ui = CalculatorUI()
        self.calculator_ui.root = MagicMock()  # Mock Tk root to avoid GUI during tests
        self.calculator_ui.display = MagicMock()

    def tearDown(self) -> None:
        """Clean up after tests"""
        del self.calculator_ui

    ## INITIALIZATION TESTS
    def test_initialization(self) -> None:
        """Test that CalculatorUI initializes correctly"""
        self.assertIsInstance(self.calculator_ui.calculator, Calculator)
        self.assertEqual(self.calculator_ui.root.title.call_args[0][0], "Simple Calculator")

    ## UI CREATION TESTS
    def test_create_ui_button_count(self) -> None:
        """Test that all buttons are created"""
        with patch('tkinter.Button') as mock_button:
            self.calculator_ui.create_ui()
            self.assertEqual(mock_button.call_count, 16)  # 16 buttons in the layout

    def test_create_ui_button_layout(self) -> None:
        """Test button grid layout positions"""
        with patch('tkinter.Button') as mock_button:
            self.calculator_ui.create_ui()
            
            # Check first button position (7)
            first_call = mock_button.call_args_list[0]
            self.assertEqual(first_call[1]['text'], '7')
            self.assertEqual(first_call[1]['width'], 5)
            self.assertEqual(first_call[1]['height'], 2)
            
            # Check last button position (+)
            last_call = mock_button.call_args_list[-1]
            self.assertEqual(last_call[1]['text'], '+')

    ## DISPLAY UPDATE TESTS
    def test_update_display(self) -> None:
        """Test display update functionality"""
        test_value = "123"
        self.calculator_ui.calculator.get_current_input = MagicMock(return_value=test_value)
        
        self.calculator_ui.update_display()
        
        self.calculator_ui.display.delete.assert_called_once_with(0, tk.END)
        self.calculator_ui.display.insert.assert_called_once_with(tk.END, test_value)

    ## BUTTON HANDLER TESTS
    def test_handle_button_click_clear(self) -> None:
        """Test clear button functionality"""
        self.calculator_ui.calculator.clear = MagicMock()
        self.calculator_ui.update_display = MagicMock()
        
        self.calculator_ui.handle_button_click('C')
        
        self.calculator_ui.calculator.clear.assert_called_once()
        self.calculator_ui.update_display.assert_called_once()

    def test_handle_button_click_equals(self) -> None:
        """Test equals button functionality"""
        self.calculator_ui.calculator.calculate = MagicMock()
        self.calculator_ui.update_display = MagicMock()
        
        self.calculator_ui.handle_button_click('=')
        
        self.calculator_ui.calculator.calculate.assert_called_once()
        self.calculator_ui.update_display.assert_called_once()

    def test_handle_button_click_operation(self) -> None:
        """Test operation buttons functionality"""
        for operation in '+-*/':
            with self.subTest(operation=operation):
                self.calculator_ui.calculator.input_operation = MagicMock()
                self.calculator_ui.update_display = MagicMock()
                
                self.calculator_ui.handle_button_click(operation)
                
                self.calculator_ui.calculator.input_operation.assert_called_once_with(operation)
                self.calculator_ui.update_display.assert_called_once()

    def test_handle_button_click_digit(self) -> None:
        """Test digit buttons functionality"""
        for digit in '0123456789':
            with self.subTest(digit=digit):
                self.calculator_ui.calculator.input_digit = MagicMock()
                self.calculator_ui.update_display = MagicMock()
                
                self.calculator_ui.handle_button_click(digit)
                
                self.calculator_ui.calculator.input_digit.assert_called_once_with(digit)
                self.calculator_ui.update_display.assert_called_once()

    ## EDGE CASES

        """Test the run method starts mainloop"""

        self.calculator_ui.run()
        self.calculator_ui.root.mainloop.assert_called_once()

if __name__ == "__main__":
    unittest.main()

"""

import unittest
from calculator_t_9.calculator import Calculator


## TestCalculator
class TestCalculator(unittest.TestCase):
    """Test cases for the Calculator class."""

    def setUp(self) -> None:
        """Set up a fresh Calculator instance before each test."""
        self.calc = Calculator()

    ## TestInitialization
    def test_initialization(self) -> None:
        """Test that the calculator initializes with default values."""
        self.assertEqual(self.calc.current_input, "")
        self.assertEqual(self.calc.operation, "")
        self.assertEqual(self.calc.result, 0.0)

    ## TestInputDigit
    def test_input_digit(self) -> None:
        """Test input_digit method with valid digits."""
        test_cases: list[str] = ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"]
        for digit in test_cases:
            self.calc.input_digit(digit)
        self.assertEqual(self.calc.get_current_input(), "1234567890")

    ## TestInputOperation
    def test_input_operation_valid(self) -> None:
        """Test input_operation with valid operations."""
        operations: list[str] = ["+", "-", "*", "/"]
        self.calc.input_digit("5")
        for op in operations:
            self.calc.input_operation(op)
            self.assertEqual(self.calc.operation, op)
            self.assertEqual(self.calc.get_current_input(), "")
            self.assertEqual(self.calc.get_result(), 5.0)

    def test_input_operation_no_input(self) -> None:
        """Test input_operation when no input is present."""
        self.calc.input_operation("+")
        self.assertEqual(self.calc.operation, "")
        self.assertEqual(self.calc.get_current_input(), "")
        self.assertEqual(self.calc.get_result(), 0.0)

    ## TestCalculate
    def test_calculate_addition(self) -> None:
        """Test calculate method with addition."""
        self.calc.input_digit("5")
        self.calc.input_operation("+")
        self.calc.input_digit("3")
        result: float = self.calc.calculate()
        self.assertEqual(result, 8.0)
        self.assertEqual(self.calc.get_current_input(), "")
        self.assertEqual(self.calc.operation, "")

    def test_calculate_subtraction(self) -> None:
        """Test calculate method with subtraction."""
        self.calc.input_digit("10")
        self.calc.input_operation("-")
        self.calc.input_digit("3")
        result: float = self.calc.calculate()
        self.assertEqual(result, 7.0)

    def test_calculate_multiplication(self) -> None:
        """Test calculate method with multiplication."""
        self.calc.input_digit("5")
        self.calc.input_operation("*")
        self.calc.input_digit("3")
        result: float = self.calc.calculate()
        self.assertEqual(result, 15.0)

    def test_calculate_division(self) -> None:
        """Test calculate method with division."""
        self.calc.input_digit("10")
        self.calc.input_operation("/")
        self.calc.input_digit("2")
        result: float = self.calc.calculate()
        self.assertEqual(result, 5.0)

    def test_calculate_division_by_zero(self) -> None:
        """Test calculate method with division by zero."""
        self.calc.input_digit("10")
        self.calc.input_operation("/")
        self.calc.input_digit("0")
        with self.assertRaises(ValueError):
            self.calc.calculate()

    def test_calculate_no_operation(self) -> None:
        """Test calculate when no operation is set."""
        self.calc.input_digit("5")
        result: float = self.calc.calculate()
        self.assertEqual(result, 0.0)  # Should return stored result (0.0 by default)

    def test_calculate_no_second_operand(self) -> None:
        """Test calculate when second operand is missing."""
        self.calc.input_digit("5")
        self.calc.input_operation("+")
        result: float = self.calc.calculate()
        self.assertEqual(result, 5.0)  # Should return first operand

    ## TestClear
    def test_clear(self) -> None:
        """Test clear method resets calculator state."""
        self.calc.input_digit("5")
        self.calc.input_operation("+")
        self.calc.input_digit("3")
        self.calc.clear()
        self.assertEqual(self.calc.get_current_input(), "")
        self.assertEqual(self.calc.operation, "")
        self.assertEqual(self.calc.get_result(), 0.0)

    ## TestGetCurrentInput
    def test_get_current_input(self) -> None:
        """Test get_current_input method."""
        self.calc.input_digit("123")
        self.assertEqual(self.calc.get_current_input(), "123")

    ## TestGetResult
    def test_get_result(self) -> None:
        """Test get_result method."""
        self.calc.input_digit("5")
        self.calc.input_operation("+")
        self.calc.input_digit("3")
        self.calc.calculate()
        self.assertEqual(self.calc.get_result(), 8.0)


if __name__ == "__main__":
    unittest.main()
