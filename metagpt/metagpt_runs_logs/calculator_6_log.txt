import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator application using tkinter for GUI."""

    def __init__(self) -> None:
        """Initialize the calculator application."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='solid')
        self.display.grid(row=0, column=0, columnspan=4)
        self.create_buttons()
        self.root.mainloop()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.delete(0, tk.END)  # Clear the display

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        # Button layout
        buttons = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        # Create buttons and place them in the grid
        row_val = 1
        col_val = 0
        for button in buttons:
            btn = tk.Button(self.root, text=button, width=5, height=2, font=('Arial', 18),
                            command=lambda char=button: self.button_click(char))
            btn.grid(row=row_val, column=col_val)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            current_text = self.display.get()
            new_text = current_text + char
            self.display.delete(0, tk.END)  # Clear the display
            self.display.insert(0, new_text)  # Update display with new text

    def clear_display(self) -> None:
        """Clear the calculator display."""
        self.display.delete(0, tk.END)  # Clear the display

    def calculate(self) -> None:
        """Evaluate the expression in the display and show the result."""
        try:
            expression = self.display.get()
            result = eval(expression)  # Evaluate the expression
            self.display.delete(0, tk.END)  # Clear the display
            self.display.insert(0, str(result))  # Show the result
        except Exception as e:
            messagebox.showerror("Error", "Invalid input")  # Show error message

if __name__ == "__main__":
    Calculator()  # Run the calculator application
