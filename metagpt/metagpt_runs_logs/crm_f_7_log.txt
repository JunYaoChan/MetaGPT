## main.py

import sqlite3
import streamlit as st

class Customer:
    """Class representing a customer with basic attributes."""
    
    def __init__(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        self.name = name
        self.birthday = birthday
        self.age = age
        self.sex = sex
        self.phone = phone

class DatabaseManager:
    """Class to manage database operations for customer data."""
    
    def __init__(self) -> None:
        """Initialize the database connection and cursor."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.initialize_database()

    def initialize_database(self) -> None:
        """Create the customer table if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                name TEXT PRIMARY KEY,
                birthday TEXT,
                age INTEGER,
                sex TEXT,
                phone TEXT
            )
        ''')
        self.conn.commit()

    def add_customer(self, customer: Customer) -> None:
        """Add a new customer to the database."""
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone)
            VALUES (?, ?, ?, ?, ?)
        ''', (customer.name, customer.birthday, customer.age, customer.sex, customer.phone))
        self.conn.commit()

    def get_customer(self, name: str) -> Customer:
        """Retrieve a customer by name."""
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        if row:
            return Customer(*row)
        return None

    def update_customer(self, customer: Customer) -> None:
        """Update an existing customer's information."""
        self.cursor.execute('''
            UPDATE customers
            SET birthday = ?, age = ?, sex = ?, phone = ?
            WHERE name = ?
        ''', (customer.birthday, customer.age, customer.sex, customer.phone, customer.name))
        self.conn.commit()

    def delete_customer(self, name: str) -> None:
        """Delete a customer from the database."""
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()

    def get_all_customers(self) -> list[Customer]:
        """Retrieve all customers from the database."""
        self.cursor.execute('SELECT * FROM customers')
        rows = self.cursor.fetchall()
        return [Customer(*row) for row in rows]

class StreamlitUI:
    """Class to render the Streamlit user interface for the CRM system."""
    
    def __init__(self) -> None:
        """Initialize the Streamlit UI with a DatabaseManager instance."""
        self.db_manager = DatabaseManager()

    def render(self) -> None:
        """Render the main UI with tabs for different operations."""
        st.title("Customer Relationship Management System")
        tab = st.sidebar.selectbox("Select Operation", ["Add Customer", "View Customer", "Update Customer", "Delete Customer"])
        
        if tab == "Add Customer":
            self.render_add_tab()
        elif tab == "View Customer":
            self.render_view_tab()
        elif tab == "Update Customer":
            self.render_update_tab()
        elif tab == "Delete Customer":
            self.render_delete_tab()

    def render_add_tab(self) -> None:
        """Render the tab for adding a new customer."""
        st.header("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")
        
        if st.button("Add Customer"):
            customer = Customer(name, birthday, age, sex, phone)
            self.db_manager.add_customer(customer)
            st.success(f"Customer {name} added successfully!")

    def render_view_tab(self) -> None:
        """Render the tab for viewing a customer's information."""
        st.header("View Customer")
        name = st.text_input("Enter Customer Name")
        
        if st.button("View Customer"):
            customer = self.db_manager.get_customer(name)
            if customer:
                st.write(f"Name: {customer.name}")
                st.write(f"Birthday: {customer.birthday}")
                st.write(f"Age: {customer.age}")
                st.write(f"Sex: {customer.sex}")
                st.write(f"Phone: {customer.phone}")
            else:
                st.error("Customer not found.")

    def render_update_tab(self) -> None:
        """Render the tab for updating a customer's information."""
        st.header("Update Customer")
        name = st.text_input("Enter Customer Name to Update")
        
        if st.button("Fetch Customer"):
            customer = self.db_manager.get_customer(name)
            if customer:
                birthday = st.text_input("Birthday (YYYY-MM-DD)", value=customer.birthday)
                age = st.number_input("Age", min_value=0, value=customer.age)
                sex = st.selectbox("Sex", ["Male", "Female", "Other"], index=["Male", "Female", "Other"].index(customer.sex))
                phone = st.text_input("Phone", value=customer.phone)
                
                if st.button("Update Customer"):
                    updated_customer = Customer(name, birthday, age, sex, phone)
                    self.db_manager.update_customer(updated_customer)
                    st.success(f"Customer {name} updated successfully!")
            else:
                st.error("Customer not found.")

    def render_delete_tab(self) -> None:
        """Render the tab for deleting a customer."""
        st.header("Delete Customer")
        name = st.text_input("Enter Customer Name to Delete")
        
        if st.button("Delete Customer"):
            self.db_manager.delete_customer(name)
            st.success(f"Customer {name} deleted successfully!")

if __name__ == "__main__":
    ui = StreamlitUI()
    ui.render()
