## main.py
import tkinter as tk

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator with a Tkinter root window and GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='solid')
        self.display.grid(row=0, column=0, columnspan=4)
        self.buttons = {}
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.grid(row=0, column=0, columnspan=4)

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        # Button layout
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        # Create buttons and place them in the grid
        for i, text in enumerate(button_texts):
            self.buttons[text] = tk.Button(self.root, text=text, width=5, height=2,
                                           command=lambda char=text: self.button_click(char))
            self.buttons[text].grid(row=(i // 4) + 1, column=i % 4)

    def button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            current_text = self.display.get()
            new_text = current_text + char
            self.display.delete(0, tk.END)
            self.display.insert(0, new_text)

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Perform the calculation based on the current display text."""
        try:
            # Evaluate the expression in the display
            result = eval(self.display.get())
            self.clear_display()
            self.display.insert(0, str(result))
        except Exception as e:
            self.clear_display()
            self.display.insert(0, "Error")

    def run(self) -> None:
        """Run the main loop of the Tkinter application."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
