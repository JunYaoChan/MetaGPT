```"""
Tic Tac Toe game logic implementation.
Handles board state, move validation, win checking, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game rules."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.board: List[List[str]] = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3):
            return False
            
        if self.board[row][col] != ' ':
            return False
            
        self.board[row][col] = self.current_player
        self._check_game_state()
        
        if not self.game_over:
            self._switch_player()
            
        return True

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if the current move resulted in a win or tie.
        Updates game_over and winner accordingly.
        """
        winner = self.check_winner()
        if winner:
            self.game_over = True
            self.winner = winner
        elif self.is_board_full():
            self.game_over = True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.
        
        Returns:
            Optional[str]: 'X' or 'O' if there's a winner, None otherwise
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] != ' ':
                return row[0]

        # Check columns
        for col in range(3):
            if self.board[0][col] == self.board[1][col] == self.board[2][col] != ' ':
                return self.board[0][col]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] != ' ':
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != ' ':
            return self.board[0][2]

        return None

    def is_board_full(self) -> bool:
        """
        Check if the board is full (tie game).
        
        Returns:
            bool: True if board is full, False otherwise
        """
        return all(cell != ' ' for row in self.board for cell in row)

    def reset_game(self) -> None:
        """Reset the game to initial state."""
        self.board = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = None

    def get_empty_positions(self) -> List[Tuple[int, int]]:
        """
        Get all empty positions on the board.
        
        Returns:
            List[Tuple[int, int]]: List of (row, col) tuples of empty positions
        """
        return [(r, c) for r in range(3) for c in range(3) if self.board[r][c] == ' ']

    def get_board_copy(self) -> List[List[str]]:
        """
        Create a deep copy of the current board state.
        
        Returns:
            List[List[str]]: Copy of the current board
        """
        return [row.copy() for row in self.board]

```"""
Tic Tac Toe game logic implementation.
Handles board state, move validation, win checking, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game rules."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.board: List[List[str]] = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3):
            return False
            
        if self.board[row][col] != ' ':
            return False
            
        self.board[row][col] = self.current_player
        self._check_game_state()
        
        if not self.game_over:
            self._switch_player()
            
        return True

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if the current move resulted in a win or tie.
        Updates game_over and winner accordingly.
        """
        winner = self.check_winner()
        if winner:
            self.game_over = True
            self.winner = winner
        elif self.is_board_full():
            self.game_over = True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.
        
        Returns:
            Optional[str]: 'X' or 'O' if there's a winner, None otherwise
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] != ' ':
                return row[0]

        # Check columns
        for col in range(3):
            if self.board[0][col] == self.board[1][col] == self.board[2][col] != ' ':
                return self.board[0][col]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] != ' ':
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != ' ':
            return self.board[0][2]

        return None

    def is_board_full(self) -> bool:
        """
        Check if the board is full (tie game).
        
        Returns:
            bool: True if board is full, False otherwise
        """
        return all(cell != ' ' for row in self.board for cell in row)

    def reset_game(self) -> None:
        """Reset the game to initial state."""
        self.board = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = None

    def get_empty_positions(self) -> List[Tuple[int, int]]:
        """
        Get all empty positions on the board.
        
        Returns:
            List[Tuple[int, int]]: List of (row, col) tuples of empty positions
        """
        return [(r, c) for r in range(3) for c in range(3) if self.board[r][c] == ' ']

    def get_board_copy(self) -> List[List[str]]:
        """
        Create a deep copy of the current board state.
        
        Returns:
            List[List[str]]: Copy of the current board
        """
        return [row.copy() for row in self.board]

```"""
AI module for Tic Tac Toe game implementing minimax algorithm with difficulty levels.
"""

import random
from typing import Tuple, Optional, List
from game import Game

class AI:
    """
    AI opponent for Tic Tac Toe with adjustable difficulty levels.
    
    Attributes:
        difficulty (str): Difficulty level ('easy', 'medium', 'hard')
        player_symbol (str): Symbol used by AI ('O')
        opponent_symbol (str): Symbol used by human player ('X')
    """
    
    def __init__(self, difficulty: str = 'medium') -> None:
        """
        Initialize AI with specified difficulty level.
        
        Args:
            difficulty: AI difficulty level ('easy', 'medium', 'hard')
        """
        self.difficulty = difficulty.lower()
        self.player_symbol = 'O'
        self.opponent_symbol = 'X'
        
        if self.difficulty not in ['easy', 'medium', 'hard']:
            raise ValueError("Difficulty must be 'easy', 'medium' or 'hard'")

    def get_move(self, game: Game) -> Tuple[int, int]:
        """
        Get AI move based on current game state and difficulty level.
        
        Args:
            game: Current game instance
            
        Returns:
            Tuple containing (row, col) of selected move
        """
        if self.difficulty == 'easy':
            return self._get_random_move(game)
        elif self.difficulty == 'medium':
            return self._get_minimax_move(game, depth=3)
        else:  # hard
            return self._get_minimax_move(game)

    def _get_random_move(self, game: Game) -> Tuple[int, int]:
        """
        Get a random valid move from available positions.
        
        Args:
            game: Current game instance
            
        Returns:
            Random valid move coordinates
        """
        empty_cells = [
            (i, j) 
            for i in range(3) 
            for j in range(3) 
            if game.board[i][j] == ''
        ]
        return random.choice(empty_cells)

    def _get_minimax_move(self, game: Game, depth: Optional[int] = None) -> Tuple[int, int]:
        """
        Get best move using minimax algorithm with optional depth limit.
        
        Args:
            game: Current game instance
            depth: Maximum search depth (None for unlimited)
            
        Returns:
            Best move coordinates according to minimax
        """
        best_score = -float('inf')
        best_move = (-1, -1)
        
        for i in range(3):
            for j in range(3):
                if game.board[i][j] == '':
                    game.board[i][j] = self.player_symbol
                    score = self._minimax(game, False, depth)
                    game.board[i][j] = ''  # undo move
                    
                    if score > best_score:
                        best_score = score
                        best_move = (i, j)
                        
        return best_move

    def _minimax(self, game: Game, is_maximizing: bool, 
                depth: Optional[int], current_depth: int = 0) -> int:
        """
        Recursive minimax algorithm implementation.
        
        Args:
            game: Current game instance
            is_maximizing: True if maximizing player's turn
            depth: Maximum search depth (None for unlimited)
            current_depth: Current recursion depth
            
        Returns:
            Heuristic score for current board state
        """
        # Base cases
        winner = game.check_winner()
        if winner == self.player_symbol:
            return 10 - current_depth
        elif winner == self.opponent_symbol:
            return -10 + current_depth
        elif game.is_board_full():
            return 0
            
        # Depth limit check
        if depth is not None and current_depth >= depth:
            return 0
            
        # Recursive case
        if is_maximizing:
            best_score = -float('inf')
            for i in range(3):
                for j in range(3):
                    if game.board[i][j] == '':
                        game.board[i][j] = self.player_symbol
                        score = self._minimax(game, False, depth, current_depth + 1)
                        game.board[i][j] = ''  # undo move
                        best_score = max(score, best_score)
            return best_score
        else:
            best_score = float('inf')
            for i in range(3):
                for j in range(3):
                    if game.board[i][j] == '':
                        game.board[i][j] = self.opponent_symbol
                        score = self._minimax(game, True, depth, current_depth + 1)
                        game.board[i][j] = ''  # undo move
                        best_score = min(score, best_score)
            return best_score

```python
"""
Tic Tac Toe game logic implementation.
Handles board state, move validation, win checking, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game rules."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.board: List[List[str]] = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3):
            return False
            
        if self.board[row][col] != ' ':
            return False
            
        self.board[row][col] = self.current_player
        self._check_game_state()
        
        if not self.game_over:
            self._switch_player()
            
        return True

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if the current move resulted in a win or tie.
        Updates game_over and winner accordingly.
        """
        winner = self.check_winner()
        if winner:
            self.game_over = True
            self.winner = winner
        elif self.is_board_full():
            self.game_over = True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.
        
        Returns:
            Optional[str]: 'X' or 'O' if there's a winner, None otherwise
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] != ' ':
                return row[0]

        # Check columns
        for col in range(3):
            if self.board[0][col] == self.board[1][col] == self.board[2][col] != ' ':
                return self.board[0][col]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] != ' ':
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != ' ':
            return self.board[0][2]

        return None

    def is_board_full(self) -> bool:
        """
        Check if the board is full (tie game).
        
        Returns:
            bool: True if board is full, False otherwise
        """
        for row in self.board:
            if ' ' in row:
                return False
        return True

    def reset_game(self) -> None:
        """Reset the game to initial state."""
        self.board = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = None

    def get_empty_positions(self) -> List[Tuple[int, int]]:
        """
        Get all empty positions on the board.
        
        Returns:
            List[Tuple[int, int]]: List of (row, col) tuples of empty positions
        """
        return [(r, c) for r in range(3) for c in range(3) if self.board[r][c] == ' ']

    def get_board_copy(self) -> List[List[str]]:
        """
        Create a deep copy of the current board state.
        
        Returns:
            List[List[str]]: Copy of the current board
        """
        return [row.copy() for row in self.board]

        """

            (i, j)

            for j in range(3) 
            if game.board[i][j] == ''
        ]
