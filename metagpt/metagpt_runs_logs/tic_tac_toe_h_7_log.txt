```python
"""Tic Tac Toe game implementation using Pygame.

This module contains the Game class which handles the game state,
logic, and rendering for a Tic Tac Toe game.
"""

import pygame
from typing import List, Optional, Tuple

# Constants would normally be imported from constants.py
# Defining them here temporarily for completeness
SCREEN_WIDTH = 600
SCREEN_HEIGHT = 700
BOARD_SIZE = 3
CELL_SIZE = 200
LINE_WIDTH = 15
X_COLOR = (255, 0, 0)
O_COLOR = (0, 0, 255)
LINE_COLOR = (0, 0, 0)
BG_COLOR = (255, 255, 255)
STATUS_FONT_SIZE = 40
BUTTON_COLOR = (0, 255, 0)
BUTTON_TEXT_COLOR = (0, 0, 0)
BUTTON_RECT = pygame.Rect(200, 620, 200, 50)


class Game:
    """Main game class handling Tic Tac Toe logic and rendering.

    Attributes:
        screen: The Pygame surface to draw on.
        board: 2D list representing the game board.
        current_player: Current player ('X' or 'O').
        game_over: Boolean indicating if game has ended.
        winner: The winning player ('X', 'O') or None if draw.
    """

    def __init__(self, screen: pygame.Surface) -> None:
        """Initialize the game with empty board and starting player.

        Args:
            screen: Pygame surface to render the game on.
        """
        self.screen = screen
        self.board: List[List[str]] = [[None for _ in range(BOARD_SIZE)] 
                                      for _ in range(BOARD_SIZE)]
        self.current_player = 'X'
        self.game_over = False
        self.winner: Optional[str] = None

    def handle_click(self, pos: Tuple[int, int]) -> None:
        """Handle mouse click events on the game board or restart button.

        Args:
            pos: Tuple of (x, y) coordinates of the mouse click.
        """
        if self.game_over:
            if BUTTON_RECT.collidepoint(pos):
                self.reset_game()
            return

        x, y = pos[0] // CELL_SIZE, pos[1] // CELL_SIZE
        if 0 <= x < BOARD_SIZE and 0 <= y < BOARD_SIZE and not self.board[y][x]:
            self.board[y][x] = self.current_player
            self.check_winner()
            if not self.game_over:
                self.current_player = 'O' if self.current_player == 'X' else 'X'

    def check_winner(self) -> None:
        """Check if current player has won or if game is a draw."""
        # Check rows
        for row in self.board:
            if all(cell == self.current_player for cell in row):
                self.game_over = True
                self.winner = self.current_player
                return

        # Check columns
        for col in range(BOARD_SIZE):
            if all(self.board[row][col] == self.current_player 
                   for row in range(BOARD_SIZE)):
                self.game_over = True
                self.winner = self.current_player
                return

        # Check diagonals
        if all(self.board[i][i] == self.current_player 
               for i in range(BOARD_SIZE)):
            self.game_over = True
            self.winner = self.current_player
            return

        if all(self.board[i][BOARD_SIZE-1-i] == self.current_player 
               for i in range(BOARD_SIZE)):
            self.game_over = True
            self.winner = self.current_player
            return

        # Check for draw
        if self.is_draw():
            self.game_over = True
            self.winner = None

    def is_draw(self) -> bool:
        """Check if the game is a draw (no empty spaces left).

        Returns:
            True if game is a draw, False otherwise.
        """
        return all(all(cell is not None for cell in row) for row in self.board)

    def reset_game(self) -> None:
        """Reset the game state to start a new game."""
        self.board = [[None for _ in range(BOARD_SIZE)] 
                     for _ in range(BOARD_SIZE)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = None

    def draw_board(self) -> None:
        """Draw the Tic Tac Toe board and current moves."""
        # Draw grid lines
        for i in range(1, BOARD_SIZE):
            # Vertical lines
            pygame.draw.line(
                self.screen, LINE_COLOR,
                (i * CELL_SIZE, 0), (i * CELL_SIZE, BOARD_SIZE * CELL_SIZE),
                LINE_WIDTH
            )
            # Horizontal lines
            pygame.draw.line(
                self.screen, LINE_COLOR,
                (0, i * CELL_SIZE), (BOARD_SIZE * CELL_SIZE, i * CELL_SIZE),
                LINE_WIDTH
            )

        # Draw X's and O's
        for y in range(BOARD_SIZE):
            for x in range(BOARD_SIZE):
                if self.board[y][x] == 'X':
                    self._draw_x(x, y)
                elif self.board[y][x] == 'O':
                    self._draw_o(x, y)

    def _draw_x(self, x: int, y: int) -> None:
        """Draw an X at the specified board position.

        Args:
            x: Column index (0-2).
            y: Row index (0-2).
        """
        center_x = x * CELL_SIZE + CELL_SIZE // 2
        center_y = y * CELL_SIZE + CELL_SIZE // 2
        offset = CELL_SIZE // 3

        pygame.draw.line(
            self.screen, X_COLOR,
            (center_x - offset, center_y - offset),
            (center_x + offset, center_y + offset),
            LINE_WIDTH
        )
        pygame.draw.line(
            self.screen, X_COLOR,
            (center_x + offset, center_y - offset),
            (center_x - offset, center_y + offset),
            LINE_WIDTH
        )

    def _draw_o(self, x: int, y: int) -> None:
        """Draw an O at the specified board position.

        Args:
            x: Column index (0-2).
            y: Row index (0-2).
        """
        center_x = x * CELL_SIZE + CELL_SIZE // 2
        center_y = y * CELL_SIZE + CELL_SIZE // 2
        radius = CELL_SIZE // 3

        pygame.draw.circle(
            self.screen, O_COLOR,
            (center_x, center_y), radius, LINE_WIDTH
        )

    def draw_status(self) -> None:
        """Draw the current game status (player turn or winner)."""
        font = pygame.font.Font(None, STATUS_FONT_SIZE)
        if self.game_over:
            if self.winner:
                text = f"Player {self.winner} wins!"
            else:
                text = "Game ended in a draw!"
        else:
            text = f"Player {self.current_player}'s turn"

        text_surface = font.render(text, True, LINE_COLOR)
        text_rect = text_surface.get_rect(center=(SCREEN_WIDTH//2, 650))
        self.screen.blit(text_surface, text_rect)

    def draw_restart_button(self) -> None:
        """Draw the restart game button."""
        pygame.draw.rect(self.screen, BUTTON_COLOR, BUTTON_RECT)
        font = pygame.font.Font(None, 30)
        text_surface = font.render("Restart Game", True, BUTTON_TEXT_COLOR)
        text_rect = text_surface.get_rect(center=BUTTON_RECT.center)
        self.screen.blit(text_surface, text_rect)
```

```python
"""Test cases for the Tic Tac Toe Game class.

This module contains unit tests for the Game class in game.py using Python's
unittest framework. The tests cover game initialization, move handling,
win/draw detection, and game reset functionality.
"""

import unittest
from unittest.mock import MagicMock
import pygame
from tic_tac_toe_h_7.game import Game


class TestGame(unittest.TestCase):
    """Test suite for the Game class."""

    def setUp(self) -> None:
        """Set up test fixtures."""
        # Mock pygame screen
        self.mock_screen = MagicMock()
        self.game = Game(self.mock_screen)

    ## Test Initialization
    def test_initialization(self) -> None:
        """Test that game initializes with correct default state."""
        self.assertEqual(len(self.game.board), 3, "Board should have 3 rows")
        self.assertEqual(len(self.game.board[0]), 3, "Board should have 3 columns")
        self.assertTrue(all(cell is None for row in self.game.board for cell in row),
                        "Board should be empty at start")
        self.assertEqual(self.game.current_player, 'X',
                         "First player should be X")
        self.assertFalse(self.game.game_over, "Game should not be over at start")
        self.assertIsNone(self.game.winner, "There should be no winner at start")

    ## Test Handle Click
    def test_handle_click_valid_move(self) -> None:
        """Test handling valid moves on the board."""
        # First click (X's turn)
        self.game.handle_click((50, 50))  # Should be cell (1, 1)
        self.assertEqual(self.game.board[1][1], 'X',
                         "Cell should be marked with X")
        self.assertEqual(self.game.current_player, 'O',
                         "Current player should switch to O")

        # Second click (O's turn)
        self.game.handle_click((150, 150))  # Should be cell (3, 3)
        self.assertEqual(self.game.board[3][3], 'O',
                         "Cell should be marked with O")
        self.assertEqual(self.game.current_player, 'X',
                         "Current player should switch back to X")

    def test_handle_click_out_of_bounds(self) -> None:
        """Test that clicks outside board are ignored."""
        initial_board = [row[:] for row in self.game.board]
        self.game.handle_click((-10, -10))  # Out of bounds
        self.game.handle_click((1000, 1000))  # Out of bounds
        self.assertEqual(self.game.board, initial_board,
                         "Board should not change for out of bounds clicks")

    def test_handle_click_occupied_cell(self) -> None:
        """Test that clicks on occupied cells are ignored."""
        self.game.board[1][1] = 'X'
        initial_board = [row[:] for row in self.game.board]
        self.game.handle_click((50, 50))  # Click on occupied cell
        self.assertEqual(self.game.board, initial_board,
                         "Board should not change for occupied cell clicks")

    def test_handle_click_after_game_over(self) -> None:
        """Test that board clicks are ignored after game over."""
        self.game.game_over = True
        initial_board = [row[:] for row in self.game.board]
        self.game.handle_click((50, 50))  # Click on empty cell
        self.assertEqual(self.game.board, initial_board,
                         "Board should not change after game over")

    ## Test Check Winner
    def test_check_winner_row(self) -> None:
        """Test row win detection."""
        # Set up a row win for X
        self.game.board = [
            ['X', 'X', 'X'],
            [None, None, None],
            [None, None, None]
        ]
        self.game.current_player = 'X'
        self.game.check_winner()
        self.assertTrue(self.game.game_over, "Game should be over for row win")
        self.assertEqual(self.game.winner, 'X', "X should win for row")

    def test_check_winner_column(self) -> None:
        """Test column win detection."""
        # Set up a column win for O
        self.game.board = [
            ['O', None, None],
            ['O', None, None],
            ['O', None, None]
        ]
        self.game.current_player = 'O'
        self.game.check_winner()
        self.assertTrue(self.game.game_over, "Game should be over for column win")
        self.assertEqual(self.game.winner, 'O', "O should win for column")

    def test_check_winner_diagonal(self) -> None:
        """Test diagonal win detection."""
        # Set up a diagonal win for X
        self.game.board = [
            ['X', None, None],
            [None, 'X', None],
            [None, None, 'X']
        ]
        self.game.current_player = 'X'
        self.game.check_winner()
        self.assertTrue(self.game.game_over, "Game should be over for diagonal win")
        self.assertEqual(self.game.winner, 'X', "X should win for diagonal")

    def test_check_winner_anti_diagonal(self) -> None:
        """Test anti-diagonal win detection."""
        # Set up an anti-diagonal win for O
        self.game.board = [
            [None, None, 'O'],
            [None, 'O', None],
            ['O', None, None]
        ]
        self.game.current_player = 'O'
        self.game.check_winner()
        self.assertTrue(self.game.game_over, "Game should be over for anti-diagonal win")
        self.assertEqual(self.game.winner, 'O', "O should win for anti-diagonal")

    def test_check_draw(self) -> None:
        """Test draw detection."""
        # Set up a draw scenario
        self.game.board = [
            ['X', 'O', 'X'],
            ['X', 'X', 'O'],
            ['O', 'X', 'O']
        ]
        self.game.current_player = 'O'
        self.game.check_winner()
        self.assertTrue(self.game.game_over, "Game should be over for draw")
        self.assertIsNone(self.game.winner, "Winner should be None for draw")

    ## Test Reset Game
    def test_reset_game(self) -> None:
        """Test game reset functionality."""
        # Set up a game state
        self.game.board = [

        ]
        self.game.current_player = 'O'
```
