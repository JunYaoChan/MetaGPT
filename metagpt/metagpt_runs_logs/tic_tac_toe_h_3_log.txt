        text_rect = text.get_rect(center=(SCREEN_SIZE[0] // 2, SCREEN_SIZE[1] - 30))

    def handle_click(self, pos: Tuple[int, int]) -> bool:
        """
        Handle mouse click events.

        Args:
            pos: Tuple of (x, y) coordinates of the mouse click.

        Returns:
            bool: True if the click was handled, False otherwise.
        """
        if self.game.game_over:
            self.game.reset_game()
            return True

        col = pos[0] // self.cell_size
        row = pos[1] // self.cell_size

        if 0 <= row < GRID_SIZE and 0 <= col < GRID_SIZE:
            return self.game.make_move(row, col)

        return False

"""
Test cases for Tic Tac Toe game logic implementation.
"""

import unittest
from tic_tac_toe_h_3.game import Game


class TestGame(unittest.TestCase):
    """Test suite for the Game class."""

    def setUp(self) -> None:
        """Set up a fresh Game instance before each test."""
        self.game = Game()

    ## INITIALIZATION TESTS
    def test_initialization(self) -> None:
        """Test that the game initializes with correct default values."""
        self.assertEqual(self.game.current_player, 'X')
        self.assertFalse(self.game.game_over)
        self.assertIsNone(self.game.winner)
        for row in self.game.board:
            for cell in row:
                self.assertIsNone(cell)

    ## MOVE VALIDATION TESTS
    def test_valid_move(self) -> None:
        """Test that valid moves are accepted and board is updated."""
        self.assertTrue(self.game.make_move(0, 0))
        self.assertEqual(self.game.board[0][0], 'X')
        self.assertEqual(self.game.current_player, 'O')

    def test_invalid_move_out_of_bounds(self) -> None:
        """Test that out-of-bounds moves are rejected."""
        self.assertFalse(self.game.make_move(-1, 0))
        self.assertFalse(self.game.make_move(3, 0))
        self.assertFalse(self.game.make_move(0, -1))
        self.assertFalse(self.game.make_move(0, 3))

    def test_invalid_move_occupied_cell(self) -> None:
        """Test that moves to occupied cells are rejected."""
        self.game.make_move(0, 0)
        self.assertFalse(self.game.make_move(0, 0))

    def test_invalid_move_after_game_over(self) -> None:
        """Test that moves are rejected after game is over."""
        # Force a win
        self.game.make_move(0, 0)  # X
        self.game.make_move(1, 0)  # O
        self.game.make_move(0, 1)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(0, 2)  # X wins
        self.assertTrue(self.game.game_over)
        self.assertFalse(self.game.make_move(2, 2))

    ## WIN DETECTION TESTS
    def test_row_win(self) -> None:
        """Test that row wins are detected correctly."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(1, 0)  # O
        self.game.make_move(0, 1)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(0, 2)  # X wins
        self.assertTrue(self.game.game_over)
        self.assertEqual(self.game.winner, 'X')

    def test_column_win(self) -> None:
        """Test that column wins are detected correctly."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(0, 1)  # O
        self.game.make_move(1, 0)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(2, 0)  # X wins
        self.assertTrue(self.game.game_over)
        self.assertEqual(self.game.winner, 'X')

    def test_diagonal_win(self) -> None:
        """Test that diagonal wins are detected correctly."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(0, 1)  # O
        self.game.make_move(1, 1)  # X
        self.game.make_move(0, 2)  # O
        self.game.make_move(2, 2)  # X wins
        self.assertTrue(self.game.game_over)
        self.assertEqual(self.game.winner, 'X')

    def test_anti_diagonal_win(self) -> None:
        """Test that anti-diagonal wins are detected correctly."""
        self.game.make_move(0, 2)  # X
        self.game.make_move(0, 1)  # O
        self.game.make_move(1, 1)  # X
        self.game.make_move(0, 0)  # O
        self.game.make_move(2, 0)  # X wins
        self.assertTrue(self.game.game_over)
        self.assertEqual(self.game.winner, 'X')

    ## DRAW DETECTION TESTS
    def test_draw_detection(self) -> None:
        """Test that draws are detected correctly."""
        # X | O | X
        # X | O | O
        # O | X | X
        moves = [
            (0, 0), (0, 1), (0, 2),
            (1, 1), (1, 0), (1, 2),
            (2, 1), (2, 0), (2, 2)
        ]
        for row, col in moves:
            self.game.make_move(row, col)
        self.assertTrue(self.game.game_over)
        self.assertIsNone(self.game.winner)

    ## PLAYER SWITCHING TESTS
    def test_player_switching(self) -> None:
        """Test that players switch turns correctly."""
        self.assertEqual(self.game.current_player, 'X')
        self.game.make_move(0, 0)
        self.assertEqual(self.game.current_player, 'O')
        self.game.make_move(0, 1)
        self.assertEqual(self.game.current_player, 'X')

    def test_no_player_switch_after_win(self) -> None:
        """Test that player doesn't switch after game is won."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(1, 0)  # O
        self.game.make_move(0, 1)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(0, 2)  # X wins
        self.assertEqual(self.game.current_player, 'X')

    ## GAME RESET TESTS
    def test_game_reset(self) -> None:
        """Test that game reset works correctly."""
        self.game.make_move(0, 0)
        self.game.make_move(0, 1)
        self.game.reset_game()
        self.assertEqual(self.game.current_player, 'X')
        self.assertFalse(self.game.game_over)
        self.assertIsNone(self.game.winner)
        for row in self.game.board:
            for cell in row:
                self.assertIsNone(cell)

    ## CURRENT PLAYER TESTS
    def test_get_current_player(self) -> None:
        """Test that current player is returned correctly."""
        self.assertEqual(self.game.get_current_player(), 'X')
        self.game.make_move(0, 0)
        self.assertEqual(self.game.get_current_player(), 'O')
