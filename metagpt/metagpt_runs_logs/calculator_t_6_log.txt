```python
## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='solid')
        self.display.grid(row=0, column=0, columnspan=4)
        self.buttons = {}
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.delete(0, tk.END)  # Clear the display

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        # Define button layout
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            '0', 'C', '=', '+'
        ]
        
        # Create buttons and place them in the grid
        for index, text in enumerate(button_texts):
            button = tk.Button(self.root, text=text, width=5, height=2, font=('Arial', 18),
                               command=lambda t=text: self.add_to_display(t))
            button.grid(row=(index // 4) + 1, column=index % 4)
            self.buttons[text] = button

    def add_to_display(self, value: str) -> None:
        """Add a value to the display."""
        if value == 'C':
            self.clear_display()
        elif value == '=':
            self.calculate()
        else:
            self.display.insert(tk.END, value)

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Calculate the result of the expression in the display."""
        try:
            # Evaluate the expression in the display
            result = eval(self.display.get())
            self.clear_display()
            self.display.insert(tk.END, str(result))
        except Exception as e:
            # Show an error message if the calculation fails
            messagebox.showerror("Error", "Invalid input or calculation error.")
            self.clear_display()

    def run(self) -> None:
        """Run the main loop of the calculator."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
```

- The `create_buttons` method generates the calculator buttons and arranges them in a grid layout.
- The `add_to_display` method handles button presses, including clearing the display and calculating results.
- The `clear_display` method clears the current input in the display.
- The `calculate` method evaluates the expression entered in the display and handles any errors that may occur during evaluation.
- The `run` method starts the Tkinter main loop to display the GUI.
