## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class using Tkinter for GUI."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = self.create_display()
        self.buttons = self.create_buttons()
        self.current_input = ""

    def create_display(self) -> tk.Entry:
        """Create and return the display for the calculator."""
        display = tk.Entry(self.root, font=("Arial", 24), bd=10, insertwidth=2, width=14, borderwidth=4)
        display.grid(row=0, column=0, columnspan=4)
        return display

    def create_buttons(self) -> dict:
        """Create and return a dictionary of buttons for the calculator."""
        buttons = {
            '7': (1, 0), '8': (1, 1), '9': (1, 2), '/': (1, 3),
            '4': (2, 0), '5': (2, 1), '6': (2, 2), '*': (2, 3),
            '1': (3, 0), '2': (3, 1), '3': (3, 2), '-': (3, 3),
            '0': (4, 0), 'C': (4, 1), '=': (4, 2), '+': (4, 3)
        }
        for text, grid_position in buttons.items():
            button = tk.Button(self.root, text=text, padx=20, pady=20, font=("Arial", 18),
                               command=lambda t=text: self.on_button_click(t))
            button.grid(row=grid_position[0], column=grid_position[1])
        return buttons

    def on_button_click(self, value: str) -> None:
        """Handle button click events."""
        if value == 'C':
            self.clear_display()
        elif value == '=':
            self.calculate()
        else:
            self.add_to_display(value)

    def add_to_display(self, value: str) -> None:
        """Add a value to the current input display."""
        self.current_input += value
        self.display.delete(0, tk.END)  # Clear the display
        self.display.insert(tk.END, self.current_input)  # Show current input

    def clear_display(self) -> None:
        """Clear the current input and reset the display."""
        self.current_input = ""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Calculate the result of the current input and display it."""
        try:
            # Evaluate the expression in the current input
            result = eval(self.current_input)
            self.display.delete(0, tk.END)  # Clear the display
            self.display.insert(tk.END, str(result))  # Show result
            self.current_input = str(result)  # Update current input to result
        except ZeroDivisionError:
            self.handle_error("Error: Division by zero.")
        except Exception as e:
            self.handle_error(f"Error: {str(e)}")

    def handle_error(self, message: str) -> None:
        """Display an error message in a message box."""
        messagebox.showerror("Calculation Error", message)
        self.clear_display()  # Clear the display after showing error

    def run(self) -> None:
        """Run the main loop of the Tkinter application."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
