 __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.reset_game()

    def reset_game(self) -> None:
        """Reset the game state to start a new game."""
        self.board: List[List[Optional[str]]] = [
            [None, None, None],
            [None, None, None],
            [None, None, None],
        ]
        self.current_player: str = "X"
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.

        Args:
            row: Row index (0-2)
            col: Column index (0-2)

        Returns:
            bool: True if move was successful, False if invalid
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3):
            return False

        if self.board[row][col] is not None:
            return False

        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self.current_player = "O" if self.current_player == "X" else "X"
        return True

    def _check_game_state(self) -> None:
        """Check if current move resulted in a win or draw."""
        winner = self._check_winner()
        if winner:
            self.winner = winner
            self.game_over = True
        elif self._is_draw():
            self.game_over = True

    def _check_winner(self) -> Optional[str]:
        """
        Check if the current player has won.

        Returns:
            Optional[str]: The winning player ("X" or "O") or None if no winner
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] and row[0] is not None:
                return row[0]

        # Check columns
        for col in range(3):
            if (
                self.board[0][col] == self.board[1][col] == self.board[2][col]
                and self.board[0][col] is not None
            ):
                return self.board[0][col]

        # Check diagonals
        if (
            self.board[0][0] == self.board[1][1] == self.board[2][2]
            and self.board[0][0] is not None
        ):
            return self.board[0][0]
        if (
            self.board[0][2] == self.board[1][1] == self.board[2][0]
            and self.board[0][2] is not None
        ):
            return self.board[0][2]

        return None

    def _is_draw(self) -> bool:
        """
        Check if the game is a draw (no winner and board full).

        Returns:
            bool: True if game is a draw, False otherwise
        """
        return all(cell is not None for row in self.board for cell in row)

class TestGame(unittest.TestCase):

    ## INITIALIZATION TESTS
    def test_initialization(self) -> None:
        """Test game initializes with correct default state."""
        self.assertEqual(self.game.board, [
            [None, None, None],
            [None, None, None],
            [None, None, None],
        ])
        self.assertEqual(self.game.current_player, "X")
        self.assertFalse(self.game.game_over)
        self.assertIsNone(self.game.winner)

    ## RESET GAME TESTS
    def test_reset_game(self) -> None:
        """Test game reset functionality."""
        # Make some moves and change state
        self.game.make_move(0, 0)
        self.game.make_move(1, 1)
        self.game.game_over = True
        self.game.winner = "X"

        # Reset and verify
        self.game.reset_game()
        self.assertEqual(self.game.board, [
            [None, None, None],
            [None, None, None],
            [None, None, None],
        ])
        self.assertEqual(self.game.current_player, "X")
        self.assertFalse(self.game.game_over)
        self.assertIsNone(self.game.winner)

    ## MOVE VALIDATION TESTS
    def test_valid_move(self) -> None:
        """Test valid move is accepted and board updated."""
        result = self.game.make_move(1, 1)
        self.assertTrue(result)
        self.assertEqual(self.game.board[1][1], "X")
        self.assertEqual(self.game.current_player, "O")

    def test_invalid_move_out_of_bounds(self) -> None:
        """Test moves outside board boundaries are rejected."""
        self.assertFalse(self.game.make_move(-1, 0))
        self.assertFalse(self.game.make_move(3, 0))
        self.assertFalse(self.game.make_move(0, -1))
        self.assertFalse(self.game.make_move(0, 3))

    def test_invalid_move_occupied_cell(self) -> None:
        """Test moves to occupied cells are rejected."""
        self.game.make_move(0, 0)
        self.assertFalse(self.game.make_move(0, 0))

    def test_move_after_game_over(self) -> None:
        """Test moves are rejected after game ends."""
        # Force game over state
        self.game.game_over = True
        self.assertFalse(self.game.make_move(0, 0))

    ## WIN DETECTION TESTS
    def test_row_win(self) -> None:
        """Test row win detection."""
        moves = [(0, 0), (1, 0), (0, 1), (1, 1), (0, 2)]
