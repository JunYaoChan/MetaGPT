```"""
Tic Tac Toe game core logic implementation.
Handles board state, move validation, win/draw checking, and game reset.
"""

from typing import List, Optional


class Game:
    """Main game class managing the Tic Tac Toe board and game state."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.board: List[List[str]] = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.

        Args:
            row: Row index (0-2)
            col: Column index (0-2)

        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3):
            return False

        if self.board[row][col] != ' ':
            return False

        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self._switch_player()
        return True

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if the current player has won or if the game is a draw.
        Updates game_over and winner accordingly.
        """
        winner = self._check_winner()
        if winner:
            self.game_over = True
            self.winner = winner
        elif self._is_draw():
            self.game_over = True

    def _check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.

        Returns:
            Optional[str]: The winning player ('X' or 'O') or None if no winner
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] != ' ':
                return row[0]

        # Check columns
        for col in range(3):
            if self.board[0][col] == self.board[1][col] == self.board[2][col] != ' ':
                return self.board[0][col]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] != ' ':
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != ' ':
            return self.board[0][2]

        return None

    def _is_draw(self) -> bool:
        """
        Check if the game is a draw (no winner and board full).

        Returns:
            bool: True if game is a draw, False otherwise
        """
        return all(cell != ' ' for row in self.board for cell in row)

    def reset_game(self) -> None:
        """Reset the game to initial state."""
        self.board = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = None

    def get_board(self) -> List[List[str]]:
        """
        Get the current board state.

        Returns:
            List[List[str]]: 3x3 board state
        """
        return self.board

    def get_current_player(self) -> str:
        """
        Get the current player.

        Returns:
            str: Current player ('X' or 'O')
        """
        return self.current_player

    def is_game_over(self) -> bool:
        """
        Check if the game is over.

        Returns:
            bool: True if game is over, False otherwise
        """
        return self.game_over

    def get_winner(self) -> Optional[str]:
        """
        Get the winner if game is over.

        Returns:
            Optional[str]: Winner ('X' or 'O') or None if no winner/draw
        """
        return self.winner

```"""
Tic Tac Toe game core logic implementation.
Handles board state, move validation, win/draw checking, and game reset.
"""

from typing import List, Optional


class Game:
    """Main game class managing the Tic Tac Toe board and game state."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.board: List[List[str]] = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.

        Args:
            row: Row index (0-2)
            col: Column index (0-2)

        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3):
            return False

        if self.board[row][col] != ' ':
            return False

        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self._switch_player()
        return True

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if the current player has won or if the game is a draw.
        Updates game_over and winner accordingly.
        """
        winner = self._check_winner()
        if winner:
            self.game_over = True
            self.winner = winner
        elif self._is_draw():
            self.game_over = True

    def _check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.

        Returns:
            Optional[str]: The winning player ('X' or 'O') or None if no winner
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] != ' ':
                return row[0]

        # Check columns
        for col in range(3):
            if self.board[0][col] == self.board[1][col] == self.board[2][col] != ' ':
                return self.board[0][col]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] != ' ':
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != ' ':
            return self.board[0][2]

        return None

    def _is_draw(self) -> bool:
        """
        Check if the game is a draw (no winner and board full).

        Returns:
            bool: True if game is a draw, False otherwise
        """
        return all(cell != ' ' for row in self.board for cell in row)

    def reset_game(self) -> None:
        """Reset the game to initial state."""
        self.board = [[' ' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = None

    def get_board(self) -> List[List[str]]:
        """
        Get the current board state.

        Returns:
            List[List[str]]: 3x3 board state
        """
        return self.board

    def get_current_player(self) -> str:
        """
        Get the current player.

        Returns:
            str: Current player ('X' or 'O')
        """
        return self.current_player

    def is_game_over(self) -> bool:
        """
        Check if the game is over.

        Returns:
            bool: True if game is over, False otherwise
        """
        return self.game_over

    def get_winner(self) -> Optional[str]:
        """
        Get the winner if game is over.

        Returns:
            Optional[str]: Winner ('X' or 'O') or None if no winner/draw
        """
        return self.winner

```"""AI module for Tic Tac Toe game implementing different difficulty levels."""

import random
from typing import Tuple
from game import Game


class AI:
    """AI player for Tic Tac Toe with configurable difficulty levels.
    
    Attributes:
        difficulty: The difficulty level of the AI ('easy', 'medium', 'hard').
        player_symbol: The symbol the AI uses ('X' or 'O').
    """

    def __init__(self, difficulty: str = 'medium', player_symbol: str = 'O') -> None:
        """Initialize AI with specified difficulty and player symbol.
        
        Args:
            difficulty: AI difficulty level ('easy', 'medium', 'hard').
            player_symbol: The symbol the AI will use ('X' or 'O').
        """
        self.difficulty = difficulty.lower()
        self.player_symbol = player_symbol.upper()
        self.opponent_symbol = 'X' if self.player_symbol == 'O' else 'O'

    def get_move(self, game: Game) -> Tuple[int, int]:
        """Get the AI's next move based on current game state and difficulty.
        
        Args:
            game: The current game instance.
            
        Returns:
            A tuple (row, col) representing the AI's move.
        """
        if self.difficulty == 'easy':
            return self._get_random_move(game)
        elif self.difficulty == 'medium':
            return self._get_medium_move(game)
        else:  # hard difficulty
            return self._get_hard_move(game)

    def _get_random_move(self, game: Game) -> Tuple[int, int]:
        """Get a random valid move.
        
        Args:
            game: The current game instance.
            
        Returns:
            A random valid (row, col) move.
        """
        valid_moves = []
        for row in range(3):
            for col in range(3):
                if game.board[row][col] == '':
                    valid_moves.append((row, col))
        return random.choice(valid_moves)

    def _get_medium_move(self, game: Game) -> Tuple[int, int]:
        """Get a move with basic strategy (win if possible, block if needed).
        
        Args:
            game: The current game instance.
            
        Returns:
            A strategic (row, col) move.
        """
        # First check for winning move
        move = self._find_winning_move(game, self.player_symbol)
        if move:
            return move

        # Then check for blocking opponent's winning move
        move = self._find_winning_move(game, self.opponent_symbol)
        if move:
            return move

        # Otherwise make random move
        return self._get_random_move(game)

    def _get_hard_move(self, game: Game) -> Tuple[int, int]:
        """Get the best move using minimax algorithm.
        
        Args:
            game: The current game instance.
            
        Returns:
            The optimal (row, col) move.
        """
        best_score = -float('inf')
        best_move = None

        for row in range(3):
            for col in range(3):
                if game.board[row][col] == '':
                    game.board[row][col] = self.player_symbol
                    score = self._minimax(game, False)
                    game.board[row][col] = ''  # Undo move

                    if score > best_score:
                        best_score = score
                        best_move = (row, col)

        return best_move or self._get_random_move(game)

    def _minimax(self, game: Game, is_maximizing: bool) -> int:
        """Minimax algorithm implementation for optimal move selection.
        
        Args:
            game: The current game instance.
            is_maximizing: Whether current player is maximizing or minimizing.
            
        Returns:
            The score of the current board state.
        """
        winner = game.check_winner()
        if winner == self.player_symbol:
            return 1
        elif winner == self.opponent_symbol:
            return -1
        elif game.is_draw():
            return 0

        if is_maximizing:
            best_score = -float('inf')
            for row in range(3):
                for col in range(3):
                    if game.board[row][col] == '':
                        game.board[row][col] = self.player_symbol
                        score = self._minimax(game, False)
                        game.board[row][col] = ''  # Undo move
                        best_score = max(score, best_score)
            return best_score
        else:
            best_score = float('inf')
            for row in range(3):
                for col in range(3):
                    if game.board[row][col] == '':
                        game.board[row][col] = self.opponent_symbol
                        score = self._minimax(game, True)
                        game.board[row][col] = ''  # Undo move
                        best_score = min(score, best_score)
            return best_score

    def _find_winning_move(self, game: Game, symbol: str) -> Tuple[int, int] | None:
        """Find a winning move for the specified symbol if one exists.
        
        Args:
            game: The current game instance.
            symbol: The symbol to check for ('X' or 'O').
            
        Returns:
            The winning (row, col) move if found, else None.
        """
        # Check rows
        for row in range(3):
            if game.board[row].count(symbol) == 2 and '' in game.board[row]:
                col = game.board[row].index('')
                return (row, col)

        # Check columns
        for col in range(3):
            column = [game.board[row][col] for row in range(3)]
            if column.count(symbol) == 2 and '' in column:
                row = column.index('')
                return (row, col)

        # Check diagonals
        diagonal1 = [game.board[i][i] for i in range(3)]
        if diagonal1.count(symbol) == 2 and '' in diagonal1:
            pos = diagonal1.index('')
            return (pos, pos)

        diagonal2 = [game.board[i][2-i] for i in range(3)]
        if diagonal2.count(symbol) == 2 and '' in diagonal2:
            pos = diagonal2.index('')
            return (pos, 2-pos)

        return None

        """Process the AI's turn if it's their move."""
        if not self.game.game_over and self.game.current_player == self.ai.player_symbol:

        """Redraw the entire game board and status."""
