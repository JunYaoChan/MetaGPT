class Calculator:
    def __init__(self) -> None:
        """Initialize the Calculator class, setting up the main window and components."""
        self.root = tk.Tk()  # Create the main window
        self.root.title("Simple Calculator")  # Set the window title
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='ridge')  # Create display
        self.buttons = []  # List to hold button references
        self.create_display()  # Call method to create display
        self.create_buttons()  # Call method to create buttons
        self.display.grid(row=0, column=0, columnspan=4)  # Place display in the grid
        self.root.mainloop()  # Start the main event loop

    def create_display(self) -> None:
        """Create the display for the calculator."""
        # The display is already created in the __init__ method, so no additional code is needed here.

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        # Define button layout
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        # Create buttons and place them in the grid
        for index, text in enumerate(button_texts):
            button = tk.Button(self.root, text=text, width=5, height=2, font=('Arial', 18),
                               command=lambda t=text: self.button_click(t))  # Bind button click to handler
            button.grid(row=(index // 4) + 1, column=index % 4)  # Calculate grid position
            self.buttons.append(button)  # Store button reference

    def button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == 'C':
            self.clear_display()  # Clear display if 'C' is pressed
        elif char == '=':
            self.calculate()  # Calculate result if '=' is pressed
        else:
            current_text = self.display.get()  # Get current text from display
            new_text = current_text + char  # Append clicked character
            self.display.delete(0, tk.END)  # Clear display
            self.display.insert(0, new_text)  # Update display with new text

    def calculate(self) -> None:
        """Perform the calculation based on the current display text."""
        try:
            result = eval(self.display.get())  # Evaluate the expression in the display
            self.display.delete(0, tk.END)  # Clear display
            self.display.insert(0, str(result))  # Show result in display
        except Exception as e:
            self.display.delete(0, tk.END)  # Clear display on error
            self.display.insert(0, "Error")  # Show error message

    def clear_display(self) -> None:
        """Clear the calculator display."""
        self.display.delete(0, tk.END)  # Clear the display

    Calculator()  # Create an instance of the Calculator to run the application

from tkinter import font as tkfont


class Calculator:
    """A simple calculator application using tkinter."""
    
    def __init__(self, root: tk.Tk) -> None:
        """Initialize the calculator application.
        
        Args:
            root: The main tkinter window.
        """
        self.root = root
        self.root.title("Calculator")
        self.root.resizable(False, False)
        self.root.configure(bg='#f0f0f0')
        
        # Custom font for buttons and display
        self.button_font = tkfont.Font(family='Helvetica', size=12, weight='bold')
        self.display_font = tkfont.Font(family='Helvetica', size=20)
        
        self.display_var = tk.StringVar()
        self.buttons = []
        
        self.create_display()
        self.create_buttons()
    
    def create_display(self) -> None:
        """Create the calculator display area."""
        display_frame = tk.Frame(self.root, height=50, bg='#f0f0f0')
        display_frame.pack(expand=True, fill='both', padx=5, pady=5)
        
        self.display = tk.Entry(
            display_frame,
            textvariable=self.display_var,
            font=self.display_font,
            borderwidth=0,
            relief='flat',
            justify='right',
            bg='#ffffff',
            fg='#333333',
            insertwidth=0
        )
        self.display.pack(expand=True, fill='both', ipady=10)
        self.display.bind('<Key>', lambda e: 'break')  # Disable keyboard input
    
    def create_buttons(self) -> None:
        """Create all calculator buttons."""
        buttons_frame = tk.Frame(self.root, bg='#f0f0f0')
        buttons_frame.pack(expand=True, fill='both', padx=5, pady=5)
        
        button_layout = [
            ('7', '8', '9', '/'),
            ('4', '5', '6', '*'),
            ('1', '2', '3', '-'),
            ('C', '0', '=', '+')
        ]
        
        for row_idx, row in enumerate(button_layout):
            button_row = []
            for col_idx, button_text in enumerate(row):
                button = tk.Button(
                    buttons_frame,
                    text=button_text,
                    font=self.button_font,
                    borderwidth=1,
                    relief='raised',
                    command=lambda text=button_text: self.button_click(text),
                    bg='#e0e0e0',
                    fg='#333333',
                    activebackground='#d0d0d0'
                )
                button.grid(
                    row=row_idx,
                    column=col_idx,
                    sticky='nsew',
                    padx=2,
                    pady=2
                )
                buttons_frame.grid_columnconfigure(col_idx, weight=1)
                button_row.append(button)
            buttons_frame.grid_rowconfigure(row_idx, weight=1)
            self.buttons.append(button_row)
    
    def button_click(self, char: str) -> None:
        """Handle button clicks.
        
        Args:
            char: The character/operation from the button clicked.
        """
        current_text = self.display_var.get()
        
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            self.display_var.set(current_text + char)
    
    def calculate(self) -> None:
        """Perform the calculation and display the result."""
        try:
            expression = self.display_var.get()
            if expression:
                result = str(eval(expression))
                self.display_var.set(result)
        except Exception:
            self.display_var.set("Error")
    
    def clear_display(self) -> None:
        """Clear the calculator display."""
        self.display_var.set('')

from tkinter import font as tkfont


class Calculator:
    """A simple calculator application using tkinter."""
    
    def __init__(self, root: tk.Tk) -> None:
        """Initialize the calculator application.
        
        Args:
            root: The main tkinter window.
        """
        self.root = root
        self.root.title("Calculator")
        self.root.resizable(False, False)
        self.root.configure(bg='#f0f0f0')
        
        # Custom font for buttons and display
        self.button_font = tkfont.Font(family='Helvetica', size=12, weight='bold')
        self.display_font = tkfont.Font(family='Helvetica', size=20)
        
        self.display_var = tk.StringVar()
        self.buttons = []
        
        self.create_display()
        self.create_buttons()
    
    def create_display(self) -> None:
        """Create the calculator display area."""
        display_frame = tk.Frame(self.root, height=50, bg='#f0f0f0')
        display_frame.pack(expand=True, fill='both', padx=5, pady=5)
        
        self.display = tk.Entry(
            display_frame,
            textvariable=self.display_var,
            font=self.display_font,
            borderwidth=0,
            relief='flat',
            justify='right',
            bg='#ffffff',
            fg='#333333',
            insertwidth=0
        )
        self.display.pack(expand=True, fill='both', ipady=10)
        self.display.bind('<Key>', lambda e: 'break')  # Disable keyboard input
    
    def create_buttons(self) -> None:
        """Create all calculator buttons."""
        buttons_frame = tk.Frame(self.root, bg='#f0f0f0')
        buttons_frame.pack(expand=True, fill='both', padx=5, pady=5)
        
        button_layout = [
            ('7', '8', '9', '/'),
            ('4', '5', '6', '*'),
            ('1', '2', '3', '-'),
            ('C', '0', '=', '+')
        ]
        
        for row_idx, row in enumerate(button_layout):
            button_row = []
            for col_idx, button_text in enumerate(row):
                button = tk.Button(
                    buttons_frame,
                    text=button_text,
                    font=self.button_font,
                    borderwidth=1,
                    relief='raised',

                    fg='#333333',
                    activebackground='#d0d0d0'

                button.grid(
                    row=row_idx,
                    column=col_idx,
                    sticky='nsew',
                    padx=2,
                    pady=2
                )
                buttons_frame.grid_columnconfigure(col_idx, weight=1)
                button_row.append(button)
            buttons_frame.grid_rowconfigure(row_idx, weight=1)
            self.buttons.append(button_row)
    
    def button_click(self, char: str) -> None:
        """Handle button clicks.
        
        Args:
            char: The character/operation from the button clicked.
        """
        current_text = self.display_var.get()
        
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            self.display_var.set(current_text + char)
    
    def calculate(self) -> None:
        """Perform the calculation and display the result."""
        try:
            expression = self.display_var.get()
            if expression:
                result = str(eval(expression))
                self.display_var.set(result)
        except Exception:
            self.display_var.set("Error")

import tkinter as tk
from tkinter import font as tkfont


class Calculator:
    """Main calculator class handling UI and calculation logic."""

    def __init__(self, root: tk.Tk) -> None:
        """Initialize the calculator application.
        
        Args:
            root: The main tkinter window.
        """
        self.root = root
        self.root.title('Calculator')
        self.root.resizable(False, False)
        
        # Configure styles
        self.button_font = tkfont.Font(family='Helvetica', size=12, weight='bold')
        self.display_font = tkfont.Font(family='Helvetica', size=20)
        
        self.display_var = tk.StringVar()
        self.buttons = []
        
        self.create_display()
        self.create_buttons()

    def create_display(self) -> None:
        """Create the calculator display area."""
        display_frame = tk.Frame(self.root, height=100)
        display_frame.pack(expand=True, fill='both')
        
        display = tk.Entry(
            display_frame,
            textvariable=self.display_var,
            font=self.display_font,
            borderwidth=0,
            justify='right',
            state='readonly',
            readonlybackground='white'
        )
        display.pack(expand=True, fill='both', padx=10, pady=10)

    def create_buttons(self) -> None:
        """Create all calculator buttons."""
        button_frame = tk.Frame(self.root)
        button_frame.pack(expand=True, fill='both')
        
        button_layout = [
            ('7', '8', '9', '/'),
            ('4', '5', '6', '*'),
            ('1', '2', '3', '-'),
            ('0', 'C', '=', '+')
        ]
        
        for row_idx, row in enumerate(button_layout):
            button_frame.rowconfigure(row_idx, weight=1)
            for col_idx, char in enumerate(row):
                button_frame.columnconfigure(col_idx, weight=1)
                
                button = tk.Button(
                    button_frame,
                    text=char,
                    font=self.button_font,
                    borderwidth=1,
                    command=lambda c=char: self.button_click(c)
                )
                button.grid(
                    row=row_idx,
                    column=col_idx,
                    sticky='nsew',
                    padx=2,
                    pady=2
                )
                self.buttons.append(button)

    def button_click(self, char: str) -> None:

        Args:
            char: The character/operation from the clicked button.

        current_text = self.display_var.get()
        
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            new_text = current_text + char

        """Perform the calculation and display the result."""
        try:
            expression = self.display_var.get()
            if not expression:
                return
                
            result = str(eval(expression))  # pylint: disable=eval-used
            self.display_var.set(result)
        except Exception:  # pylint: disable=broad-except
            self.display_var.set('Error')

    def clear_display(self) -> None:
        """Clear the calculator display."""
        self.display_var.set('')


def main() -> None:
    """Main function to start the calculator application."""
    root = tk.Tk()

    root.mainloop()


if __name__ == '__main__':

"""A simple calculator application using tkinter."""

import tkinter as tk
from tkinter import font as tkfont


class Calculator:
    """Main calculator class handling UI and calculation logic."""

    def __init__(self, root: tk.Tk) -> None:
        """Initialize the calculator application.
        
        Args:
            root: The main tkinter window.
        """
        self.root = root
        self.root.title('Calculator')
        self.root.resizable(False, False)
        
        # Configure styles
        self.button_font = tkfont.Font(family='Helvetica', size=12, weight='bold')
        self.display_font = tkfont.Font(family='Helvetica', size=20)
        
        self.display_var = tk.StringVar()
        self.buttons = []
        
        self.create_display()
        self.create_buttons()

    def create_display(self) -> None:
        """Create the calculator display area."""
        display_frame = tk.Frame(self.root, height=100)
        display_frame.pack(expand=True, fill='both')
        
        display = tk.Entry(
            display_frame,
            textvariable=self.display_var,
            font=self.display_font,
            borderwidth=0,
            justify='right',
            state='readonly',
            readonlybackground='white'
        )
        display.pack(expand=True, fill='both', padx=10, pady=10)

    def create_buttons(self) -> None:
        """Create all calculator buttons."""
        button_frame = tk.Frame(self.root)
        button_frame.pack(expand=True, fill='both')
        
        button_layout = [
            ('7', '8', '9', '/'),
            ('4', '5', '6', '*'),
            ('1', '2', '3', '-'),
            ('0', 'C', '=', '+')
        ]
        
        for row_idx, row in enumerate(button_layout):
            button_frame.rowconfigure(row_idx, weight=1)
            for col_idx, char in enumerate(row):
                button_frame.columnconfigure(col_idx, weight=1)
                
                button = tk.Button(
                    button_frame,
                    text=char,
                    font=self.button_font,
                    borderwidth=1,
                    command=lambda c=char: self.button_click(c)
                )
                button.grid(
                    row=row_idx,
                    column=col_idx,
                    sticky='nsew',
                    padx=2,
                    pady=2
                )
                self.buttons.append(button)

    def button_click(self, char: str) -> None:
        """Handle button click events.
        
        Args:
            char: The character/operation from the clicked button.
        """
        current_text = self.display_var.get()
        
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            new_text = current_text + char
            self.display_var.set(new_text)

    def calculate(self) -> None:
        """Perform the calculation and display the result."""
        try:
            expression = self.display_var.get()
            if not expression:
                return
                
            result = str(eval(expression))  # pylint: disable=eval-used
            self.display_var.set(result)
        except Exception:  # pylint: disable=broad-except
            self.display_var.set('Error')

    def clear_display(self) -> None:
        """Clear the calculator display."""
        self.display_var.set('')


def main() -> None:
    """Main function to start the calculator application."""
    root = tk.Tk()
    calculator = Calculator(root)
    root.mainloop()


if __name__ == '__main__':
    main()
