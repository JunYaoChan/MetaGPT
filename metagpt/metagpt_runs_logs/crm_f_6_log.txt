## main.py

import sqlite3
import streamlit as st

class Customer:
    """Class representing a customer."""
    
    def __init__(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Initialize a Customer object with provided attributes.
        
        Args:
            name (str): The name of the customer.
            birthday (str): The birthday of the customer in YYYY-MM-DD format.
            age (int): The age of the customer.
            sex (str): The sex of the customer (e.g., 'Male', 'Female').
            phone (str): The phone number of the customer.
        """
        self.name = name
        self.birthday = birthday
        self.age = age
        self.sex = sex
        self.phone = phone

class CustomerDB:
    """Class to handle database operations for customers."""
    
    def __init__(self) -> None:
        """Initialize the database connection and create the customer table if it doesn't exist."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()

    def create_table(self) -> None:
        """Create the customer table in the database."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                name TEXT PRIMARY KEY,
                birthday TEXT,
                age INTEGER,
                sex TEXT,
                phone TEXT
            )
        ''')
        self.conn.commit()

    def add_customer(self, customer: Customer) -> None:
        """Add a new customer to the database.
        
        Args:
            customer (Customer): The customer object to add.
        """
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone) VALUES (?, ?, ?, ?, ?)
        ''', (customer.name, customer.birthday, customer.age, customer.sex, customer.phone))
        self.conn.commit()

    def get_customer(self, name: str) -> Customer:
        """Retrieve a customer from the database by name.
        
        Args:
            name (str): The name of the customer to retrieve.
        
        Returns:
            Customer: The customer object retrieved from the database.
        """
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        if row:
            return Customer(*row)
        return None

    def update_customer(self, name: str, customer: Customer) -> None:
        """Update an existing customer in the database.
        
        Args:
            name (str): The name of the customer to update.
            customer (Customer): The updated customer object.
        """
        self.cursor.execute('''
            UPDATE customers SET birthday = ?, age = ?, sex = ?, phone = ? WHERE name = ?
        ''', (customer.birthday, customer.age, customer.sex, customer.phone, name))
        self.conn.commit()

    def delete_customer(self, name: str) -> None:
        """Delete a customer from the database by name.
        
        Args:
            name (str): The name of the customer to delete.
        """
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()

class MainApp:
    """Main application class to run the Streamlit app."""
    
    def __init__(self) -> None:
        """Initialize the main application and database."""
        self.db: CustomerDB = CustomerDB()

    def run(self) -> None:
        """Run the Streamlit application."""
        st.title("Customer Relationship Management (CRM) System")

        menu = ["Add Customer", "Search Customer", "Update Customer", "Delete Customer"]
        choice = st.sidebar.selectbox("Select Operation", menu)

        if choice == "Add Customer":
            st.subheader("Add Customer")
            name = st.text_input("Name")
            birthday = st.text_input("Birthday (YYYY-MM-DD)")
            age = st.number_input("Age", min_value=0, max_value=120)
            sex = st.selectbox("Sex", ["Male", "Female"])
            phone = st.text_input("Phone Number")
            if st.button("Add Customer"):
                new_customer = Customer(name, birthday, age, sex, phone)
                self.db.add_customer(new_customer)
                st.success(f"Customer {name} added successfully!")

        elif choice == "Search Customer":
            st.subheader("Search Customer")
            name = st.text_input("Enter Customer Name")
            if st.button("Search"):
                customer = self.db.get_customer(name)
                if customer:
                    st.write(customer.__dict__)
                else:
                    st.error("Customer not found.")

        elif choice == "Update Customer":
            st.subheader("Update Customer")
            name = st.text_input("Enter Customer Name to Update")
            customer = self.db.get_customer(name)
            if customer:
                birthday = st.text_input("Birthday (YYYY-MM-DD)", value=customer.birthday)
                age = st.number_input("Age", min_value=0, max_value=120, value=customer.age)
                sex = st.selectbox("Sex", ["Male", "Female"], index=["Male", "Female"].index(customer.sex))
                phone = st.text_input("Phone Number", value=customer.phone)
                if st.button("Update Customer"):
                    updated_customer = Customer(name, birthday, age, sex, phone)
                    self.db.update_customer(name, updated_customer)
                    st.success(f"Customer {name} updated successfully!")
            else:
                st.error("Customer not found.")

        elif choice == "Delete Customer":
            st.subheader("Delete Customer")
            name = st.text_input("Enter Customer Name to Delete")
            if st.button("Delete Customer"):
                self.db.delete_customer(name)
                st.success(f"Customer {name} deleted successfully!")

if __name__ == "__main__":
    app = MainApp()
    app.run()
