"""A user interface module implementing common UI components and patterns.

This module provides reusable UI components following Google Python Style Guide.
Components are designed to be modular, extensible, and maintainable.
"""

from typing import Optional, List, Dict, Any, Callable
import abc
import logging

# Configure module-level logger
_LOGGER = logging.getLogger(__name__)


class UIComponent(abc.ABC):
    """Abstract base class for all UI components."""

    def __init__(self, name: str, visible: bool = True):
        """Initialize the UI component.

        Args:
            name: Unique identifier for the component.
            visible: Whether the component is initially visible. Defaults to True.
        """
        self._name = name
        self._visible = visible
        self._parent = None

    @property
    def name(self) -> str:
        """Get the component's name."""
        return self._name

    @property
    def visible(self) -> bool:
        """Get the component's visibility status."""
        return self._visible

    @visible.setter
    def visible(self, value: bool) -> None:
        """Set the component's visibility status."""
        self._visible = value
        self._on_visibility_changed()

    @property
    def parent(self) -> Optional['UIContainer']:
        """Get the parent container of this component."""
        return self._parent

    @parent.setter
    def parent(self, value: Optional['UIContainer']) -> None:
        """Set the parent container of this component."""
        self._parent = value

    @abc.abstractmethod
    def render(self) -> str:
        """Render the component to its string representation.

        Returns:
            The rendered component as a string.
        """
        pass

    def _on_visibility_changed(self) -> None:
        """Callback when visibility changes."""
        _LOGGER.debug("Visibility changed for component %s to %s",
                     self._name, self._visible)


class UIContainer(UIComponent):
    """A container that can hold other UI components."""

    def __init__(self, name: str, visible: bool = True):
        """Initialize the container.

        Args:
            name: Unique identifier for the container.
            visible: Whether the container is initially visible. Defaults to True.
        """
        super().__init__(name, visible)
        self._children: List[UIComponent] = []

    def add_child(self, component: UIComponent) -> None:
        """Add a child component to this container.

        Args:
            component: The component to add.
        """
        if component in self._children:
            _LOGGER.warning("Component %s already exists in container %s",
                           component.name, self.name)
            return

        component.parent = self
        self._children.append(component)
        _LOGGER.debug("Added component %s to container %s",
                     component.name, self.name)

    def remove_child(self, component: UIComponent) -> bool:
        """Remove a child component from this container.

        Args:
            component: The component to remove.

        Returns:
            True if the component was removed, False otherwise.
        """
        try:
            self._children.remove(component)
            component.parent = None
            _LOGGER.debug("Removed component %s from container %s",
                         component.name, self.name)
            return True
        except ValueError:
            _LOGGER.warning("Component %s not found in container %s",
                           component.name, self.name)
            return False

    def get_child(self, name: str) -> Optional[UIComponent]:
        """Get a child component by name.

        Args:
            name: The name of the child component to find.

        Returns:
            The component if found, None otherwise.
        """
        for child in self._children:
            if child.name == name:
                return child
        return None

    def render(self) -> str:
        """Render the container and all visible children.

        Returns:
            The rendered container as a string.
        """
        if not self.visible:
            return ""

        rendered_children = [child.render() for child in self._children
                           if child.visible]
        return f"<div id='{self.name}'>\n" + "\n".join(rendered_children) + "\n</div>"


class UILabel(UIComponent):
    """A simple text label UI component."""

    def __init__(self, name: str, text: str = "", visible: bool = True):
        """Initialize the label.

        Args:
            name: Unique identifier for the label.
            text: The label text. Defaults to empty string.
            visible: Whether the label is initially visible. Defaults to True.
        """
        super().__init__(name, visible)
        self._text = text

    @property
    def text(self) -> str:
        """Get the label text."""
        return self._text

    @text.setter
    def text(self, value: str) -> None:
        """Set the label text."""
        self._text = value

    def render(self) -> str:
        """Render the label.

        Returns:
            The rendered label as a string.
        """
        if not self.visible:
            return ""
        return f"<label id='{self.name}'>{self._text}</label>"


class UIButton(UIComponent):
    """A clickable button UI component."""

    def __init__(self,
                 name: str,
                 text: str = "",
                 visible: bool = True,
                 on_click: Optional[Callable[[], None]] = None):
        """Initialize the button.

        Args:
            name: Unique identifier for the button.
            text: The button text. Defaults to empty string.
            visible: Whether the button is initially visible. Defaults to True.
            on_click: Callback function when button is clicked. Defaults to None.
        """
        super().__init__(name, visible)
        self._text = text
        self._on_click = on_click

    @property
    def text(self) -> str:
        """Get the button text."""
        return self._text

    @text.setter
    def text(self, value: str) -> None:
        """Set the button text."""
        self._text = value

    def click(self) -> None:
        """Simulate a button click."""
        if self._on_click is not None:
            self._on_click()

    def render(self) -> str:
        """Render the button.

        Returns:
            The rendered button as a string.
        """
        if not self.visible:
            return ""
        return (f"<button id='{self.name}' "
                f"onclick='document.getElementById(\"{self.name}\").click()'>"
                f"{self._text}</button>")
