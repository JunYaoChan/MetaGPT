## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, font=("Arial", 24), borderwidth=2, relief="groove")
        self.display.grid(row=0, column=0, columnspan=4, padx=10, pady=10)
        self.buttons = {}
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.delete(0, tk.END)

    def create_buttons(self) -> None:
        """Create buttons for numbers and operations."""
        # Define button layout
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        # Create buttons and place them in the grid
        for i, text in enumerate(button_texts):
            self.buttons[text] = tk.Button(self.root, text=text, font=("Arial", 18),
                                            command=lambda t=text: self.on_button_click(t))
            self.buttons[text].grid(row=(i // 4) + 1, column=i % 4, sticky="nsew", padx=5, pady=5)

        # Configure grid weights for responsive design
        for i in range(4):
            self.root.grid_columnconfigure(i, weight=1)

    def on_button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            self.add_to_display(char)

    def add_to_display(self, char: str) -> None:
        """Add a character to the display."""
        self.display.insert(tk.END, char)

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Calculate the result of the expression in the display."""
        try:
            expression = self.display.get()
            result = eval(expression)  # Use eval to calculate the expression
            self.clear_display()
            self.add_to_display(str(result))
        except Exception as e:
            messagebox.showerror("Error", f"Invalid input: {e}")
            self.clear_display()

    def run(self) -> None:
        """Run the main loop of the calculator."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
