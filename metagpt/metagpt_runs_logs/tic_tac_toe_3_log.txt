import random

class Game:
    """Class to manage the Tic Tac Toe game state and logic."""

    def __init__(self) -> None:
        """Initialize the game board and set the current player."""
        self.board: list[list[str]] = [['' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # X always starts first
        self.winner: str = ''  # No winner at the start

    def make_move(self, row: int, col: int) -> bool:
        """Make a move on the board if the cell is empty.

        Args:
            row (int): The row index for the move.
            col (int): The column index for the move.

        Returns:
            bool: True if the move was successful, False otherwise.
        """
        if self.board[row][col] == '' and self.winner == '':
            self.board[row][col] = self.current_player
            return True
        return False

    def check_winner(self) -> str:
        """Check the current board for a winner.

        Returns:
            str: The winner ('X', 'O') or an empty string if no winner.
        """
        # Check rows, columns, and diagonals for a win
        for i in range(3):
            if self.board[i][0] == self.board[i][1] == self.board[i][2] != '':
                self.winner = self.board[i][0]
                return self.winner
            if self.board[0][i] == self.board[1][i] == self.board[2][i] != '':
                self.winner = self.board[0][i]
                return self.winner

        if self.board[0][0] == self.board[1][1] == self.board[2][2] != '':
            self.winner = self.board[0][0]
            return self.winner
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != '':
            self.winner = self.board[0][2]
            return self.winner

        return ''  # No winner yet

    def reset_game(self) -> None:
        """Reset the game to the initial state."""
        self.board = [['' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.winner = ''

Test cases for main.py in the Tic Tac Toe game implementation.
These tests verify the correct initialization and execution flow of the main game components.
"""

import unittest
from unittest.mock import patch, MagicMock
import pygame
import sys
from tic_tac_toe_3.main import main
from tic_tac_toe_3.game import Game
from tic_tac_toe_3.ai import AI
from tic_tac_toe_3.ui import UI


class TestMain(unittest.TestCase):
    """Test cases for the main game execution flow."""

    @patch('tic_tac_toe_3.main.pygame.init')
    @patch('tic_tac_toe_3.main.Game')
    @patch('tic_tac_toe_3.main.AI')
    @patch('tic_tac_toe_3.main.UI')
    @patch('tic_tac_toe_3.main.pygame.quit')
    def test_main_function_flow(
        self,
        mock_pygame_quit: MagicMock,
        mock_ui: MagicMock,
        mock_ai: MagicMock,
        mock_game: MagicMock,
        mock_pygame_init: MagicMock
    ) -> None:
        """
        Test the complete execution flow of the main function.
        Verifies proper initialization and method calls.
        """
        # Call the main function
        with patch.object(sys, 'argv', ['test_main.py']):
            main()

        # Verify pygame.init was called
        mock_pygame_init.assert_called_once()

        # Verify Game instance was created
        mock_game.assert_called_once()

        # Verify AI instance was created
        mock_ai.assert_called_once()

        # Verify UI instance was created with correct parameters
        mock_ui.assert_called_once_with(mock_game.return_value, mock_ai.return_value)

        # Verify AI difficulty was set
        mock_ai.return_value.set_difficulty.assert_called_once_with('Medium')

        # Verify main_loop was called on UI instance
        mock_ui.return_value.main_loop.assert_called_once()

        # Verify pygame.quit was called
        mock_pygame_quit.assert_called_once()

    @patch('tic_tac_toe_3.main.pygame.init')
    @patch('tic_tac_toe_3.main.Game')
    @patch('tic_tac_toe_3.main.AI')
    @patch('tic_tac_toe_3.main.UI')
    def test_main_as_module(
        self,
        mock_ui: MagicMock,
        mock_ai: MagicMock,
        mock_game: MagicMock,
        mock_pygame_init: MagicMock
    ) -> None:
        """
        Test that main() can be properly imported and executed as a module.
        """
        # Call main directly (simulating import)
        main()

        # Verify basic initialization occurred
        mock_pygame_init.assert_called_once()
        mock_game.assert_called_once()
        mock_ai.assert_called_once()

    @patch('tic_tac_toe_3.main.pygame.init')
    @patch('tic_tac_toe_3.main.Game')
    @patch('tic_tac_toe_3.main.AI')
    @patch('tic_tac_toe_3.main.UI')
    @patch('tic_tac_toe_3.main.pygame.quit')
    def test_pygame_quit_on_exception(
        self,
        mock_pygame_quit: MagicMock,
        mock_ui: MagicMock,
        mock_ai: MagicMock,
        mock_game: MagicMock,
        mock_pygame_init: MagicMock
    ) -> None:
        """
        Test that pygame.quit() is called even if an exception occurs.
        """
        # Set up mock to raise exception
        mock_ui.return_value.main_loop.side_effect = RuntimeError("Test error")

        # Verify pygame.quit is called despite exception
        with self.assertRaises(RuntimeError):
            main()
        mock_pygame_quit.assert_called_once()

    @patch('tic_tac_toe_3.main.pygame.init')
    @patch('tic_tac_toe_3.main.Game')
    @patch('tic_tac_toe_3.main.AI')
    @patch('tic_tac_toe_3.main.UI')
    def test_ai_difficulty_setting(
        self,
        mock_ui: MagicMock,
        mock_ai: MagicMock,
        mock_game: MagicMock,
        mock_pygame_init: MagicMock
    ) -> None:
        """
        Test that AI difficulty is properly set to 'Medium' by default.
        """
        main()
        mock_ai.return_value.set_difficulty.assert_called_once_with('Medium')


if __name__ == '__main__':
    unittest.main()
