## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator with a Tkinter root and UI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='solid')
        self.display.grid(row=0, column=0, columnspan=4)

        self.buttons = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        self.create_buttons()

    def create_buttons(self) -> None:
        """Create buttons for the calculator and place them in the grid."""
        row_val = 1
        col_val = 0
        for button in self.buttons:
            action = lambda x=button: self.add_to_display(x)
            if button == '=':
                action = self.calculate
            elif button == 'C':
                action = self.clear_display

            tk.Button(self.root, text=button, width=5, height=2, command=action).grid(row=row_val, column=col_val)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def add_to_display(self, value: str) -> None:
        """Add a value to the display."""
        current_text = self.display.get()
        new_text = current_text + value
        self.display.delete(0, tk.END)  # Clear the display
        self.display.insert(0, new_text)  # Insert the new text

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Calculate the result of the expression in the display."""
        try:
            result = eval(self.display.get())  # Evaluate the expression
            self.display.delete(0, tk.END)  # Clear the display
            self.display.insert(0, str(result))  # Show the result
        except ZeroDivisionError:
            self.handle_error("Error: Division by zero.")
        except Exception as e:
            self.handle_error(f"Error: {str(e)}")

    def handle_error(self, message: str) -> None:
        """Handle errors by displaying a message box."""
        messagebox.showerror("Calculation Error", message)

    def run(self) -> None:
        """Run the Tkinter main loop."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
