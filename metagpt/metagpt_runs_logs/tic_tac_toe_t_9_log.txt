    def get_move(self, game) -> Tuple[int, int]:
        """Get the AI's move based on current game state and difficulty level.
        
        Args:
            game: The current game instance
            
        Returns:
            Tuple[int, int]: Row and column indices for the move
        """
        if self.difficulty == 'easy':
            return self._get_random_move(game)
        elif self.difficulty == 'medium':
            return self._get_medium_move(game)
        else:  # hard
            return self._get_best_move(game)

    def _get_random_move(self, game) -> Tuple[int, int]:
        """Get a completely random valid move.
        
        Args:
            game: The current game instance
            
        Returns:
            Tuple[int, int]: Random valid move coordinates
        """
        valid_moves = []
        for row in range(3):
            for col in range(3):
                if game.board[row][col] == EMPTY:
                    valid_moves.append((row, col))
        return random.choice(valid_moves)

    def _get_medium_move(self, game) -> Tuple[int, int]:
        """Get a move with some intelligence (blocks wins and takes wins).
        
        Args:
            game: The current game instance
            
        Returns:
            Tuple[int, int]: Move coordinates
        """
        # First check if AI can win immediately
        ai_move = self._find_winning_move(game, game.current_player)
        if ai_move:
            return ai_move

        # Then check if player can win next move and block
        opponent = PLAYER_X if game.current_player == PLAYER_O else PLAYER_O
        block_move = self._find_winning_move(game, opponent)
        if block_move:
            return block_move

        # Otherwise make a random move
        return self._get_random_move(game)

    def _find_winning_move(self, game, player: str) -> Optional[Tuple[int, int]]:
        """Find a winning move for the specified player if one exists.
        
        Args:
            game: The current game instance
            player: The player to check for (PLAYER_X or PLAYER_O)
            
        Returns:
            Optional[Tuple[int, int]]: Winning move if found, else None
        """
        for row in range(3):
            for col in range(3):
                if game.board[row][col] == EMPTY:
                    # Simulate move
                    game.board[row][col] = player
                    winner = game.check_winner()
                    # Undo move
                    game.board[row][col] = EMPTY
                    if winner == player:
                        return (row, col)
        return None

    def _get_best_move(self, game) -> Tuple[int, int]:
        """Get the best possible move using minimax algorithm.
        
        Args:
            game: The current game instance
            
        Returns:
            Tuple[int, int]: Best move coordinates
        """
        best_score = -float('inf')
        best_move = None
        for row in range(3):
            for col in range(3):
                if game.board[row][col] == EMPTY:
                    game.board[row][col] = game.current_player
                    score = self._minimax(game, False, game.current_player)
                    game.board[row][col] = EMPTY
                    if score > best_score:
                        best_score = score
                        best_move = (row, col)
        return best_move or self._get_random_move(game)

    def _minimax(
        self,
        game,
        is_maximizing: bool,
        ai_player: str,
        depth: int = 0,
        alpha: float = -float('inf'),
        beta: float = float('inf')
    ) -> float:
        """Minimax algorithm implementation with alpha-beta pruning.
        
        Args:
            game: The current game instance
            is_maximizing: Whether current player is maximizing or minimizing
            ai_player: The AI's player symbol (PLAYER_X or PLAYER_O)
            depth: Current depth in the game tree
            alpha: Alpha value for pruning
            beta: Beta value for pruning
            
        Returns:
            float: Score for the current board state
        """
        winner = game.check_winner()
        if winner == ai_player:
            return 10 - depth
        elif winner != EMPTY:  # Other player won
            return depth - 10
        elif game.is_board_full():
            return 0

        if is_maximizing:
            best_score = -float('inf')
            for row in range(3):
                for col in range(3):
                    if game.board[row][col] == EMPTY:
                        game.board[row][col] = ai_player
                        score = self._minimax(game, False, ai_player, depth + 1, alpha, beta)
                        game.board[row][col] = EMPTY
                        best_score = max(score, best_score)
                        alpha = max(alpha, best_score)
                        if beta <= alpha:
                            break
            return best_score
        else:
            best_score = float('inf')
            human_player = PLAYER_O if ai_player == PLAYER_X else PLAYER_X
            for row in range(3):
                for col in range(3):
                    if game.board[row][col] == EMPTY:
                        game.board[row][col] = human_player
                        score = self._minimax(game, True, ai_player, depth + 1, alpha, beta)
                        game.board[row][col] = EMPTY
                        best_score = min(score, best_score)
                        beta = min(beta, best_score)
                        if beta <= alpha:
                            break
            return best_score
```

   - Easy: Makes random moves
   - Medium: Blocks opponent wins and takes immediate wins
   - Hard: Uses minimax algorithm with alpha-beta pruning for optimal play

3. Proper error handling for invalid difficulty levels

4. Helper methods for each difficulty level's logic

5. The minimax algorithm implementation with:
   - Depth consideration to prefer faster wins
   - Alpha-beta pruning for optimization
   - Proper scoring for wins/losses/draws

6. All methods are private except the public interface `get_move`

    ## TestDivideOperation

        """Test the divide operation with various inputs."""
        # Test with positive numbers
        self.assertEqual(self.calc.divide(6, 3), 2)
        self.assertEqual(self.calc.divide(1, 2), 0.5)
        self.assertEqual(self.calc.divide(-4, 2), -2)
        self.assertEqual(self.calc.divide(5, 2.5), 2.0)

        # Test division by zero
        with self.assertRaises(ValueError):
            self.calc.divide(5, 0)

    ## TestEdgeCases
    def test_edge_cases(self) -> None:
