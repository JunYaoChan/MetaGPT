## main.py
import tkinter as tk
from tkinter import Entry, Button, StringVar, messagebox

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self):
        """Initialize the calculator with default values."""
        self.current_input: str = ""
        self.previous_input: str = ""
        self.operation: str = ""
        
        # Create the main window
        self.window: tk.Tk = tk.Tk()
        self.window.title("Simple Calculator")
        
        # Create display and buttons
        self.display: Entry = self.create_display()
        self.buttons: dict = self.create_buttons()
        
        # Layout the buttons in a grid
        self.layout_buttons()

    def create_display(self) -> Entry:
        """Create and return the display for the calculator."""
        display_var: StringVar = StringVar()
        display: Entry = Entry(self.window, textvariable=display_var, font=("Arial", 24), bd=10, insertwidth=2, width=14, borderwidth=4)
        display.grid(row=0, column=0, columnspan=4)
        return display

    def create_buttons(self) -> dict:
        """Create and return a dictionary of buttons for the calculator."""
        buttons: dict = {
            '7': (1, 0), '8': (1, 1), '9': (1, 2), '/': (1, 3),
            '4': (2, 0), '5': (2, 1), '6': (2, 2), '*': (2, 3),
            '1': (3, 0), '2': (3, 1), '3': (3, 2), '-': (3, 3),
            '0': (4, 0), 'C': (4, 1), '=': (4, 2), '+': (4, 3)
        }
        return buttons

    def layout_buttons(self):
        """Layout the buttons in the calculator window."""
        for text, (row, col) in self.buttons.items():
            button: Button = Button(self.window, text=text, padx=20, pady=20, font=("Arial", 18),
                                    command=lambda t=text: self.button_click(t))
            button.grid(row=row, column=col)

    def button_click(self, number: str):
        """Handle button click events for numbers and operations."""
        if number == 'C':
            self.clear()
        elif number == '=':
            self.calculate()
        else:
            self.current_input += number
            self.update_display()

    def operation_click(self, operation: str):
        """Store the current input and the operation to be performed."""
        if self.current_input:
            self.previous_input = self.current_input
            self.current_input = ""
            self.operation = operation

    def clear(self):
        """Clear the current input and reset the calculator."""
        self.current_input = ""
        self.previous_input = ""
        self.operation = ""
        self.update_display()

    def calculate(self):
        """Perform the calculation based on the current and previous inputs."""
        if self.previous_input and self.current_input:
            try:
                if self.operation == '+':
                    result = float(self.previous_input) + float(self.current_input)
                elif self.operation == '-':
                    result = float(self.previous_input) - float(self.current_input)
                elif self.operation == '*':
                    result = float(self.previous_input) * float(self.current_input)
                elif self.operation == '/':
                    if float(self.current_input) == 0:
                        raise ZeroDivisionError("Cannot divide by zero.")
                    result = float(self.previous_input) / float(self.current_input)
                else:
                    return
                self.current_input = str(result)
                self.previous_input = ""
                self.operation = ""
                self.update_display()
            except ZeroDivisionError as e:
                messagebox.showerror("Error", str(e))
                self.clear()

    def update_display(self):
        """Update the calculator display with the current input."""
        self.display.delete(0, tk.END)
        self.display.insert(0, self.current_input)

    def run(self):
        """Run the calculator application."""
        self.window.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
