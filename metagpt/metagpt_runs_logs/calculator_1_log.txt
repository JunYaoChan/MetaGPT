import tkinter as tk

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='ridge')
        self.display.grid(row=0, column=0, columnspan=4)
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.delete(0, tk.END)  # Clear the display

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        # Define button layout
        buttons = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]

        # Create buttons and place them in the grid
        row_val = 1
        col_val = 0
        for button in buttons:
            action = lambda x=button: self.add_to_display(x)
            if button == '=':
                action = self.calculate
            elif button == 'C':
                action = self.clear_display

            tk.Button(self.root, text=button, width=5, height=2, command=action).grid(row=row_val, column=col_val)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def add_to_display(self, char: str) -> None:
        """Add a character to the display."""
        current_text = self.display.get()
        self.display.delete(0, tk.END)  # Clear the display
        self.display.insert(0, current_text + char)  # Append new character

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)  # Clear the display

    def calculate(self) -> None:
        """Calculate the result of the expression in the display."""
        try:
            result = eval(self.display.get())  # Evaluate the expression
            self.clear_display()  # Clear the display before showing result
            self.display.insert(0, str(result))  # Show result
        except Exception as e:
            self.clear_display()  # Clear the display on error
            self.display.insert(0, "Error")  # Show error message

    def run(self) -> None:
        """Run the main loop of the calculator."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()  # Create a Calculator instance
    calculator.run()  # Run the calculator
