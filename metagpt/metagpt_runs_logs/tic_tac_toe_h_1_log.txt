```"""
Tic Tac Toe game core logic implementation.
Handles game state, move validation, win checking, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game rules."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.board: List[List[str]] = [['' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3):
            return False
            
        if self.board[row][col] != '':
            return False
            
        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self._switch_player()
        return True

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if the current move resulted in a win or draw.
        Updates game_over and winner accordingly.
        """
        winner = self.check_winner()
        if winner:
            self.game_over = True
            self.winner = winner
        elif self.is_board_full():
            self.game_over = True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.
        
        Returns:
            Optional[str]: 'X' or 'O' if there's a winner, None otherwise
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] != '':
                return row[0]

        # Check columns
        for col in range(3):
            if self.board[0][col] == self.board[1][col] == self.board[2][col] != '':
                return self.board[0][col]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] != '':
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != '':
            return self.board[0][2]

        return None

    def is_board_full(self) -> bool:
        """
        Check if the board is completely filled.
        
        Returns:
            bool: True if board is full, False otherwise
        """
        return all(cell != '' for row in self.board for cell in row)

    def reset_game(self) -> None:
        """Reset the game to initial state."""
        self.board = [['' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = None

    def get_available_moves(self) -> List[Tuple[int, int]]:
        """
        Get all available moves on the current board.
        
        Returns:

```"""
Tic Tac Toe game core logic implementation.
Handles game state, move validation, win checking, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game rules."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.board: List[List[str]] = [['' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3):
            return False
            
        if self.board[row][col] != '':
            return False
            
        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self._switch_player()
        return True

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if the current move resulted in a win or draw.
        Updates game_over and winner accordingly.
        """
        winner = self.check_winner()
        if winner:
            self.game_over = True
            self.winner = winner
        elif self.is_board_full():
            self.game_over = True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.
        
        Returns:
            Optional[str]: 'X' or 'O' if there's a winner, None otherwise
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] != '':
                return row[0]

        # Check columns
        for col in range(3):
            if self.board[0][col] == self.board[1][col] == self.board[2][col] != '':
                return self.board[0][col]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] != '':
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != '':
            return self.board[0][2]

        return None

    def is_board_full(self) -> bool:
        """
        Check if the board is completely filled.
        
        Returns:
            bool: True if board is full, False otherwise
        """
        return all(cell != '' for row in self.board for cell in row)

    def reset_game(self) -> None:
        """Reset the game to initial state."""
        self.board = [['' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = None

    def get_available_moves(self) -> List[Tuple[int, int]]:
        """
        Get all available moves on the current board.
        
        Returns:
            List[Tuple[int, int]]: List of (row, col) tuples of empty cells
        """
        return [(i, j) for i in range(3) for j in range(3) if self.board[i][j] == '']

    def get_board_state(self) -> List[List[str]]:
        """
        Get a deep copy of the current board state.
        
        Returns:
            List[List[str]]: Copy of the current board
        """
        return [row.copy() for row in self.board]

```"""AI module for Tic Tac Toe game, implementing different difficulty levels."""

import random
from typing import Tuple
from game import Game


class AI:
    """AI player for Tic Tac Toe with configurable difficulty levels.
    
    Attributes:
        difficulty: The difficulty level of the AI ('easy', 'medium', 'hard').
    """
    
    def __init__(self, difficulty: str = 'medium') -> None:
        """Initialize AI with specified difficulty level.
        
        Args:
            difficulty: The difficulty level ('easy', 'medium', 'hard').
                        Defaults to 'medium'.
        """
        self.difficulty = difficulty.lower()
        if self.difficulty not in ('easy', 'medium', 'hard'):
            raise ValueError("Difficulty must be 'easy', 'medium' or 'hard'")

    def get_move(self, game: Game) -> Tuple[int, int]:
        """Get the AI's next move based on current game state.
        
        Args:
            game: The current game instance.
            
        Returns:
            A tuple (row, col) representing the AI's move.
        """
        if self.difficulty == 'easy':
            return self._get_random_move(game)
        elif self.difficulty == 'medium':
            return self._get_medium_move(game)
        else:  # hard
            return self._get_hard_move(game)

    def _get_random_move(self, game: Game) -> Tuple[int, int]:
        """Get a random valid move (easy difficulty).
        
        Args:
            game: The current game instance.
            
        Returns:
            A tuple (row, col) representing a random valid move.
        """
        empty_cells = [
            (i, j)
            for i in range(3)
            for j in range(3)
            if game.board[i][j] == ''
        ]
        return random.choice(empty_cells)

    def _get_medium_move(self, game: Game) -> Tuple[int, int]:
        """Get a move with basic strategy (medium difficulty).
        
        Strategy:
        1. Try to win if possible
        2. Block opponent's winning move
        3. Otherwise make random move
        
        Args:
            game: The current game instance.
            
        Returns:
            A tuple (row, col) representing the AI's move.
        """
        # Check for winning move
        for i in range(3):
            for j in range(3):
                if game.board[i][j] == '':
                    game.board[i][j] = 'O'  # AI is always 'O'
                    if game.check_winner() == 'O':
                        game.board[i][j] = ''  # Undo test move
                        return (i, j)
                    game.board[i][j] = ''  # Undo test move

        # Check for blocking move
        for i in range(3):
            for j in range(3):
                if game.board[i][j] == '':
                    game.board[i][j] = 'X'  # Player is always 'X'
                    if game.check_winner() == 'X':
                        game.board[i][j] = ''  # Undo test move
                        return (i, j)
                    game.board[i][j] = ''  # Undo test move

        # Default to random move
        return self._get_random_move(game)

    def _get_hard_move(self, game: Game) -> Tuple[int, int]:
        """Get the best move using minimax algorithm (hard difficulty).
        
        Args:
            game: The current game instance.
            
        Returns:
            A tuple (row, col) representing the optimal move.
        """
        best_score = -float('inf')
        best_move = (-1, -1)
        
        for i in range(3):
            for j in range(3):
                if game.board[i][j] == '':
                    game.board[i][j] = 'O'
                    score = self._minimax(game, False)
                    game.board[i][j] = ''
                    
                    if score > best_score:
                        best_score = score
                        best_move = (i, j)
        
        return best_move

    def _minimax(self, game: Game, is_maximizing: bool) -> int:
        """Recursive minimax algorithm to evaluate board positions.
        
        Args:
            game: The current game instance.
            is_maximizing: Whether current player is maximizing (AI) or not.
            
        Returns:
            The score of the current board position.
        """
        winner = game.check_winner()
        if winner == 'O':
            return 1
        elif winner == 'X':
            return -1
        elif game.is_board_full():
            return 0

        if is_maximizing:
            best_score = -float('inf')
            for i in range(3):
                for j in range(3):
                    if game.board[i][j] == '':
                        game.board[i][j] = 'O'
                        score = self._minimax(game, False)
                        game.board[i][j] = ''
                        best_score = max(score, best_score)
            return best_score
        else:
            best_score = float('inf')
            for i in range(3):
                for j in range(3):
                    if game.board[i][j] == '':
                        game.board[i][j] = 'X'
                        score = self._minimax(game, True)
                        game.board[i][j] = ''
                        best_score = min(score, best_score)
            return best_score
