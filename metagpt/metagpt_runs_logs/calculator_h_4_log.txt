## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class using Tkinter for GUI."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = ""
        self.first_operand = None
        self.operation = None
        self.create_display()
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display_var = tk.StringVar()
        self.display_entry = tk.Entry(self.root, textvariable=self.display_var, font=("Arial", 24), justify='right')
        self.display_entry.grid(row=0, column=0, columnspan=4, sticky="nsew")

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        buttons = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        row_val = 1
        col_val = 0
        
        for button in buttons:
            button_command = lambda char=button: self.button_click(char)
            tk.Button(self.root, text=button, command=button_command, font=("Arial", 18), width=5, height=2).grid(row=row_val, column=col_val, sticky="nsew")
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

        # Configure grid weights for responsive design
        for i in range(4):
            self.root.grid_columnconfigure(i, weight=1)
        for i in range(5):
            self.root.grid_rowconfigure(i, weight=1)

    def button_click(self, char: str) -> None:
        """Handle button clicks and update the display."""
        if char == 'C':
            self.clear()
        elif char == '=':
            self.calculate()
        else:
            self.display += char
            self.display_var.set(self.display)

    def calculate(self) -> None:
        """Perform the calculation based on the current input."""
        try:
            if self.display:
                # Evaluate the expression safely
                result = eval(self.display)
                self.display_var.set(result)
                self.display = str(result)  # Update display for further calculations
            else:
                messagebox.showerror("Error", "No input to calculate.")
        except ZeroDivisionError:
            messagebox.showerror("Error", "Division by zero is not allowed.")
            self.clear()
        except Exception as e:
            messagebox.showerror("Error", f"An error occurred: {str(e)}")
            self.clear()

    def clear(self) -> None:
        """Clear the display and reset stored values."""
        self.display = ""
        self.display_var.set(self.display)

    def run(self) -> None:
        """Run the main loop of the Tkinter application."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
