## main.py
import tkinter as tk

class Calculator:
    """A simple calculator class using Tkinter for GUI."""

    def __init__(self):
        """Initialize the calculator and create the UI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='ridge')
        self.display.grid(row=0, column=0, columnspan=4)

        self.buttons = {}
        self.create_buttons()
        self.create_display()

    def create_display(self):
        """Create the display area for the calculator."""
        self.display.insert(0, "0")  # Set default value to 0

    def create_buttons(self):
        """Create buttons for the calculator."""
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            '0', 'C', '=', '+'
        ]

        row_val = 1
        col_val = 0

        for text in button_texts:
            self.buttons[text] = tk.Button(self.root, text=text, width=5, height=2,
                                            command=lambda t=text: self.button_click(t))
            self.buttons[text].grid(row=row_val, column=col_val)

            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def button_click(self, char: str):
        """Handle button click events."""
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            current_text = self.display.get()
            if current_text == "0":  # Replace default value if it's still 0
                self.display.delete(0, tk.END)
            self.display.insert(tk.END, char)

    def clear_display(self):
        """Clear the display area."""
        self.display.delete(0, tk.END)
        self.display.insert(0, "0")  # Reset to default value

    def calculate(self):
        """Perform the calculation based on the current display content."""
        try:
            result = eval(self.display.get())  # Evaluate the expression
            self.display.delete(0, tk.END)
            self.display.insert(0, str(result))  # Show the result
        except Exception as e:
            self.display.delete(0, tk.END)
            self.display.insert(0, "Error")  # Show error if calculation fails

    def run(self):
        """Run the main loop of the Tkinter application."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
