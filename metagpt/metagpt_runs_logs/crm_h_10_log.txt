## main.py

import sqlite3
import streamlit as st

class CustomerCRUD:
    """Class to handle CRUD operations for customer data in SQLite database."""

    def __init__(self) -> None:
        """Initialize the database connection and cursor."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()

    def create_table(self) -> None:
        """Create the customer table if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                birthday TEXT NOT NULL,
                age INTEGER NOT NULL,
                sex TEXT NOT NULL,
                phone TEXT NOT NULL
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> bool:
        """Add a new customer to the database.

        Args:
            name (str): The name of the customer.
            birthday (str): The birthday of the customer.
            age (int): The age of the customer.
            sex (str): The sex of the customer.
            phone (str): The phone number of the customer.

        Returns:
            bool: True if the customer was added successfully, False otherwise.
        """
        try:
            self.cursor.execute('''
                INSERT INTO customers (name, birthday, age, sex, phone)
                VALUES (?, ?, ?, ?, ?)
            ''', (name, birthday, age, sex, phone))
            self.conn.commit()
            return True
        except Exception as e:
            st.error(f"Error adding customer: {e}")
            return False

    def get_customer(self, name: str) -> dict:
        """Retrieve a customer's data by name.

        Args:
            name (str): The name of the customer.

        Returns:
            dict: A dictionary containing customer data or an empty dict if not found.
        """
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        if row:
            return {
                'id': row[0],
                'name': row[1],
                'birthday': row[2],
                'age': row[3],
                'sex': row[4],
                'phone': row[5]
            }
        return {}

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> bool:
        """Update an existing customer's data.

        Args:
            name (str): The name of the customer to update.
            birthday (str): The new birthday of the customer.
            age (int): The new age of the customer.
            sex (str): The new sex of the customer.
            phone (str): The new phone number of the customer.

        Returns:
            bool: True if the customer was updated successfully, False otherwise.
        """
        try:
            self.cursor.execute('''
                UPDATE customers
                SET birthday = ?, age = ?, sex = ?, phone = ?
                WHERE name = ?
            ''', (birthday, age, sex, phone, name))
            self.conn.commit()
            return True
        except Exception as e:
            st.error(f"Error updating customer: {e}")
            return False

    def delete_customer(self, name: str) -> bool:
        """Delete a customer from the database.

        Args:
            name (str): The name of the customer to delete.

        Returns:
            bool: True if the customer was deleted successfully, False otherwise.
        """
        try:
            self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
            self.conn.commit()
            return True
        except Exception as e:
            st.error(f"Error deleting customer: {e}")
            return False


class StreamlitUI:
    """Class to manage the Streamlit user interface for the CRM system."""

    def __init__(self, crud: CustomerCRUD) -> None:
        """Initialize the Streamlit UI with a CustomerCRUD instance.

        Args:
            crud (CustomerCRUD): An instance of the CustomerCRUD class.
        """
        self.crud: CustomerCRUD = crud

    def show_add_form(self) -> None:
        """Display the form to add a new customer."""
        st.header("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")

        if st.button("Add Customer"):
            if self.crud.add_customer(name, birthday, age, sex, phone):
                st.success("Customer added successfully!")

    def show_search_form(self) -> None:
        """Display the form to search for a customer."""
        st.header("Search Customer")
        name = st.text_input("Enter customer name")

        if st.button("Search"):
            customer = self.crud.get_customer(name)
            if customer:
                st.write(customer)
            else:
                st.warning("Customer not found.")

    def show_update_form(self) -> None:
        """Display the form to update an existing customer."""
        st.header("Update Customer")
        name = st.text_input("Enter customer name to update")
        birthday = st.text_input("New Birthday (YYYY-MM-DD)")
        age = st.number_input("New Age", min_value=0)
        sex = st.selectbox("New Sex", ["Male", "Female", "Other"])
        phone = st.text_input("New Phone")

        if st.button("Update Customer"):
            if self.crud.update_customer(name, birthday, age, sex, phone):
                st.success("Customer updated successfully!")

    def show_delete_form(self) -> None:
        """Display the form to delete a customer."""
        st.header("Delete Customer")
        name = st.text_input("Enter customer name to delete")

        if st.button("Delete Customer"):
            if self.crud.delete_customer(name):
                st.success("Customer deleted successfully!")

    def run(self) -> None:
        """Run the Streamlit application."""
        st.title("Customer Relationship Management (CRM) System")
        operation = st.selectbox("Select Operation", ["Add", "Search", "Update", "Delete"])

        if operation == "Add":
            self.show_add_form()
        elif operation == "Search":
            self.show_search_form()
        elif operation == "Update":
            self.show_update_form()
        elif operation == "Delete":
            self.show_delete_form()


if __name__ == "__main__":
    # Initialize the CustomerCRUD and StreamlitUI classes and run the application
    crud = CustomerCRUD()
    ui = StreamlitUI(crud)
    ui.run()
