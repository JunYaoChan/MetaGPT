## main.py

import sqlite3
import streamlit as st

class CustomerManager:
    """Class to manage customer data in the SQLite database."""

    def __init__(self, db_name: str = "customers.db") -> None:
        """Initialize the CustomerManager with the database name and create the table if it doesn't exist."""
        self.db_name: str = db_name
        self.conn: sqlite3.Connection = sqlite3.connect(self.db_name)
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()

    def create_table(self) -> None:
        """Create the customer table if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                birthday TEXT NOT NULL,
                age INTEGER NOT NULL,
                sex TEXT NOT NULL,
                phone TEXT NOT NULL
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Add a new customer to the database."""
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone)
            VALUES (?, ?, ?, ?, ?)
        ''', (name, birthday, age, sex, phone))
        self.conn.commit()

    def get_customer(self, name: str) -> dict:
        """Retrieve customer data by name."""
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        if row:
            return {
                "id": row[0],
                "name": row[1],
                "birthday": row[2],
                "age": row[3],
                "sex": row[4],
                "phone": row[5]
            }
        return {}

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Update existing customer data."""
        self.cursor.execute('''
            UPDATE customers
            SET birthday = ?, age = ?, sex = ?, phone = ?
            WHERE name = ?
        ''', (birthday, age, sex, phone, name))
        self.conn.commit()

    def delete_customer(self, name: str) -> None:
        """Delete a customer from the database."""
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()

    def close_connection(self) -> None:
        """Close the database connection."""
        self.conn.close()


class StreamlitUI:
    """Class to render the Streamlit user interface."""

    def __init__(self, customer_manager: CustomerManager) -> None:
        """Initialize the StreamlitUI with a CustomerManager instance."""
        self.customer_manager: CustomerManager = customer_manager

    def render(self) -> None:
        """Render the Streamlit application interface."""
        st.title("Customer Relationship Management (CRM) System")

        operation = st.selectbox("Select Operation", ["Add Customer", "View Customer", "Update Customer", "Delete Customer"])

        if operation == "Add Customer":
            self.show_add_form()
        elif operation == "View Customer":
            self.show_view_form()
        elif operation == "Update Customer":
            self.show_update_form()
        elif operation == "Delete Customer":
            self.show_delete_form()

    def show_add_form(self) -> None:
        """Display the form to add a new customer."""
        st.subheader("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")

        if st.button("Add Customer"):
            self.customer_manager.add_customer(name, birthday, age, sex, phone)
            st.success("Customer added successfully!")

    def show_view_form(self) -> None:
        """Display the form to view a customer's details."""
        st.subheader("View Customer")
        name = st.text_input("Enter Customer Name")

        if st.button("View Customer"):
            customer = self.customer_manager.get_customer(name)
            if customer:
                st.write(customer)
            else:
                st.error("Customer not found.")

    def show_update_form(self) -> None:
        """Display the form to update an existing customer's details."""
        st.subheader("Update Customer")
        name = st.text_input("Enter Customer Name to Update")
        birthday = st.text_input("New Birthday (YYYY-MM-DD)")
        age = st.number_input("New Age", min_value=0)
        sex = st.selectbox("New Sex", ["Male", "Female", "Other"])
        phone = st.text_input("New Phone")

        if st.button("Update Customer"):
            self.customer_manager.update_customer(name, birthday, age, sex, phone)
            st.success("Customer updated successfully!")

    def show_delete_form(self) -> None:
        """Display the form to delete a customer."""
        st.subheader("Delete Customer")
        name = st.text_input("Enter Customer Name to Delete")

        if st.button("Delete Customer"):
            self.customer_manager.delete_customer(name)
            st.success("Customer deleted successfully!")


if __name__ == "__main__":
    # Initialize the CustomerManager and StreamlitUI
    customer_manager = CustomerManager()
    ui = StreamlitUI(customer_manager)
    ui.render()
    customer_manager.close_connection()
