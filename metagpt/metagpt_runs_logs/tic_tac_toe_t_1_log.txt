        if level.lower() in valid_levels:
            self._difficulty = level.lower()
        else:
            raise ValueError(f"Invalid difficulty level. Must be one of {valid_levels}")
    
    def get_move(self, game: GameLogic) -> Tuple[int, int]:
        """Get the AI's move based on current game state.
        
        Args:
            game: Current game state
            
        Returns:
            Tuple of (row, col) representing the AI's move
        """

            if random.random() < 0.5:
                return self._get_random_move(board)
            return self._get_optimal_move(board, player)
        else:  # hard
            return self._get_optimal_move(board, player)
    
    def _get_random_move(self, board: List[List[str]]) -> Tuple[int, int]:

        Args:
            board: Current game board

            Random valid move as (row, col)

            (i, j) 
            for i in range(3) 
            for j in range(3) 
            if board[i][j] == ''
        ]
        return random.choice(empty_positions)

        """Get the optimal move using minimax algorithm.
        
        Args:
            board: Current game board
            player: Current player ('X' or 'O')

            Optimal move as (row, col)

                    board[i][j] = player
                    score = self._minimax(board, 0, False, player)
                    board[i][j] = ''
                    if score > best_score:
                        best_score = score
                        best_move = (i, j)

```"""
Tic-Tac-Toe game logic implementation.
Handles board state, move validation, win/draw checking, and game reset.
"""

from typing import List, Optional, Tuple


class GameLogic:
    """Core game logic for Tic-Tac-Toe."""

    def __init__(self) -> None:
        """Initialize game with empty board and starting player."""
        self._board: List[List[str]] = [[" " for _ in range(3)] for _ in range(3)]
        self._current_player: str = "X"
        self._game_over: bool = False

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self._game_over or not self._is_valid_position(row, col):
            return False

        if self._board[row][col] != " ":
            return False

        self._board[row][col] = self._current_player
        self._switch_player()
        return True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.
        
        Returns:
            Optional[str]: 'X' or 'O' if winner found, None otherwise
        """
        # Check rows
        for row in self._board:
            if row[0] != " " and row[0] == row[1] == row[2]:
                self._game_over = True
                return row[0]

        # Check columns
        for col in range(3):
            if (self._board[0][col] != " " and 
                self._board[0][col] == self._board[1][col] == self._board[2][col]):
                self._game_over = True
                return self._board[0][col]

        # Check diagonals
        if (self._board[0][0] != " " and 
            self._board[0][0] == self._board[1][1] == self._board[2][2]):
            self._game_over = True
            return self._board[0][0]

        if (self._board[0][2] != " " and 
            self._board[0][2] == self._board[1][1] == self._board[2][0]):
            self._game_over = True
            return self._board[0][2]

        return None

    def is_draw(self) -> bool:
        """
        Check if the game is a draw (no winner and board full).
        
        Returns:
            bool: True if game is a draw, False otherwise
        """
        if self.check_winner() is not None:
            return False

        for row in self._board:
            if " " in row:
                return False

        self._game_over = True
        return True

    def reset_board(self) -> None:
        """Reset the game board and state."""
        self._board = [[" " for _ in range(3)] for _ in range(3)]
        self._current_player = "X"
        self._game_over = False

    def get_board(self) -> List[List[str]]:
        """
        Get current board state.
        
        Returns:
            List[List[str]]: Copy of current board state
        """
        return [row.copy() for row in self._board]

    def get_current_player(self) -> str:
        """
        Get current player ('X' or 'O').
        
        Returns:
            str: Current player
        """
        return self._current_player

    def is_game_over(self) -> bool:
        """
        Check if game is over.
        
        Returns:
            bool: True if game is over, False otherwise
        """
        return self._game_over

    def _switch_player(self) -> None:
        """Switch current player between X and O."""
        self._current_player = "O" if self._current_player == "X" else "X"

    def _is_valid_position(self, row: int, col: int) -> bool:
        """
        Check if position is valid (within board bounds).
        
        Args:
            row: Row index
            col: Column index
            
        Returns:
            bool: True if position is valid, False otherwise
        """
        return 0 <= row < 3 and 0 <= col < 3
```

```"""
Tic-Tac-Toe game logic implementation.
Handles board state, move validation, win/draw checking, and game reset.
"""

from typing import List, Optional, Tuple


class GameLogic:
    """Core game logic for Tic-Tac-Toe."""

    def __init__(self) -> None:
        """Initialize game with empty board and starting player."""
        self._board: List[List[str]] = [[" " for _ in range(3)] for _ in range(3)]
        self._current_player: str = "X"
        self._game_over: bool = False

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self._game_over or not self._is_valid_position(row, col):
            return False

        if self._board[row][col] != " ":
            return False

        self._board[row][col] = self._current_player
        self._switch_player()
        return True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.
        
        Returns:
            Optional[str]: 'X' or 'O' if winner found, None otherwise
        """
        # Check rows
        for row in self._board:
            if row[0] != " " and row[0] == row[1] == row[2]:
                self._game_over = True
                return row[0]

        # Check columns
        for col in range(3):
            if (self._board[0][col] != " " and 
                self._board[0][col] == self._board[1][col] == self._board[2][col]):
                self._game_over = True
                return self._board[0][col]

        # Check diagonals
        if (self._board[0][0] != " " and 
            self._board[0][0] == self._board[1][1] == self._board[2][2]):
            self._game_over = True
            return self._board[0][0]

        if (self._board[0][2] != " " and 
            self._board[0][2] == self._board[1][1] == self._board[2][0]):
            self._game_over = True
            return self._board[0][2]

        return None

    def is_draw(self) -> bool:
        """
        Check if the game is a draw (no winner and board full).
        
        Returns:
            bool: True if game is a draw, False otherwise
        """
        if self.check_winner() is not None:
            return False

        for row in self._board:
            if " " in row:
                return False

        self._game_over = True
        return True

    def reset_board(self) -> None:
        """Reset the game board and state."""
        self._board = [[" " for _ in range(3)] for _ in range(3)]
        self._current_player = "X"
        self._game_over = False

    def get_board(self) -> List[List[str]]:
        """
        Get current board state.
        
        Returns:
            List[List[str]]: Copy of current board state
        """
        return [row.copy() for row in self._board]

    def get_current_player(self) -> str:
        """
        Get current player ('X' or 'O').
        
        Returns:
            str: Current player
        """
        return self._current_player

    def is_game_over(self) -> bool:
        """
        Check if game is over.
        
        Returns:
            bool: True if game is over, False otherwise
        """
        return self._game_over

    def _switch_player(self) -> None:
        """Switch current player between X and O."""
        self._current_player = "O" if self._current_player == "X" else "X"

    def _is_valid_position(self, row: int, col: int) -> bool:
        """
        Check if position is valid (within board bounds).
        
        Args:
            row: Row index
            col: Column index
            
        Returns:
            bool: True if position is valid, False otherwise
        """
        return 0 <= row < 3 and 0 <= col < 3
```

```"""AI module implementing minimax algorithm for Tic-Tac-Toe game."""
    def _minimax(self, board: List[List[str]], depth: int, is_maximizing: bool, 
                player: str) -> int:
        """Minimax algorithm implementation with alpha-beta pruning.
        
        Args:
            board: Current game board
            depth: Current depth in game tree
            is_maximizing: Whether current player is maximizing
            player: The AI player ('X' or 'O')
            
        Returns:
            Score of the current board state
        """
        opponent = 'X' if player == 'O' else 'O'
        
        # Check terminal states
        winner = self._check_winner(board)
        if winner == player:
            return 10 - depth
        elif winner == opponent:
            return depth - 10
        elif self._is_board_full(board):
            return 0
        
        if is_maximizing:
            best_score = -math.inf
            for i in range(3):
                for j in range(3):
                    if board[i][j] == '':
                        board[i][j] = player
                        score = self._minimax(board, depth + 1, False, player)
                        board[i][j] = ''
                        best_score = max(score, best_score)
            return best_score
        else:
            best_score = math.inf
            for i in range(3):
                for j in range(3):
                    if board[i][j] == '':
                        board[i][j] = opponent
                        score = self._minimax(board, depth + 1, True, player)
                        board[i][j] = ''
                        best_score = min(score, best_score)
            return best_score
    
    def _check_winner(self, board: List[List[str]]) -> Optional[str]:
        """Check if there's a winner on the board.
        
        Args:
            board: Current game board
            
        Returns:
            'X' or 'O' if winner found, None otherwise
        """
        # Check rows
        for row in board:
            if row[0] == row[1] == row[2] != '':
                return row[0]
        
        # Check columns
        for col in range(3):
            if board[0][col] == board[1][col] == board[2][col] != '':
                return board[0][col]
        
        # Check diagonals
        if board[0][0] == board[1][1] == board[2][2] != '':
            return board[0][0]
        if board[0][2] == board[1][1] == board[2][0] != '':
            return board[0][2]
        
        return None
    
    def _is_board_full(self, board: List[List[str]]) -> bool:
        """Check if the board is full (draw).

            board: Current game board
            
        Returns:
            True if board is full, False otherwise
        """
        return all(cell != '' for row in board for cell in row)
```

```"""Score tracking and persistence for Tic-Tac-Toe game using JSON."""

import json
import os
from typing import Dict, Optional
from pathlib import Path


class ScoreTracker:
    """Handles score persistence and management using JSON file storage."""

    DEFAULT_SCORES = {
        'X': 0,
        'O': 0,
        'draw': 0
    }

    def __init__(self, file_path: str = 'scores.json') -> None:
        """Initialize score tracker with specified file path.
        
        Args:
            file_path: Path to JSON file for score storage
        """
        self._file_path = Path(file_path)
        self._scores: Dict[str, int] = self.DEFAULT_SCORES.copy()
        self.load_scores()

    def load_scores(self) -> None:
        """Load scores from JSON file. Creates file with default scores if not exists."""
        try:
            if self._file_path.exists():
                with open(self._file_path, 'r', encoding='utf-8') as f:
                    self._scores = json.load(f)
                    # Validate loaded scores
                    for key in self.DEFAULT_SCORES:
                        if key not in self._scores:
                            self._scores[key] = 0
            else:
                self._scores = self.DEFAULT_SCORES.copy()
                self.save_scores()
        except (json.JSONDecodeError, IOError):
            # If file is corrupted, reset to defaults
            self._scores = self.DEFAULT_SCORES.copy()
            self.save_scores()

    def save_scores(self) -> None:
        """Save current scores to JSON file."""
        try:
            with open(self._file_path, 'w', encoding='utf-8') as f:
                json.dump(self._scores, f, indent=4)
        except IOError:
            # Silently fail if unable to save
            pass

    def update_score(self, winner: Optional[str]) -> None:
        """Update scores based on game outcome.
        
        Args:
            winner: 'X', 'O', or None for draw
        """
        if winner is None:
            self._scores['draw'] += 1
        elif winner in ('X', 'O'):
            self._scores[winner] += 1
        self.save_scores()

    def get_scores(self) -> Dict[str, int]:
        """Get current scores.
        
        Returns:
            Dictionary with keys 'X', 'O', 'draw' and their counts
        """
        return self._scores.copy()

    def reset_scores(self) -> None:
        """Reset all scores to zero and save."""
        self._scores = self.DEFAULT_SCORES.copy()
        self.save_scores()
```

"""
Tic-Tac-Toe game logic implementation.
Handles board state, move validation, win/draw checking, and game reset.
"""

from typing import List, Optional, Tuple


class GameLogic:
    """Core game logic for Tic-Tac-Toe."""

    def __init__(self) -> None:
        """Initialize game with empty board and starting player."""
        self._board: List[List[str]] = [[" " for _ in range(3)] for _ in range(3)]
        self._current_player: str = "X"
        self._game_over: bool = False

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self._game_over or not self._is_valid_position(row, col):
            return False

        if self._board[row][col] != " ":
            return False

        self._board[row][col] = self._current_player
        self._switch_player()
        return True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.
        
        Returns:
            Optional[str]: 'X' or 'O' if winner found, None otherwise
        """
        # Check rows
        for row in self._board:
            if row[0] != " " and row[0] == row[1] == row[2]:
                self._game_over = True
                return row[0]

        # Check columns
        for col in range(3):
            if (self._board[0][col] != " " and 
                self._board[0][col] == self._board[1][col] == self._board[2][col]):
                self._game_over = True
                return self._board[0][col]

        # Check diagonals
        if (self._board[0][0] != " " and 
            self._board[0][0] == self._board[1][1] == self._board[2][2]):
            self._game_over = True
            return self._board[0][0]

        if (self._board[0][2] != " " and 
            self._board[0][2] == self._board[1][1] == self._board[2][0]):
            self._game_over = True
            return self._board[0][2]

        return None

    def is_draw(self) -> bool:
        """
        Check if the game is a draw (no winner and board full).
        
        Returns:
            bool: True if game is a draw, False otherwise
        """
        if self.check_winner() is not None:
            return False

        for row in self._board:
            if " " in row:
                return False

        self._game_over = True
        return True

    def reset_board(self) -> None:
        """Reset the game board and state."""
        self._board = [[" " for _ in range(3)] for _ in range(3)]
        self._current_player = "X"
        self._game_over = False

        """
        Get current board state.
        
        Returns:
            List[List[str]]: Copy of current board state

```python
"""Score tracking and persistence for Tic-Tac-Toe game using JSON."""

import json
import os
from typing import Dict, Optional
from pathlib import Path


class ScoreTracker:
    """Handles score persistence and management using JSON file storage."""

    DEFAULT_SCORES = {
        'X': 0,
        'O': 0,
        'draw': 0
    }

    def __init__(self, file_path: str = 'scores.json') -> None:
        """Initialize score tracker with specified file path.
        
        Args:
            file_path: Path to JSON file for score storage
        """
        self._file_path = Path(file_path)
        self._scores: Dict[str, int] = self.DEFAULT_SCORES.copy()
        self.load_scores()

    def load_scores(self) -> None:
        """Load scores from JSON file. Creates file with default scores if not exists."""
        try:
            if self._file_path.exists():
                with open(self._file_path, 'r', encoding='utf-8') as f:
                    self._scores = json.load(f)
                    # Validate loaded scores
                    for key in self.DEFAULT_SCORES:
                        if key not in self._scores:
                            self._scores[key] = 0
            else:
                self._scores = self.DEFAULT_SCORES.copy()
                self.save_scores()
        except (json.JSONDecodeError, IOError):
            # If file is corrupted, reset to defaults
            self._scores = self.DEFAULT_SCORES.copy()
            self.save_scores()

    def save_scores(self) -> None:
        """Save current scores to JSON file."""
        try:
            with open(self._file_path, 'w', encoding='utf-8') as f:
                json.dump(self._scores, f, indent=4)
        except IOError:
            # Silently fail if unable to save
            pass

    def update_score(self, winner: Optional[str]) -> None:
        """Update scores based on game outcome.
        
        Args:
            winner: 'X', 'O', or None for draw
        """
        if winner is None:
            self._scores['draw'] += 1
        elif winner in ('X', 'O'):
            self._scores[winner] += 1
        self.save_scores()

    def get_scores(self) -> Dict[str, int]:
        """Get current scores.
        
        Returns:
            Dictionary with keys 'X', 'O', 'draw' and their counts
        """
        return self._scores.copy()

    def reset_scores(self) -> None:
        """Reset all scores to zero and save."""
        self._scores = self.DEFAULT_SCORES.copy()
        self.save_scores()
```
