import unittest
from typing import List

class TestGame(unittest.TestCase):

    def setUp(self) -> None:
        """Set up a fresh Game instance before each test."""
        self.game: Game = Game()

    ## INITIALIZATION TESTS
    def test_initialization(self) -> None:
        """Test that the game initializes with correct default values."""
        self.assertEqual(len(self.game.get_board()), constants.BOARD_SIZE)
        for row in self.game.get_board():
            self.assertEqual(len(row), constants.BOARD_SIZE)
            for cell in row:
                self.assertEqual(cell, " ")
        self.assertEqual(self.game.get_current_player(), constants.PLAYER_X)
        self.assertFalse(self.game.is_game_over())
        self.assertEqual(self.game.get_winner(), "")

    ## MOVE TESTS
    def test_make_valid_move(self) -> None:
        """Test making a valid move updates the board and switches player."""
        self.assertTrue(self.game.make_move(0, 0))
        self.assertEqual(self.game.get_board()[0][0], constants.PLAYER_X)
        self.assertEqual(self.game.get_current_player(), constants.PLAYER_O)

    def test_make_invalid_move_occupied(self) -> None:
        """Test making a move on an occupied cell returns False."""
        self.game.make_move(0, 0)
        self.assertFalse(self.game.make_move(0, 0))
        self.assertEqual(self.game.get_current_player(), constants.PLAYER_O)

    def test_make_invalid_move_out_of_bounds(self) -> None:
        """Test making a move out of bounds raises IndexError."""
        with self.assertRaises(IndexError):
            self.game.make_move(-1, 0)
        with self.assertRaises(IndexError):
            self.game.make_move(0, constants.BOARD_SIZE)

    def test_make_move_after_game_over(self) -> None:
        """Test making a move after game is over returns False."""
        # Force game over by creating a winner
        self.game.make_move(0, 0)  # X
        self.game.make_move(1, 0)  # O
        self.game.make_move(0, 1)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(0, 2)  # X wins
        self.assertTrue(self.game.is_game_over())
        self.assertFalse(self.game.make_move(2, 2))

    ## WINNER TESTS
    def test_check_winner_row(self) -> None:
        """Test row win condition detection."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(1, 0)  # O
        self.game.make_move(0, 1)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(0, 2)  # X wins
        self.assertEqual(self.game.check_winner(), constants.PLAYER_X)
        self.assertTrue(self.game.is_game_over())
        self.assertEqual(self.game.get_winner(), constants.PLAYER_X)

    def test_check_winner_column(self) -> None:
        """Test column win condition detection."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(0, 1)  # O
        self.game.make_move(1, 0)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(2, 0)  # X wins
        self.assertEqual(self.game.check_winner(), constants.PLAYER_X)
        self.assertTrue(self.game.is_game_over())

    def test_check_winner_diagonal(self) -> None:
        """Test diagonal win condition detection."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(0, 1)  # O
        self.game.make_move(1, 1)  # X
        self.game.make_move(0, 2)  # O
        self.game.make_move(2, 2)  # X wins
        self.assertEqual(self.game.check_winner(), constants.PLAYER_X)
        self.assertTrue(self.game.is_game_over())

    def test_check_winner_anti_diagonal(self) -> None:
        """Test anti-diagonal win condition detection."""
        self.game.make_move(0, 2)  # X
        self.game.make_move(0, 0)  # O
        self.game.make_move(1, 1)  # X
        self.game.make_move(1, 0)  # O
        self.game.make_move(2, 0)  # X wins
        self.assertEqual(self.game.check_winner(), constants.PLAYER_X)
        self.assertTrue(self.game.is_game_over())

    def test_check_no_winner(self) -> None:
        """Test no winner is detected when there isn't one."""
        self.assertEqual(self.game.check_winner(), "")
        self.assertFalse(self.game.is_game_over())

    ## DRAW TESTS
    def test_is_draw_true(self) -> None:
        """Test draw condition is detected."""
        # X O X
        # X O O
        # O X X
        moves = [(0, 0), (0, 1), (0, 2),
                 (1, 1), (1, 0), (1, 2),
                 (2, 1), (2, 0), (2, 2)]
        for i, (row, col) in enumerate(moves):
            if i % 2 == 0:
                self.game._current_player = constants.PLAYER_X
            else:
                self.game._current_player = constants.PLAYER_O
            self.game.make_move(row, col)
        self.assertTrue(self.game.is_draw())
        self.assertTrue(self.game.is_game_over())
        self.assertEqual(self.game.get_winner(), "")

    def test_is_draw_false(self) -> None:
        """Test draw is not detected when game isn't over."""
        self.assertFalse(self.game.is_draw())
        self.game.make_move(0, 0)
        self.assertFalse(self.game.is_draw())

    ## RESET TESTS
    def test_reset(self) -> None:
        """Test reset returns game to initial state."""
        self.game.make_move(0, 0)
        self.game.make_move(1, 1)
        self.game._game_over = True
        self.game._winner = constants.PLAYER_X
        self.game.reset()
        self.assertEqual(len(self.game.get_board()), constants.BOARD_SIZE)
        for row in self.game.get_board():

        self.assertEqual(self.game.get_current_player(), constants.PLAYER_X)
        self.assertFalse(self.game.is_game_over())
        self.assertEqual(self.game.get_winner(), "")
