## main.py

import sqlite3
import streamlit as st

class CustomerDB:
    """Class to handle database operations for customer data."""

    def __init__(self) -> None:
        """Initialize the database connection and cursor."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table_if_not_exists()

    def create_table_if_not_exists(self) -> None:
        """Create the customer table if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                birthday TEXT NOT NULL,
                age INTEGER NOT NULL,
                sex TEXT NOT NULL,
                phone TEXT NOT NULL
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Add a new customer to the database."""
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone)
            VALUES (?, ?, ?, ?, ?)
        ''', (name, birthday, age, sex, phone))
        self.conn.commit()

    def get_customer(self, name: str) -> dict:
        """Retrieve a customer's data by name."""
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        if row:
            return {
                'id': row[0],
                'name': row[1],
                'birthday': row[2],
                'age': row[3],
                'sex': row[4],
                'phone': row[5]
            }
        return {}

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Update an existing customer's data."""
        self.cursor.execute('''
            UPDATE customers
            SET birthday = ?, age = ?, sex = ?, phone = ?
            WHERE name = ?
        ''', (birthday, age, sex, phone, name))
        self.conn.commit()

    def delete_customer(self, name: str) -> None:
        """Delete a customer from the database."""
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()

    def close(self) -> None:
        """Close the database connection."""
        self.conn.close()


class CustomerApp:
    """Class to handle the user interface for the CRM system."""

    def __init__(self) -> None:
        """Initialize the application and database."""
        self.db: CustomerDB = CustomerDB()

    def run(self) -> None:
        """Run the main application loop."""
        st.title("Customer Relationship Management System")
        menu = ["Add Customer", "Search Customer", "Edit Customer", "Delete Customer"]
        choice = st.sidebar.selectbox("Select an option", menu)

        if choice == "Add Customer":
            self.show_add_form()
        elif choice == "Search Customer":
            self.show_search_form()
        elif choice == "Edit Customer":
            self.show_edit_form()
        elif choice == "Delete Customer":
            self.show_delete_form()

    def show_add_form(self) -> None:
        """Display the form to add a new customer."""
        st.subheader("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0, max_value=120)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")

        if st.button("Add Customer"):
            self.db.add_customer(name, birthday, age, sex, phone)
            st.success(f"Customer {name} added successfully!")

    def show_search_form(self) -> None:
        """Display the form to search for a customer."""
        st.subheader("Search Customer")
        name = st.text_input("Enter the name of the customer")

        if st.button("Search"):
            customer = self.db.get_customer(name)
            if customer:
                st.write(customer)
            else:
                st.warning("Customer not found.")

    def show_edit_form(self) -> None:
        """Display the form to edit an existing customer."""
        st.subheader("Edit Customer")
        name = st.text_input("Enter the name of the customer to edit")

        if st.button("Load Customer"):
            customer = self.db.get_customer(name)
            if customer:
                birthday = st.text_input("Birthday (YYYY-MM-DD)", value=customer['birthday'])
                age = st.number_input("Age", min_value=0, max_value=120, value=customer['age'])
                sex = st.selectbox("Sex", ["Male", "Female", "Other"], index=["Male", "Female", "Other"].index(customer['sex']))
                phone = st.text_input("Phone", value=customer['phone'])

                if st.button("Update Customer"):
                    self.db.update_customer(name, birthday, age, sex, phone)
                    st.success(f"Customer {name} updated successfully!")
            else:
                st.warning("Customer not found.")

    def show_delete_form(self) -> None:
        """Display the form to delete a customer."""
        st.subheader("Delete Customer")
        name = st.text_input("Enter the name of the customer to delete")

        if st.button("Delete Customer"):
            self.db.delete_customer(name)
            st.success(f"Customer {name} deleted successfully!")

# Run the application
if __name__ == "__main__":
    app = CustomerApp()
    app.run()
