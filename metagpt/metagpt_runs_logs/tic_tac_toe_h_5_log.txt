    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.
        
        Args:
            row: Row index (0-2)
            col: Column index (0-2)
            
        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3) or self.board[row][col] is not None:
            return False

        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self.current_player = "O" if self.current_player == "X" else "X"
        return True

    def _check_game_state(self) -> None:
        """Check if current move resulted in a win or draw."""
        winner = self._check_winner()
        if winner:
            self.winner = winner
            self.game_over = True
        elif self._is_draw():
            self.game_over = True

    def _check_winner(self) -> Optional[str]:
        """Check if current player has won.
        
        Returns:
            Optional[str]: Winning player ("X" or "O") or None if no winner
        """
        # Check rows
        for row in range(3):
            if self.board[row][0] == self.board[row][1] == self.board[row][2] and self.board[row][0] is not None:
                return self.board[row][0]

        # Check columns
        for col in range(3):
            if self.board[0][col] == self.board[1][col] == self.board[2][col] and self.board[0][col] is not None:
                return self.board[0][col]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] and self.board[0][0] is not None:
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] and self.board[0][2] is not None:
            return self.board[0][2]

        return None

"""User Interface module implementing common UI components and patterns."""

from typing import Optional, Callable, Any
import dataclasses
import abc


@dataclasses.dataclass
class Size:
    """Represents dimensions of a UI component."""
    width: int = 100
    height: int = 50


class UIComponent(abc.ABC):
    """Abstract base class for all UI components."""
    
    def __init__(self, name: str, size: Optional[Size] = None) -> None:
        """Initializes the UI component.
        
        Args:
            name: Unique identifier for the component.
            size: Dimensions of the component. Defaults to 100x50.
        """
        self._name = name
        self._size = size if size else Size()
        self._visible = True
        self._enabled = True
    
    @property
    def name(self) -> str:
        """Returns the component's name."""
        return self._name
    
    @property
    def size(self) -> Size:
        """Returns the component's size."""
        return self._size
    
    @property
    def visible(self) -> bool:
        """Returns whether the component is visible."""
        return self._visible
    
    @visible.setter
    def visible(self, value: bool) -> None:
        """Sets the component's visibility."""
        self._visible = value
    
    @property
    def enabled(self) -> bool:
        """Returns whether the component is enabled."""
        return self._enabled
    
    @enabled.setter
    def enabled(self, value: bool) -> None:
        """Sets whether the component is enabled."""
        self._enabled = value
    
    @abc.abstractmethod
    def render(self) -> None:
        """Renders the component to the display."""
        pass


class Button(UIComponent):
    """A clickable button UI component."""
    
    def __init__(
        self,
        name: str,
        text: str,
        on_click: Optional[Callable[[], Any]] = None,
        size: Optional[Size] = None
    ) -> None:
        """Initializes the button.
        
        Args:
            name: Unique identifier for the button.
            text: Display text on the button.
            on_click: Callback when button is clicked.
            size: Dimensions of the button.
        """
        super().__init__(name, size)
        self._text = text
        self._on_click = on_click
    
    @property
    def text(self) -> str:
        """Returns the button's text."""
        return self._text
    
    @text.setter
    def text(self, value: str) -> None:
        """Sets the button's text."""
        self._text = value
    
    def click(self) -> Optional[Any]:
        """Triggers the button's click action.
        
        Returns:
            The result of the on_click callback if provided.
        """
        if self._enabled and self._on_click:
            return self._on_click()
        return None
    
    def render(self) -> None:
        """Renders the button."""
        if not self._visible:
            return
        print(f"Rendering Button '{self._name}': {self._text}")


class TextInput(UIComponent):
    """A text input UI component."""
    
    def __init__(
        self,
        name: str,
        placeholder: str = "",
        size: Optional[Size] = None
    ) -> None:
        """Initializes the text input.
        
        Args:
            name: Unique identifier for the input.
            placeholder: Placeholder text when empty.
            size: Dimensions of the input.
        """
        super().__init__(name, size)
        self._placeholder = placeholder
        self._value = ""
    
    @property
    def value(self) -> str:
        """Returns the current input value."""
        return self._value
    
    @value.setter
    def value(self, text: str) -> None:
        """Sets the input value."""
        self._value = text
    
    @property
    def placeholder(self) -> str:
        """Returns the placeholder text."""
        return self._placeholder
    
    @placeholder.setter
    def placeholder(self, text: str) -> None:
        """Sets the placeholder text."""
        self._placeholder = text
    
    def clear(self) -> None:
        """Clears the input value."""
        self._value = ""
    
    def render(self) -> None:
        """Renders the text input."""
        if not self._visible:
            return
        display_text = self._value if self._value else f"({self._placeholder})"
        print(f"Rendering TextInput '{self._name}': {display_text}")


class UIManager:
    """Manages a collection of UI components."""
    
    def __init__(self) -> None:
        """Initializes the UI manager."""
        self._components: dict[str, UIComponent] = {}
    
    def add_component(self, component: UIComponent) -> None:
        """Adds a component to be managed.
        
        Args:
            component: The UI component to add.
            
        Raises:
            ValueError: If a component with the same name already exists.
        """
        if component.name in self._components:
            raise ValueError(f"Component with name '{component.name}' already exists")
        self._components[component.name] = component
    
    def get_component(self, name: str) -> Optional[UIComponent]:
        """Gets a component by name.
        
        Args:
            name: Name of the component to retrieve.
            
        Returns:
            The component if found, None otherwise.
        """
        return self._components.get(name)
    
    def remove_component(self, name: str) -> None:
        """Removes a component by name.
        
        Args:
            name: Name of the component to remove.
        """
        self._components.pop(name, None)
    
    def render_all(self) -> None:
        """Renders all visible components."""
        for component in self._components.values():
            component.render()

from tic_tac_toe_h_5.game import Game

    """Test suite for the Game class."""

        """Set up a fresh game instance before each test."""
        self.game: Game = Game()

    ## Test Initialization
    def test_initial_state(self) -> None:
        """Test that game initializes with correct default values."""
        self.assertEqual(self.game.current_player, "X")
        self.assertFalse(self.game.game_over)
        self.assertIsNone(self.game.winner)
        for row in range(3):
            for col in range(3):
                self.assertIsNone(self.game.board[row][col])

    ## Test make_move
    def test_valid_move(self) -> None:
        """Test that valid moves are accepted and board is updated."""
        self.assertTrue(self.game.make_move(0, 0))
        self.assertEqual(self.game.board[0][0], "X")
        self.assertEqual(self.game.current_player, "O")

    def test_invalid_move_out_of_bounds(self) -> None:
        """Test that out-of-bounds moves are rejected."""
        self.assertFalse(self.game.make_move(-1, 0))
        self.assertFalse(self.game.make_move(3, 0))
        self.assertFalse(self.game.make_move(0, -1))
        self.assertFalse(self.game.make_move(0, 3))

    def test_invalid_move_occupied_cell(self) -> None:
        """Test that moves to occupied cells are rejected."""
        self.game.make_move(0, 0)
        self.assertFalse(self.game.make_move(0, 0))

    def test_invalid_move_after_game_over(self) -> None:
        """Test that moves are rejected after game is over."""
        # Force game over state
        self.game.game_over = True
        self.assertFalse(self.game.make_move(0, 0))

    ## Test _check_winner
    def test_check_winner_rows(self) -> None:
        """Test row win conditions."""
        # First row win
        self.game.board = [["X", "X", "X"], [None, None, None], [None, None, None]]
        self.assertEqual(self.game._check_winner(), "X")

        # Second row win
        self.game.board = [[None, None, None], ["O", "O", "O"], [None, None, None]]
        self.assertEqual(self.game._check_winner(), "O")

        self.game.board = [[None, "O", None], [None, "O", None], [None, "O", None]]
