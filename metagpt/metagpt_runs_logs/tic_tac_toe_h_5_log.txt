```"""
Tic Tac Toe game logic implementation.
Handles board state, move validation, win/draw detection, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the Tic Tac Toe board and rules."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.reset_game()

    def reset_game(self) -> None:
        """Reset the game state to initial conditions."""
        self.board: List[List[Optional[str]]] = [[None for _ in range(3)] for _ in range(3)]
        self.current_player: str = "X"  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.

        Args:
            row: Row index (0-2)
            col: Column index (0-2)

        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not (0 <= row < 3) or not (0 <= col < 3) or self.board[row][col] is not None:
            return False

        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self._switch_player()
        return True

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = "O" if self.current_player == "X" else "X"

    def _check_game_state(self) -> None:
        """
        Check if the current player has won or if the game is a draw.
        Updates game_over and winner accordingly.
        """
        winner = self.check_winner()
        if winner:
            self.game_over = True
            self.winner = winner
        elif self.is_draw():
            self.game_over = True

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.

        Returns:
            Optional[str]: The winning player ("X" or "O") or None if no winner
        """
        # Check rows and columns
        for i in range(3):
            if self.board[i][0] == self.board[i][1] == self.board[i][2] and self.board[i][0] is not None:
                return self.board[i][0]
            if self.board[0][i] == self.board[1][i] == self.board[2][i] and self.board[0][i] is not None:
                return self.board[0][i]

        # Check diagonals
        if self.board[0][0] == self.board[1][1] == self.board[2][2] and self.board[0][0] is not None:
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] and self.board[0][2] is not None:
            return self.board[0][2]

        return None

    def is_draw(self) -> bool:
        """
        Check if the game is a draw (no winner and board full).

        Returns:
            bool: True if game is a draw, False otherwise
        """
        return all(cell is not None for row in self.board for cell in row) and self.check_winner() is None

    def get_board(self) -> List[List[Optional[str]]]:
        """
        Get the current board state.

        Returns:
            List[List[Optional[str]]]: 3x3 board representation
        """
        return self.board

    def get_current_player(self) -> str:
        """
        Get the current player (X or O).

        Returns:
            str: Current player ("X" or "O")
        """
        return self.current_player

    def is_game_over(self) -> bool:
        """
        Check if the game is over.

        Returns:
            bool: True if game is over, False otherwise
        """
        return self.game_over

```"""Main module for Tic Tac Toe game using Pygame."""

import pygame
import sys
from typing import Tuple, Optional
from game import Game
from ai import AI

# Constants
SCREEN_SIZE = (600, 700)
BOARD_SIZE = 600
GRID_SIZE = BOARD_SIZE // 3
LINE_WIDTH = 10
O_COLOR = (0, 0, 255)  # Blue
X_COLOR = (255, 0, 0)  # Red
LINE_COLOR = (0, 0, 0)  # Black
BG_COLOR = (255, 255, 255)  # White
STATUS_HEIGHT = 100
BUTTON_COLOR = (100, 100, 100)
BUTTON_TEXT_COLOR = (255, 255, 255)
BUTTON_RECT = pygame.Rect(200, 620, 200, 50)


class Main:
    """Main class handling the game interface and loop."""

    def __init__(self) -> None:
        """Initialize the game components and Pygame."""
        pygame.init()
        self.screen = pygame.display.set_mode(SCREEN_SIZE)
        pygame.display.set_caption("Tic Tac Toe")
        self.clock = pygame.time.Clock()
        self.font = pygame.font.SysFont('Arial', 36)
        self.button_font = pygame.font.SysFont('Arial', 24)
        
        self.game = Game()
        self.ai = AI("hard")  # Default to hard difficulty

    def run(self) -> None:
        """Run the main game loop."""
        running = True
        while running:
            self.handle_events()
            self.draw()
            self.clock.tick(60)
        pygame.quit()
        sys.exit()

    def handle_events(self) -> None:
        """Handle Pygame events including mouse clicks."""
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit()
                sys.exit()
            
            if event.type == pygame.MOUSEBUTTONDOWN:
                pos = pygame.mouse.get_pos()
                
                # Check restart button click
                if BUTTON_RECT.collidepoint(pos) and self.game.game_over:
                    self.game.reset_game()
                    return
                
                # Handle player move if game is active and it's player's turn
                if not self.game.game_over and self.game.current_player == "X":
                    row, col = self._get_board_position(pos)
                    if row is not None and col is not None:
                        if self.game.make_move(row, col):
                            # Check if game ended after player move
                            if not self.game.game_over:
                                self._make_ai_move()

    def _make_ai_move(self) -> None:
        """Make AI move if it's AI's turn."""
        if not self.game.game_over and self.game.current_player == "O":
            row, col = self.ai.get_move(self.game)
            self.game.make_move(row, col)

    def _get_board_position(self, pos: Tuple[int, int]) -> Tuple[Optional[int], Optional[int]]:
        """Convert screen position to board row and column."""
        x, y = pos
        if x < 0 or x >= BOARD_SIZE or y < 0 or y >= BOARD_SIZE:
            return None, None
        return y // GRID_SIZE, x // GRID_SIZE

    def draw(self) -> None:
        """Draw all game elements."""
        self.screen.fill(BG_COLOR)
        self._draw_board()
        self._draw_status()
        if self.game.game_over:
            self._draw_restart_button()
        pygame.display.flip()

    def _draw_board(self) -> None:
        """Draw the Tic Tac Toe board and current moves."""
        # Draw grid lines
        for i in range(1, 3):
            # Vertical lines
            pygame.draw.line(
                self.screen, LINE_COLOR,
                (i * GRID_SIZE, 0), (i * GRID_SIZE, BOARD_SIZE),
                LINE_WIDTH
            )
            # Horizontal lines
            pygame.draw.line(
                self.screen, LINE_COLOR,
                (0, i * GRID_SIZE), (BOARD_SIZE, i * GRID_SIZE),
                LINE_WIDTH
            )
        
        # Draw X's and O's
        for row in range(3):
            for col in range(3):
                cell = self.game.get_board()[row][col]
                center_x = col * GRID_SIZE + GRID_SIZE // 2
                center_y = row * GRID_SIZE + GRID_SIZE // 2
                
                if cell == "X":
                    self._draw_x(center_x, center_y)
                elif cell == "O":
                    self._draw_o(center_x, center_y)

    def _draw_x(self, x: int, y: int) -> None:
        """Draw an X at the specified position."""
        offset = GRID_SIZE // 3
        pygame.draw.line(
            self.screen, X_COLOR,
            (x - offset, y - offset), (x + offset, y + offset),
            LINE_WIDTH
        )
        pygame.draw.line(
            self.screen, X_COLOR,
            (x + offset, y - offset), (x - offset, y + offset),
            LINE_WIDTH
        )

    def _draw_o(self, x: int, y: int) -> None:
        """Draw an O at the specified position."""
        radius = GRID_SIZE // 3
        pygame.draw.circle(
            self.screen, O_COLOR,
            (x, y), radius, LINE_WIDTH
        )

    def _draw_status(self) -> None:
        """Draw the current game status."""
        status_text = ""
        if self.game.game_over:
            if self.game.winner:
                status_text = f"Player {self.game.winner} wins!"
            else:
                status_text = "Game ended in a draw!"
        else:
            status_text = f"Player {self.game.current_player}'s turn"
        
        text_surface = self.font.render(status_text, True, LINE_COLOR)
        text_rect = text_surface.get_rect(center=(BOARD_SIZE // 2, BOARD_SIZE + STATUS_HEIGHT // 2))
        self.screen.blit(text_surface, text_rect)

    def _draw_restart_button(self) -> None:
        """Draw the restart button."""
        pygame.draw.rect(self.screen, BUTTON_COLOR, BUTTON_RECT)
        text_surface = self.button_font.render("Play Again", True, BUTTON_TEXT_COLOR)
        text_rect = text_surface.get_rect(center=BUTTON_RECT.center)
        self.screen.blit(text_surface, text_rect)


if __name__ == "__main__":
    game = Main()
    game.run()

        """

            Tuple[int, int]: (row, col) coordinates for the move

        """

                if board[row][col] is None:

                    board[row][col] = "O"
                    score = self._minimax(game, False)
                    board[row][col] = None  # Undo move

                    if score > best_score:
                        best_score = score
                        best_move = (row, col)

        return best_move

    def _minimax(self, game: Game, is_maximizing: bool) -> int:
        """
        Minimax algorithm implementation for move evaluation.

        Args:
            game: Current game instance
            is_maximizing: Whether current player is maximizing (O) or minimizing (X)

        Returns:
            int: Score for the current board state
        """
        # Check terminal states
        winner = game.check_winner()
        if winner == "O":
            return 1
        elif winner == "X":
            return -1
        elif game.is_draw():
            return 0

        board = game.get_board()
        if is_maximizing:
            best_score = -float('inf')
            for row in range(3):
                for col in range(3):
                    if board[row][col] is None:
                        board[row][col] = "O"
                        score = self._minimax(game, False)
                        board[row][col] = None
                        best_score = max(score, best_score)
            return best_score
        else:
            best_score = float('inf')
            for row in range(3):
                for col in range(3):
                    if board[row][col] is None:
                        board[row][col] = "X"
                        score = self._minimax(game, True)
                        board[row][col] = None
                        best_score = min(score, best_score)
            return best_score
