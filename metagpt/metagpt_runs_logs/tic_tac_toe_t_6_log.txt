    def draw_status(self) -> None:
        """Draw the current game status (turn or winner)."""
        status_rect = pygame.Rect(0, BOARD_SIZE, BOARD_SIZE, 100)
        pygame.draw.rect(self.screen, WHITE, status_rect)
        
        if self.game.winner:
            text = f"Player {self.game.winner} wins!" if self.game.winner != 'draw' else "It's a draw!"
            color = GREEN if self.game.winner != 'draw' else BLACK
        else:
            text = f"Player {self.game.current_player}'s turn"
            color = BLACK
            
        text_surface = self.font.render(text, True, color)
        text_rect = text_surface.get_rect(center=(BOARD_SIZE // 2, BOARD_SIZE + 50))
        self.screen.blit(text_surface, text_rect)

    def draw_restart_button(self) -> None:
        """Draw the restart game button."""
        button_rect = pygame.Rect(
            (BOARD_SIZE - BUTTON_WIDTH) // 2,
            BOARD_SIZE + 100,
            BUTTON_WIDTH,
            BUTTON_HEIGHT
        )
        pygame.draw.rect(self.screen, GRAY, button_rect)
        pygame.draw.rect(self.screen, BLACK, button_rect, 2)
        
        text_surface = self.button_font.render("Restart Game", True, BLACK)
        text_rect = text_surface.get_rect(center=button_rect.center)
        self.screen.blit(text_surface, text_rect)
        
        return button_rect

    def handle_click(self, pos: Tuple[int, int]) -> bool:
        """Handle mouse click events.
        
        Args:
            pos: The (x, y) position of the mouse click.
            
        Returns:
            bool: True if the game state was changed, False otherwise.
        """
        # Check if click is on the board and game is still ongoing
        if (pos[0] < 0 or pos[0] >= BOARD_SIZE or 
            pos[1] < 0 or pos[1] >= BOARD_SIZE or 
            self.game.game_over):
            return False
            
        row, col = pos[1] // GRID_SIZE, pos[0] // GRID_SIZE
        
        # Make the player's move
        if self.game.make_move(row, col):
            return True
        return False

    def run(self) -> None:
        """Run the main game loop."""
        restart_button = None
        clock = pygame.time.Clock()
        running = True
        
        while running:
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    running = False
                elif event.type == pygame.MOUSEBUTTONDOWN:
                    pos = pygame.mouse.get_pos()
                    
                    # Check if restart button was clicked
                    if (restart_button and 
                        restart_button.collidepoint(pos)):
                        self.game.reset_game()
                    else:
                        if self.handle_click(pos) and not self.game.game_over:
                            # AI makes move if game isn't over
                            if not self.game.game_over and self.game.current_player == 'O':
                                row, col = self.ai.get_move(self.game)
                                self.game.make_move(row, col)
            
            # Draw all UI elements
            self.draw_board()
            self.draw_status()
            restart_button = self.draw_restart_button()
            
            pygame.display.flip()
            clock.tick(60)
        
        pygame.quit()
