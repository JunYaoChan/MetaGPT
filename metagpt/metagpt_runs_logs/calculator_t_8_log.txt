## main.py
import tkinter as tk  # Importing the tkinter library for GUI

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator with a Tkinter root and display."""
        self.root = tk.Tk()  # Create the main window
        self.root.title("Simple Calculator")  # Set the window title
        self.current_expression = ""  # Initialize the current expression
        self.create_display()  # Create the display for the calculator
        self.create_buttons()  # Create the buttons for the calculator

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display = tk.Entry(self.root, font=("Arial", 24), borderwidth=2, relief="solid")
        self.display.grid(row=0, column=0, columnspan=4, padx=10, pady=10)  # Position the display

    def create_buttons(self) -> None:
        """Create buttons for numbers and operations."""
        # Define button layout
        buttons = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]

        # Create buttons and place them in the grid
        row_val = 1
        col_val = 0
        for button in buttons:
            action = lambda x=button: self.add_to_display(x) if x != '=' and x != 'C' else self.evaluate_expression() if x == '=' else self.clear_display()
            tk.Button(self.root, text=button, font=("Arial", 18), command=action).grid(row=row_val, column=col_val, sticky="nsew", padx=5, pady=5)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

        # Configure grid weights for responsive design
        for i in range(4):
            self.root.grid_columnconfigure(i, weight=1)
        for i in range(5):
            self.root.grid_rowconfigure(i, weight=1)

    def add_to_display(self, value: str) -> None:
        """Add a value to the current expression displayed."""
        self.current_expression += str(value)  # Append the value to the current expression
        self.display.delete(0, tk.END)  # Clear the display
        self.display.insert(tk.END, self.current_expression)  # Show the updated expression

    def clear_display(self) -> None:
        """Clear the current expression."""
        self.current_expression = ""  # Reset the current expression
        self.display.delete(0, tk.END)  # Clear the display

    def evaluate_expression(self) -> None:
        """Evaluate the current expression and display the result."""
        try:
            # Evaluate the expression and update the display
            result = str(eval(self.current_expression))  # Use eval to calculate the result
            self.display.delete(0, tk.END)  # Clear the display
            self.display.insert(tk.END, result)  # Show the result
            self.current_expression = result  # Update the current expression to the result
        except Exception as e:
            self.display.delete(0, tk.END)  # Clear the display on error
            self.display.insert(tk.END, "Error")  # Show error message
            self.current_expression = ""  # Reset the current expression

    def run(self) -> None:
        """Run the main loop of the calculator."""
        self.root.mainloop()  # Start the Tkinter event loop

# Entry point for the application
if __name__ == "__main__":
    calculator = Calculator()  # Create a Calculator instance
    calculator.run()  # Run the calculator
