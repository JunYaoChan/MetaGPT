    def draw_status(self) -> None:
        """Draw the current game status (current player or winner/draw)."""
        if self.game.game_over:
            if self.game.winner:
                text = f"Player {self.game.winner} wins!"
            else:
                text = "Game ended in a draw!"
        else:
            text = f"Player {self.game.current_player}'s turn"
        
        status_text = self.font.render(text, True, LINE_COLOR)
        self.screen.blit(status_text, STATUS_POS)
    
    def draw_restart_button(self) -> None:
        """Draw the restart game button."""
        button_rect = pygame.Rect(BUTTON_POS, BUTTON_SIZE)
        pygame.draw.rect(self.screen, BUTTON_COLOR, button_rect)
        
        button_text = self.button_font.render("Restart", True, BUTTON_TEXT_COLOR)
        text_rect = button_text.get_rect(center=button_rect.center)
        self.screen.blit(button_text, text_rect)
    
    def handle_click(self, pos: tuple[int, int]) -> bool:
        """
        Handle mouse click events.
        
        Args:
            pos: Tuple of (x, y) coordinates of the click
            
        Returns:
            bool: True if the game state changed, False otherwise
        """
        # Check if restart button was clicked
        button_rect = pygame.Rect(BUTTON_POS, BUTTON_SIZE)
        if button_rect.collidepoint(pos):
            self.game.reset_game()
            return True
        
        # If game is over, ignore board clicks
        if self.game.game_over:
            return False
        
        # Convert click position to board coordinates
        cell_size = WINDOW_SIZE[0] // 3
        col = pos[0] // cell_size
        row = pos[1] // cell_size
        
        # Attempt to make move
        return self.game.make_move(row, col)
    
    def run(self) -> None:
        """Run the main game loop."""
        clock = pygame.time.Clock()
        running = True
        
        while running:
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    running = False
                elif event.type == pygame.MOUSEBUTTONDOWN:
                    if self.handle_click(event.pos):
                        # Redraw everything if game state changed
                        self.draw_board()
                        self.draw_symbols()
                        self.draw_status()
                        self.draw_restart_button()
                        pygame.display.flip()
            
            # Initial draw
            self.draw_board()
            self.draw_symbols()
            self.draw_status()
            self.draw_restart_button()
            pygame.display.flip()
            clock.tick(60)
        
        pygame.quit()
