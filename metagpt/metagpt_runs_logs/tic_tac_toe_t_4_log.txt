    def _get_random_move(self, board: List[List[str]]) -> Tuple[int, int]:
        """Get a random valid move (easy difficulty).
        
        Args:
            board: The current game board
            
        Returns:
            Random valid (row, col) move
        """
        empty_cells = [
            (i, j) 
            for i in range(3) 
            for j in range(3) 
            if board[i][j] == EMPTY
        ]
        return random.choice(empty_cells)
    
    def _get_medium_move(self, board: List[List[str]]) -> Tuple[int, int]:
        """Get a move with basic strategy (medium difficulty).
        
        Strategy:
        1. Try to win immediately
        2. Block opponent's immediate win
        3. Take center if available
        4. Take a corner if available
        5. Random move otherwise
        
        Args:
            board: The current game board
            
        Returns:
            Selected (row, col) move
        """
        # Check for winning move
        move = self._find_winning_move(board, PLAYER_O)
        if move:
            return move
        
        # Check for blocking move
        move = self._find_winning_move(board, PLAYER_X)
        if move:
            return move
        
        # Take center if available
        if board[1][1] == EMPTY:
            return (1, 1)
        
        # Take a random corner
        corners = [(0, 0), (0, 2), (2, 0), (2, 2)]
        empty_corners = [c for c in corners if board[c[0]][c[1]] == EMPTY]
        if empty_corners:
            return random.choice(empty_corners)
        
        # Fall back to random move
        return self._get_random_move(board)
    
    def _get_hard_move(self, board: List[List[str]]) -> Tuple[int, int]:
        """Get the optimal move using minimax algorithm (hard difficulty).
        
        Args:
            board: The current game board
            
        Returns:
            Optimal (row, col) move
        """
        best_score = -float('inf')
        best_move = None
        
        for i in range(3):
            for j in range(3):
                if board[i][j] == EMPTY:
                    board[i][j] = PLAYER_O
                    score = self._minimax(board, 0, False)
                    board[i][j] = EMPTY
                    if score > best_score:
                        best_score = score
                        best_move = (i, j)
        
        return best_move or self._get_random_move(board)
    
    def _minimax(self, board: List[List[str]], depth: int, is_maximizing: bool) -> int:
        """Minimax algorithm helper function for hard difficulty.
        
        Args:
            board: Current game board state
            depth: Current depth in the game tree
            is_maximizing: Whether current player is maximizing (AI) or minimizing (human)
            
        Returns:
            Score for the current board state
        """
        winner = self._check_winner(board)
        if winner == PLAYER_O:
            return 10 - depth
        elif winner == PLAYER_X:
            return depth - 10
        elif self._is_board_full(board):
            return 0
        
        if is_maximizing:
            best_score = -float('inf')
            for i in range(3):
                for j in range(3):
                    if board[i][j] == EMPTY:
                        board[i][j] = PLAYER_O
                        score = self._minimax(board, depth + 1, False)
                        board[i][j] = EMPTY
                        best_score = max(score, best_score)
            return best_score
        else:
            best_score = float('inf')
            for i in range(3):
                for j in range(3):
                    if board[i][j] == EMPTY:
                        board[i][j] = PLAYER_X
                        score = self._minimax(board, depth + 1, True)
                        board[i][j] = EMPTY
                        best_score = min(score, best_score)
            return best_score
    
    def _find_winning_move(self, board: List[List[str]], player: str) -> Optional[Tuple[int, int]]:
        """Find a winning move for the specified player if one exists.
        
        Args:
            board: Current game board
            player: Player to check for winning move (PLAYER_X or PLAYER_O)
            
        Returns:
            Winning move (row, col) if found, else None
        """
        for i in range(3):
            for j in range(3):
                if board[i][j] == EMPTY:
                    board[i][j] = player
                    if self._check_winner(board) == player:
                        board[i][j] = EMPTY
                        return (i, j)
                    board[i][j] = EMPTY
        return None
    
    def _check_winner(self, board: List[List[str]]) -> Optional[str]:
        """Check if there's a winner on the current board.
        
        Args:
            board: Current game board
            
        Returns:
            Winning player (PLAYER_X or PLAYER_O) if found, else None
        """
        # Check rows
        for row in board:
            if row[0] == row[1] == row[2] != EMPTY:
                return row[0]
        
        # Check columns
        for col in range(3):
            if board[0][col] == board[1][col] == board[2][col] != EMPTY:
                return board[0][col]
        
        # Check diagonals
        if board[0][0] == board[1][1] == board[2][2] != EMPTY:
            return board[0][0]
        if board[0][2] == board[1][1] == board[2][0] != EMPTY:
            return board[0][2]
        
        return None
    
    def _is_board_full(self, board: List[List[str]]) -> bool:
        """Check if the board is full (no empty spaces).
        
        Args:
            board: Current game board
            
        Returns:
            True if board is full, False otherwise
        """
        return all(cell != EMPTY for row in board for cell in row)

        """Test that moves outside board boundaries raise IndexError."""
        with self.assertRaises(IndexError):
            self.game.make_move(-1, 0)
        with self.assertRaises(IndexError):
            self.game.make_move(0, constants.BOARD_SIZE)

    def test_move_after_game_over(self) -> None:
        """Test that moves can't be made after game is over."""
        self.game.game_over = True
        self.assertFalse(self.game.make_move(0, 0))

    ## WINNER CHECK TESTS
    def test_check_winner_row(self) -> None:
        """Test row win detection."""
        for i in range(constants.BOARD_SIZE):
            self.game.make_move(0, i)
            if i < constants.BOARD_SIZE - 1:
                self.game.make_move(1, i)  # Opponent move to prevent other wins
        self.assertEqual(self.game.check_winner(), constants.PLAYER_X)

    def test_check_winner_column(self) -> None:
        """Test column win detection."""
        for i in range(constants.BOARD_SIZE):
            self.game.make_move(i, 0)
            if i < constants.BOARD_SIZE - 1:
                self.game.make_move(i, 1)  # Opponent move to prevent other wins
        self.assertEqual(self.game.check_winner(), constants.PLAYER_X)

    def test_check_winner_diagonal(self) -> None:
        """Test diagonal win detection."""
        for i in range(constants.BOARD_SIZE):
