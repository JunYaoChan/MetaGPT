## main.py

import tkinter as tk

class Calculator:
    """A simple calculator class using Tkinter for the UI."""

    def __init__(self) -> None:
        """Initialize the calculator and its components."""
        self.root = tk.Tk()  # Create the main window
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='groove')
        self.buttons = {}  # Dictionary to hold button references
        self.create_widgets()  # Create the UI components
        self.bind_buttons()  # Bind button actions

    def create_widgets(self) -> None:
        """Create the calculator's UI components."""
        # Set the title and geometry of the main window
        self.root.title("Simple Calculator")
        self.root.geometry("400x600")

        # Place the display at the top
        self.display.grid(row=0, column=0, columnspan=4)

        # Define button layout
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]

        # Create buttons and place them in the grid
        for index, text in enumerate(button_texts):
            row = (index // 4) + 1  # Calculate row number
            col = index % 4  # Calculate column number
            self.buttons[text] = tk.Button(self.root, text=text, font=('Arial', 18), command=lambda t=text: self.button_click(t))
            self.buttons[text].grid(row=row, column=col, sticky="nsew")

        # Configure grid weights for responsive design
        for i in range(5):
            self.root.grid_rowconfigure(i, weight=1)
            self.root.grid_columnconfigure(i, weight=1)

    def bind_buttons(self) -> None:
        """Bind button actions to their respective methods."""
        # No additional binding needed as buttons are already bound in create_widgets

    def button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == 'C':
            self.clear()  # Clear the display
        elif char == '=':
            self.calculate()  # Calculate the result
        else:
            self.display.insert(tk.END, char)  # Insert the clicked character into the display

    def calculate(self) -> None:
        """Evaluate the expression in the display and show the result."""
        try:
            expression = self.display.get()  # Get the current expression
            result = eval(expression)  # Evaluate the expression
            self.clear()  # Clear the display
            self.display.insert(tk.END, str(result))  # Show the result
        except Exception as e:
            self.clear()  # Clear the display on error
            self.display.insert(tk.END, "Error")  # Show error message

    def clear(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)  # Delete all content in the display

    def run(self) -> None:
        """Run the main loop of the application."""
        self.root.mainloop()  # Start the Tkinter event loop

if __name__ == "__main__":
    calculator = Calculator()  # Create a Calculator instance
    calculator.run()  # Run the calculator application
