## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class using tkinter for GUI."""

    def __init__(self) -> None:
        """Initialize the calculator and create the main window."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='solid')
        self.display.grid(row=0, column=0, columnspan=4)

        self.buttons = {
            '7': (1, 0), '8': (1, 1), '9': (1, 2), '/': (1, 3),
            '4': (2, 0), '5': (2, 1), '6': (2, 2), '*': (2, 3),
            '1': (3, 0), '2': (3, 1), '3': (3, 2), '-': (3, 3),
            '0': (4, 0), 'C': (4, 1), '=': (4, 2), '+': (4, 3)
        }
        self.create_widgets()

    def create_widgets(self) -> None:
        """Create buttons and assign their functionality."""
        for (text, (row, col)) in self.buttons.items():
            button = tk.Button(self.root, text=text, width=5, height=2, font=('Arial', 18),
                               command=lambda char=text: self.button_click(char))
            button.grid(row=row, column=col)

    def button_click(self, char: str) -> None:
        """Handle button clicks and update the display."""
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            current_text = self.display.get()
            new_text = current_text + char
            self.display.delete(0, tk.END)
            self.display.insert(0, new_text)

    def calculate(self) -> None:
        """Perform the calculation based on the current display content."""
        try:
            expression = self.display.get()
            result = eval(expression)  # Note: eval can be dangerous; use with caution.
            self.display.delete(0, tk.END)
            self.display.insert(0, str(result))
        except ZeroDivisionError:
            messagebox.showerror("Error", "Division by zero is not allowed.")
            self.clear_display()
        except Exception as e:
            messagebox.showerror("Error", f"Invalid input: {e}")
            self.clear_display()

    def clear_display(self) -> None:
        """Clear the calculator display."""
        self.display.delete(0, tk.END)

    def run(self) -> None:
        """Run the main loop of the calculator."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
