"""A modular UI component following Google-style Python guidelines.

This module provides a base UI class with common functionality that can be extended
for specific implementations. It follows principles of clean code and maintainability.

from typing import Optional, Any, Dict
import logging


class UIComponent:
    """Base class for UI components with common functionality.

    Attributes:
        _config: Dictionary containing configuration parameters.
        _logger: Logger instance for the component.
        _initialized: Flag indicating if component is initialized.
    """

    def __init__(self, config: Optional[Dict[str, Any]] = None) -> None:
        """Initializes the UI component with optional configuration.

        Args:
            config: Optional configuration dictionary. If None, defaults will be used.
        """
        self._config = config or {}
        self._logger = logging.getLogger(self.__class__.__name__)
        self._initialized = False

    def initialize(self) -> None:
        """Initializes the UI component resources.

        Raises:
            RuntimeError: If initialization fails.
        """
        if self._initialized:
            self._logger.warning('Component already initialized')
            return

        try:
            self._setup_resources()
            self._initialized = True
            self._logger.info('Component initialized successfully')
        except Exception as e:
            self._logger.error('Initialization failed: %s', str(e))
            raise RuntimeError('UI component initialization failed') from e

    def _setup_resources(self) -> None:
        """Sets up necessary resources for the UI component.

        This method should be overridden by subclasses for specific implementations.
        """
        raise NotImplementedError('Subclasses must implement _setup_resources')

    def render(self) -> None:
        """Renders the UI component.

        Raises:
            RuntimeError: If component is not initialized or rendering fails.
        """
        if not self._initialized:
            raise RuntimeError('Component must be initialized before rendering')

        try:
            self._logger.debug('Rendering UI component')
            self._perform_render()
        except Exception as e:
            self._logger.error('Rendering failed: %s', str(e))
            raise RuntimeError('UI rendering failed') from e

    def _perform_render(self) -> None:
        """Performs the actual rendering of the UI component.

        This method should be overridden by subclasses for specific implementations.
        """
        raise NotImplementedError('Subclasses must implement _perform_render')

    def cleanup(self) -> None:
        """Cleans up resources used by the UI component."""
        if not self._initialized:
            return

        try:
            self._release_resources()
            self._initialized = False
            self._logger.info('Component cleaned up successfully')
        except Exception as e:
            self._logger.error('Cleanup failed: %s', str(e))
            raise RuntimeError('UI component cleanup failed') from e

    def _release_resources(self) -> None:
        """Releases resources used by the UI component.

        This method should be overridden by subclasses for specific implementations.
        """
        raise NotImplementedError('Subclasses must implement _release_resources')

    @property
    def is_initialized(self) -> bool:
        """Returns whether the component is initialized.

        Returns:
            bool: True if initialized, False otherwise.
        """
        return self._initialized


def create_ui_component(config: Optional[Dict[str, Any]] = None) -> UIComponent:
    """Factory function to create a UI component instance.

    Args:
        config: Optional configuration dictionary.

    Returns:
        UIComponent: A new UI component instance.
    """
    return UIComponent(config)

"""

import unittest
from tic_tac_toe_h_8.game import Game


class TestGame(unittest.TestCase):
    """Test suite for the Tic Tac Toe Game class."""

    def setUp(self) -> None:
        """Initialize a fresh game instance before each test."""
        self.game = Game()

    ## INITIALIZATION TESTS
    def test_initial_state(self) -> None:
        """Test that game initializes with correct default values."""
        self.assertEqual(len(self.game.board), 3, "Board should have 3 rows")
        self.assertEqual(len(self.game.board[0]), 3, "Board should have 3 columns")
        self.assertTrue(all(cell is None for row in self.game.board for cell in row),
                       "All board cells should be None initially")
        self.assertEqual(self.game.current_player, 'X', "X should be the starting player")
        self.assertFalse(self.game.game_over, "Game should not be over initially")
        self.assertIsNone(self.game.winner, "There should be no winner initially")

    ## MOVE VALIDATION TESTS
    def test_valid_move(self) -> None:
        """Test that valid moves are accepted and update the board."""
        self.assertTrue(self.game.make_move(0, 0), "Valid move should return True")
        self.assertEqual(self.game.board[0][0], 'X', "Board should be updated with player's mark")
        self.assertEqual(self.game.current_player, 'O', "Player should switch after valid move")

    def test_invalid_move_out_of_bounds(self) -> None:
        """Test that out-of-bounds moves are rejected."""
        self.assertFalse(self.game.make_move(-1, 0), "Negative row should be invalid")
        self.assertFalse(self.game.make_move(3, 0), "Row > 2 should be invalid")
        self.assertFalse(self.game.make_move(0, -1), "Negative column should be invalid")
        self.assertFalse(self.game.make_move(0, 3), "Column > 2 should be invalid")

    def test_invalid_move_occupied_cell(self) -> None:
        """Test that moves to occupied cells are rejected."""
        self.game.make_move(0, 0)
        self.assertFalse(self.game.make_move(0, 0), "Move to occupied cell should be invalid")

    def test_move_after_game_over(self) -> None:
        """Test that moves are rejected after game is over."""
        # Force a win
        self.game.make_move(0, 0)  # X
        self.game.make_move(1, 0)  # O
        self.game.make_move(0, 1)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(0, 2)  # X wins
        self.assertFalse(self.game.make_move(2, 2), "Move after game over should be invalid")

    ## WIN CHECKING TESTS
    def test_row_win(self) -> None:
        """Test that row wins are correctly detected."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(1, 0)  # O
        self.game.make_move(0, 1)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(0, 2)  # X wins
        self.assertTrue(self.game.game_over, "Game should be over after win")
        self.assertEqual(self.game.winner, 'X', "X should be the winner")

    def test_column_win(self) -> None:
        """Test that column wins are correctly detected."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(0, 1)  # O
        self.game.make_move(1, 0)  # X
        self.game.make_move(1, 1)  # O
        self.game.make_move(2, 0)  # X wins
        self.assertTrue(self.game.game_over, "Game should be over after win")
        self.assertEqual(self.game.winner, 'X', "X should be the winner")

    def test_diagonal_win(self) -> None:
        """Test that diagonal wins are correctly detected."""
        self.game.make_move(0, 0)  # X
        self.game.make_move(0, 1)  # O
        self.game.make_move(1, 1)  # X
        self.game.make_move(0, 2)  # O
        self.game.make_move(2, 2)  # X wins
        self.assertTrue(self.game.game_over, "Game should be over after win")
