import numpy as np

class Game:
    """Class to manage the Tic Tac Toe game state and rules."""

    def __init__(self) -> None:
        """Initialize the game board and set the starting player."""
        self.board: list[list[str]] = [['' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # Player 'X' starts the game
        self.game_over: bool = False

    def make_move(self, row: int, col: int) -> bool:
        """Make a move on the board.

        Args:
            row (int): The row index for the move.
            col (int): The column index for the move.

        Returns:
            bool: True if the move was successful, False otherwise.
        """
        if self.board[row][col] == '' and not self.game_over:
            self.board[row][col] = self.current_player
            return True
        return False

    def check_winner(self) -> str:
        """Check for a winner in the game.

        Returns:
            str: The symbol of the winner ('X' or 'O'), or an empty string if no winner.
        """
        # Check rows, columns, and diagonals for a winner
        for i in range(3):
            if self.board[i][0] == self.board[i][1] == self.board[i][2] != '':
                return self.board[i][0]
            if self.board[0][i] == self.board[1][i] == self.board[2][i] != '':
                return self.board[0][i]

        if self.board[0][0] == self.board[1][1] == self.board[2][2] != '':
            return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != '':
            return self.board[0][2]

        return ''

    def is_draw(self) -> bool:
        """Check if the game is a draw.

        Returns:
            bool: True if the game is a draw, False otherwise.
        """
        return all(cell != '' for row in self.board for cell in row) and not self.check_winner()

    def reset_game(self) -> None:
        """Reset the game to the initial state."""
        self.board = [['' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False

_winning_move(self, game: Game, player: str) -> Tuple[int, int] | None:
        """Find a move that would win the game for the given player.
        
        Args:
            game: The current game instance.
            player: The player to check for ('X' or 'O').
            
        Returns:
            A tuple (row, col) representing the winning move, or None if none exists.
        """
        for row in range(3):
            for col in range(3):
                if game.board[row][col] == ' ':
                    game.board[row][col] = player
                    winner = game.check_winner()
                    game.board[row][col] = ' '
                    if winner == player:
                        return (row, col)
        return None
