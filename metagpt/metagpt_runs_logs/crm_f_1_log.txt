import sqlite3
import streamlit as st
from typing import List, Dict

class CustomerCRUD:
    """Class to handle all database operations related to customer records."""

    def __init__(self) -> None:
        """Initialize the database connection and create the customer table if it doesn't exist."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()

    def create_table(self) -> None:
        """Create the customer table in the database if it doesn't exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                birthday TEXT NOT NULL,
                age INTEGER NOT NULL CHECK(age > 0),
                sex TEXT NOT NULL,
                phone TEXT NOT NULL
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> bool:
        """Add a new customer to the database.

        Args:
            name (str): The name of the customer.
            birthday (str): The birthday of the customer in YYYY-MM-DD format.
            age (int): The age of the customer.
            sex (str): The sex of the customer.
            phone (str): The phone number of the customer.

        Returns:
            bool: True if the customer was added successfully, False otherwise.
        """
        try:
            self.cursor.execute('''
                INSERT INTO customers (name, birthday, age, sex, phone)
                VALUES (?, ?, ?, ?, ?)
            ''', (name, birthday, age, sex, phone))
            self.conn.commit()
            return True
        except sqlite3.Error:
            return False

    def get_customer(self, name: str) -> Dict:
        """Retrieve a customer's details by name.

        Args:
            name (str): The name of the customer.

        Returns:
            dict: A dictionary containing customer details or an empty dictionary if not found.
        """
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        return {
            'id': row[0],
            'name': row[1],
            'birthday': row[2],
            'age': row[3],
            'sex': row[4],
            'phone': row[5]
        } if row else {}

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> bool:
        """Update an existing customer's details.

        Args:
            name (str): The name of the customer to update.
            birthday (str): The new birthday of the customer.
            age (int): The new age of the customer.
            sex (str): The new sex of the customer.
            phone (str): The new phone number of the customer.

        Returns:
            bool: True if the customer was updated successfully, False otherwise.
        """
        try:
            self.cursor.execute('''
                UPDATE customers
                SET birthday = ?, age = ?, sex = ?, phone = ?
                WHERE name = ?
            ''', (birthday, age, sex, phone, name))
            self.conn.commit()
            return True
        except sqlite3.Error:
            return False

    def delete_customer(self, name: str) -> bool:
        """Delete a customer from the database.

        Args:
            name (str): The name of the customer to delete.

        Returns:
            bool: True if the customer was deleted successfully, False otherwise.
        """
        try:
            self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
            self.conn.commit()
            return True
        except sqlite3.Error:
            return False

    def get_all_customers(self) -> List[Dict]:
        """Retrieve all customers from the database.

        Returns:
            list[dict]: A list of dictionaries containing all customer details.
        """
        self.cursor.execute('SELECT * FROM customers')
        rows = self.cursor.fetchall()
        return [{
            'id': row[0],
            'name': row[1],
            'birthday': row[2],
            'age': row[3],
            'sex': row[4],
            'phone': row[5]
        } for row in rows]

class StreamlitUI:

                st.success("Customer deleted successfully!")
            else:
                st.error("Failed to delete customer.")
