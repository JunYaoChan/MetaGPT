## main.py
import tkinter as tk

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator with a Tkinter root window and UI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='groove')
        self.display.grid(row=0, column=0, columnspan=4)
        self.buttons = {}
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.grid(row=0, column=0, columnspan=4)

    def create_buttons(self) -> None:
        """Create buttons for the calculator and arrange them in a grid layout."""
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        row_val = 1
        col_val = 0
        
        for text in button_texts:
            self.buttons[text] = tk.Button(self.root, text=text, width=5, height=2, font=('Arial', 18),
                                            command=lambda t=text: self.on_button_click(t))
            self.buttons[text].grid(row=row_val, column=col_val)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def add_to_display(self, value: str) -> None:
        """Add a value to the display."""
        current_text = self.display.get()
        self.display.delete(0, tk.END)  # Clear the display
        self.display.insert(tk.END, current_text + value)  # Append new value

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Calculate the result of the expression in the display."""
        try:
            result = eval(self.display.get())  # Evaluate the expression
            self.clear_display()  # Clear the display before showing result
            self.display.insert(tk.END, str(result))  # Show result
        except Exception as e:
            self.clear_display()  # Clear display on error
            self.display.insert(tk.END, "Error")  # Show error message

    def on_button_click(self, value: str) -> None:
        """Handle button click events."""
        if value == 'C':
            self.clear_display()
        elif value == '=':
            self.calculate()
        else:
            self.add_to_display(value)

    def run(self) -> None:
        """Run the main loop of the Tkinter application."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
