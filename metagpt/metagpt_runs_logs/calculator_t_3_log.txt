## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class using Tkinter for GUI."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='solid')
        self.display.grid(row=0, column=0, columnspan=4)
        self.buttons = {}
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.delete(0, tk.END)

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        row_val = 1
        col_val = 0
        
        for text in button_texts:
            self.buttons[text] = tk.Button(self.root, text=text, width=5, height=2,
                                            command=lambda t=text: self.on_button_click(t))
            self.buttons[text].grid(row=row_val, column=col_val)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def add_to_display(self, char: str) -> None:
        """Add a character to the display."""
        current_text = self.display.get()
        self.display.delete(0, tk.END)
        self.display.insert(0, current_text + char)

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Perform the calculation based on the current display content."""
        try:
            result = eval(self.display.get())
            self.display.delete(0, tk.END)
            self.display.insert(0, str(result))
        except ZeroDivisionError:
            self.handle_error("Error: Division by zero")
        except Exception as e:
            self.handle_error(f"Error: {str(e)}")

    def handle_error(self, message: str) -> None:
        """Handle errors by displaying a message box."""
        messagebox.showerror("Calculation Error", message)

    def on_button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == '=':
            self.calculate()
        elif char == 'C':
            self.clear_display()
        else:
            self.add_to_display(char)

    def run(self) -> None:
        """Run the main loop of the calculator."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
