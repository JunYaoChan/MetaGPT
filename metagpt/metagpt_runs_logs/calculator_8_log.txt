## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = ""
        self.create_display()
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display_var = tk.StringVar()
        self.display_entry = tk.Entry(self.root, textvariable=self.display_var, font=("Arial", 24), bd=10, insertwidth=2, width=14, borderwidth=4)
        self.display_entry.grid(row=0, column=0, columnspan=4)

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        buttons = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        row_val = 1
        col_val = 0
        
        for button in buttons:
            self.create_button(button, row_val, col_val)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def create_button(self, char: str, row: int, column: int) -> None:
        """Create a button and place it on the grid."""
        button = tk.Button(self.root, text=char, padx=20, pady=20, font=("Arial", 18),
                           command=lambda: self.button_click(char))
        button.grid(row=row, column=column)

    def button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.calculate()
        else:
            self.display += char
            self.update_display()

    def update_display(self) -> None:
        """Update the display with the current input."""
        self.display_var.set(self.display)

    def calculate(self) -> None:
        """Perform the calculation based on the current input."""
        try:
            # Evaluate the expression and update the display
            result = eval(self.display)
            self.display_var.set(result)
            self.display = str(result)  # Store the result for further calculations
        except ZeroDivisionError:
            messagebox.showerror("Error", "Division by zero is not allowed.")
            self.clear_display()
        except Exception as e:
            messagebox.showerror("Error", f"Invalid input: {e}")
            self.clear_display()

    def clear_display(self) -> None:
        """Clear the display and reset the input."""
        self.display = ""
        self.update_display()

    def run(self) -> None:
        """Run the main loop of the Tkinter application."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
