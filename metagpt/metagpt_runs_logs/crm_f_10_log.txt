## main.py

import sqlite3
import streamlit as st

class CustomerCRUD:
    """Class to handle CRUD operations for customer data in SQLite database."""

    def __init__(self) -> None:
        """Initialize the database connection and create the customer table if it doesn't exist."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()

    def create_table(self) -> None:
        """Create the customer table if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                birthday TEXT NOT NULL,
                age INTEGER NOT NULL CHECK(age >= 0),
                sex TEXT NOT NULL,
                phone TEXT NOT NULL
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Add a new customer to the database."""
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone)
            VALUES (?, ?, ?, ?, ?)
        ''', (name, birthday, age, sex, phone))
        self.conn.commit()

    def search_customer(self, name: str) -> list:
        """Search for a customer by name and return their details."""
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        return self.cursor.fetchall()

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Update an existing customer's details."""
        self.cursor.execute('''
            UPDATE customers
            SET birthday = ?, age = ?, sex = ?, phone = ?
            WHERE name = ?
        ''', (birthday, age, sex, phone, name))
        self.conn.commit()

    def delete_customer(self, name: str) -> None:
        """Delete a customer from the database by name."""
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()

    def close_connection(self) -> None:
        """Close the database connection."""
        self.conn.close()


class StreamlitUI:
    """Class to render the Streamlit user interface for customer management."""

    def __init__(self, crud: CustomerCRUD) -> None:
        """Initialize the Streamlit UI with a CustomerCRUD instance."""
        self.crud: CustomerCRUD = crud

    def render_add_section(self) -> None:
        """Render the section for adding a new customer."""
        st.header("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")

        if st.button("Add Customer"):
            self.crud.add_customer(name, birthday, age, sex, phone)
            st.success("Customer added successfully!")

    def render_search_section(self) -> None:
        """Render the section for searching a customer."""
        st.header("Search Customer")
        name = st.text_input("Enter Name to Search")

        if st.button("Search"):
            results = self.crud.search_customer(name)
            if results:
                for row in results:
                    st.write(f"Name: {row[1]}, Birthday: {row[2]}, Age: {row[3]}, Sex: {row[4]}, Phone: {row[5]}")
            else:
                st.warning("No customer found.")

    def render_update_section(self) -> None:
        """Render the section for updating a customer's details."""
        st.header("Update Customer")
        name = st.text_input("Enter Name to Update")
        birthday = st.text_input("New Birthday (YYYY-MM-DD)")
        age = st.number_input("New Age", min_value=0)
        sex = st.selectbox("New Sex", ["Male", "Female", "Other"])
        phone = st.text_input("New Phone")

        if st.button("Update Customer"):
            self.crud.update_customer(name, birthday, age, sex, phone)
            st.success("Customer updated successfully!")

    def render_delete_section(self) -> None:
        """Render the section for deleting a customer."""
        st.header("Delete Customer")
        name = st.text_input("Enter Name to Delete")

        if st.button("Delete Customer"):
            self.crud.delete_customer(name)
            st.success("Customer deleted successfully!")

    def run(self) -> None:
        """Run the Streamlit application."""
        st.title("Customer Relationship Management (CRM) Tool")
        self.render_add_section()
        self.render_search_section()
        self.render_update_section()
        self.render_delete_section()


if __name__ == "__main__":
    # Initialize the CustomerCRUD and StreamlitUI classes
    crud = CustomerCRUD()
    ui = StreamlitUI(crud)
    ui.run()
    crud.close_connection()
