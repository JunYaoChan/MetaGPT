## main.py
import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator and create the GUI components."""
        self.current_expression: str = ""
        self.window: tk.Tk = tk.Tk()
        self.window.title("Simple Calculator")
        self.display: tk.Entry = self.create_display()
        self.buttons: dict = self.create_buttons()
        self.layout_components()

    def create_display(self) -> tk.Entry:
        """Create and return the display entry widget."""
        display = tk.Entry(self.window, font=("Arial", 24), borderwidth=2, relief="ridge")
        display.grid(row=0, column=0, columnspan=4, padx=10, pady=10)
        return display

    def create_buttons(self) -> dict:
        """Create and return a dictionary of buttons for the calculator."""
        buttons = {
            '7': (1, 0), '8': (1, 1), '9': (1, 2), '/': (1, 3),
            '4': (2, 0), '5': (2, 1), '6': (2, 2), '*': (2, 3),
            '1': (3, 0), '2': (3, 1), '3': (3, 2), '-': (3, 3),
            '0': (4, 0), 'C': (4, 1), '=': (4, 2), '+': (4, 3)
        }
        for text, position in buttons.items():
            button = tk.Button(self.window, text=text, font=("Arial", 18),
                               command=lambda t=text: self.on_button_click(t))
            button.grid(row=position[0], column=position[1], sticky="nsew", padx=5, pady=5)
        return buttons

    def layout_components(self) -> None:
        """Configure the grid layout for the buttons."""
        for i in range(5):
            self.window.grid_rowconfigure(i, weight=1)
            self.window.grid_columnconfigure(i, weight=1)

    def on_button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == 'C':
            self.clear_display()
        elif char == '=':
            self.evaluate_expression()
        else:
            self.update_display(char)

    def update_display(self, value: str) -> None:
        """Update the display with the given value."""
        self.current_expression += str(value)
        self.display.delete(0, tk.END)
        self.display.insert(tk.END, self.current_expression)

    def evaluate_expression(self) -> None:
        """Evaluate the current expression and update the display."""
        try:
            result = eval(self.current_expression)
            self.display.delete(0, tk.END)
            self.display.insert(tk.END, str(result))
            self.current_expression = str(result)  # Update current expression to result
        except ZeroDivisionError:
            messagebox.showerror("Error", "Division by zero is not allowed.")
            self.clear_display()
        except Exception as e:
            messagebox.showerror("Error", f"Invalid input: {e}")
            self.clear_display()

    def clear_display(self) -> None:
        """Clear the display and reset the current expression."""
        self.current_expression = ""
        self.display.delete(0, tk.END)

    def run(self) -> None:
        """Run the main loop of the calculator application."""
        self.window.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
