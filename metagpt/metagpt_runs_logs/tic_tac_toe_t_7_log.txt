# Game symbols
EMPTY = ""
X = "X"
O = "O"

class Game:
    """Manages the game state and logic."""
    
    def __init__(self) -> None:
        """Initialize the game with empty board and starting player."""
        self.board: List[List[str]] = [[EMPTY for _ in range(BOARD_SIZE)] 
                                     for _ in range(BOARD_SIZE)]
        self.current_player: str = X
        self.game_over: bool = False
        self.winner: Optional[str] = None
    
    def handle_click(self, pos: Tuple[int, int]) -> bool:
        """
        Handle a player's move at the given position.
        
        Args:
            pos: The (row, col) position where the player clicked
            
        Returns:
            bool: True if the move was valid and processed, False otherwise
        """
        if self.game_over:
            return False
            
        row, col = pos
        if self.board[row][col] != EMPTY:
            return False
            
        self.board[row][col] = self.current_player
        
        # Check for winner or draw
        self.winner = self.check_winner()
        if self.winner or self.is_draw():
            self.game_over = True
        else:
            self.current_player = O if self.current_player == X else X
            
        return True
    
    def check_winner(self) -> Optional[str]:
        """Check if there's a winner on the current board."""
        # Check rows
        for row in self.board:
            if row[0] != EMPTY and all(cell == row[0] for cell in row):
                return row[0]
        
        # Check columns
        for col in range(BOARD_SIZE):
            if self.board[0][col] != EMPTY and all(
                self.board[row][col] == self.board[0][col] for row in range(BOARD_SIZE)
            ):
                return self.board[0][col]
        
        # Check diagonals
        if self.board[0][0] != EMPTY and all(
            self.board[i][i] == self.board[0][0] for i in range(BOARD_SIZE)
        ):
            return self.board[0][0]
        
        if self.board[0][BOARD_SIZE-1] != EMPTY and all(
            self.board[i][BOARD_SIZE-1-i] == self.board[0][BOARD_SIZE-1] 
            for i in range(BOARD_SIZE)
        ):
            return self.board[0][BOARD_SIZE-1]
        
        return None
    
    def is_draw(self) -> bool:
        """Check if the game is a draw (no winner and board full)."""
        return all(cell != EMPTY for row in self.board for cell in row) and not self.winner
    
    def reset(self) -> None:
        """Reset the game to initial state."""
        self.__init__()
    
    def get_board(self) -> List[List[str]]:
        """Get the current board state."""
        return self.board
    
    def get_current_player(self) -> str:
        """Get the current player (X or O)."""
        return self.current_player
    
    def is_game_over(self) -> bool:
        """Check if the game is over."""
        return self.game_over
    
    def get_winner(self) -> Optional[str]:
        """Get the winner (X or O) if there is one."""
        return self.winner
