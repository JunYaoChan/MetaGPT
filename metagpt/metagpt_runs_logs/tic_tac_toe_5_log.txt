import numpy as np

class Game:
    """Class to manage the Tic Tac Toe game state and logic."""

    def __init__(self) -> None:
        """Initialize the game with an empty board and set the first player."""
        self.board: list[list[str]] = [['' for _ in range(3)] for _ in range(3)]
        self.current_player: str = 'X'  # Player 'X' starts the game
        self.game_over: bool = False
        self.winner: str = ''

    def make_move(self, row: int, col: int) -> bool:
        """Make a move on the board if the position is valid.

        Args:
            row (int): The row index for the move.
            col (int): The column index for the move.

        Returns:
            bool: True if the move was successful, False otherwise.
        """
        if self.board[row][col] == '' and not self.game_over:
            self.board[row][col] = self.current_player
            return True
        return False

    def check_winner(self) -> str:
        """Check for a winner in the game.

        Returns:
            str: The winner ('X' or 'O') or an empty string if no winner.
        """
        # Check rows, columns, and diagonals for a win
        for i in range(3):
            if self.board[i][0] == self.board[i][1] == self.board[i][2] != '':
                self.winner = self.board[i][0]
                self.game_over = True
                return self.winner
            if self.board[0][i] == self.board[1][i] == self.board[2][i] != '':
                self.winner = self.board[0][i]
                self.game_over = True
                return self.winner

        if self.board[0][0] == self.board[1][1] == self.board[2][2] != '':
            self.winner = self.board[0][0]
            self.game_over = True
            return self.winner
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != '':
            self.winner = self.board[0][2]
            self.game_over = True
            return self.winner

        return ''

    def is_draw(self) -> bool:
        """Check if the game is a draw.

        Returns:
            bool: True if the game is a draw, False otherwise.
        """
        if all(cell != '' for row in self.board for cell in row) and not self.game_over:
            self.game_over = True
            return True
        return False

    def reset_game(self) -> None:
        """Reset the game to the initial state."""
        self.board = [['' for _ in range(3)] for _ in range(3)]
        self.current_player = 'X'
        self.game_over = False
        self.winner = ''

    def get_board(self) -> list[list[str]]:
        """Get the current state of the board.

        Returns:
            list[list[str]]: The current board state.
        """
        return self.board

    def get_current_player(self) -> str:
        """Get the current player.

        Returns:
            str: The current player ('X' or 'O').
        """
        return self.current_player

    def is_game_over(self) -> bool:
        """Check if the game is over.

        Returns:
            bool: True if the game is over, False otherwise.
        """
        return self.game_over

    def get_winner(self) -> str:
        """Get the winner of the game.

        Returns:
            str: The winner ('X' or 'O') or an empty string if no winner.
        """
        return self.winner

    def run(self) -> None:
        """Main game loop."""
        clock = pygame.time.Clock()
        running = True
        
        while running:
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    running = False
                elif event.type == pygame.MOUSEBUTTONDOWN:
                    if event.button == 1:  # Left mouse button
                        self.handle_click(event.pos)
            
            self.draw_board()
            self.draw_status()
            self.draw_buttons()
            
            # Show celebration if game is over
            if self.game.is_game_over():
                if self.game.get_winner():
                    self.show_celebration(f"Player {self.game.get_winner()} wins!")
                else:
                    self.show_celebration("Draw!")
            
            pygame.display.flip()
            clock.tick(60)  # Cap at 60 FPS
        
        pygame.quit()

        moves = [(0, 0), (0, 1), (0, 2),

                 (2, 1), (2, 0), (2, 2)]
        for i, (row, col) in enumerate(moves):

            self.game.board[row][col] = player

        self.assertTrue(self.game.is_game_over())

    def test_not_draw(self) -> None:

        self.assertFalse(self.game.is_game_over())
