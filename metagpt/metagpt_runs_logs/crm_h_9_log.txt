            name TEXT NOT NULL,
            birthday TEXT NOT NULL,
            sex TEXT NOT NULL,
            phone TEXT NOT NULL
        )
        '''
        self.cursor.execute(create_table_query)
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, sex: str, phone: str) -> bool:
        """Add a new customer to the database.

        Args:
            name (str): The name of the customer.
            birthday (str): The birthday of the customer in YYYY-MM-DD format.
            sex (str): The sex of the customer.
            phone (str): The phone number of the customer.

        Returns:
            bool: True if the customer was added successfully, False otherwise.
        """
        if not self.validate_phone(phone):
            return False
        
        insert_query = '''
        INSERT INTO customers (name, birthday, sex, phone)
        VALUES (?, ?, ?, ?)
        '''
        self.cursor.execute(insert_query, (name, birthday, sex, phone))
        self.conn.commit()
        return True

    def get_customer(self, name: str) -> dict:
        """Retrieve a customer's data by name.

        Args:
            name (str): The name of the customer.

        Returns:
            dict: A dictionary containing customer data.
        """
        select_query = 'SELECT * FROM customers WHERE name = ?'
        self.cursor.execute(select_query, (name,))
        customer = self.cursor.fetchone()
        if customer:
            return {
                'id': customer[0],
                'name': customer[1],
                'birthday': customer[2],
                'sex': customer[3],
                'phone': customer[4],
                'age': self.calculate_age(customer[2])
            }
        return {}

    def update_customer(self, name: str, birthday: str, sex: str, phone: str) -> bool:
        """Update an existing customer's data.

        Args:
            name (str): The name of the customer to update.
            birthday (str): The new birthday of the customer.
            sex (str): The new sex of the customer.
            phone (str): The new phone number of the customer.

        Returns:
            bool: True if the customer was updated successfully, False otherwise.
        """
        if not self.validate_phone(phone):
            return False
        
        update_query = '''
        UPDATE customers

        WHERE name = ?
        '''
        self.cursor.execute(update_query, (birthday, sex, phone, name))

    def delete_customer(self, name: str) -> bool:
        """Delete a customer from the database.

        Args:
            name (str): The name of the customer to delete.

        Returns:
            bool: True if the customer was deleted successfully, False otherwise.
        """
        delete_query = 'DELETE FROM customers WHERE name = ?'
        self.cursor.execute(delete_query, (name,))
        self.conn.commit()
        return True

    def calculate_age(self, birthday: str) -> int:
        """Calculate the age of a customer based on their birthday.

        Args:
            birthday (str): The birthday of the customer in YYYY-MM-DD format.

        Returns:
            int: The age of the customer.
        """
        birth_date = datetime.strptime(birthday, '%Y-%m-%d')
        today = datetime.today()
        age = today.year - birth_date.year - ((today.month, today.day) < (birth_date.month, birth_date.day))
        return age

    def validate_phone(self, phone: str) -> bool:
        """Validate the phone number format.

        Args:
            phone (str): The phone number to validate.

        Returns:
            bool: True if the phone number is valid, False otherwise.
        """
        # Basic validation: check if the phone number is numeric and has 10 digits
        return phone.isdigit() and len(phone) == 10
