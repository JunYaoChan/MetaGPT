    def _get_random_move(self, game) -> Tuple[int, int]:
        """Get a random valid move (easy difficulty)."""
        valid_moves = self._get_valid_moves(game)
        return random.choice(valid_moves)
    
    def _get_medium_move(self, game) -> Tuple[int, int]:
        """Get a move with some intelligence (medium difficulty)."""
        # 50% chance to make a random move, 50% to make optimal move
        if random.random() < 0.5:
            return self._get_random_move(game)
        return self._get_best_move(game)
    
    def _get_best_move(self, game) -> Tuple[int, int]:
        """Get the optimal move using minimax algorithm (hard difficulty)."""
        best_score = -math.inf
        best_move = (-1, -1)
        
        for move in self._get_valid_moves(game):
            row, col = move
            game.board[row][col] = PLAYER_O
            score = self._minimax(game, 0, False)
            game.board[row][col] = EMPTY
            
            if score > best_score:
                best_score = score
                best_move = move
                
        return best_move
    
    def _minimax(self, game, depth: int, is_maximizing: bool) -> int:
        """Minimax algorithm implementation with depth limiting.
        
        Args:
            game: Current game state
            depth: Current depth in the game tree
            is_maximizing: Whether current player is maximizing (AI)
            
        Returns:
            Score for the current board state
        """
        winner = game.check_winner()
        if winner == PLAYER_O:
            return 10 - depth
        elif winner == PLAYER_X:
            return depth - 10
        elif game.is_board_full():
            return 0
            
        if is_maximizing:
            best_score = -math.inf
            for move in self._get_valid_moves(game):
                row, col = move
                game.board[row][col] = PLAYER_O
                score = self._minimax(game, depth + 1, False)
                game.board[row][col] = EMPTY
                best_score = max(score, best_score)
            return best_score
        else:
            best_score = math.inf
            for move in self._get_valid_moves(game):
                row, col = move
                game.board[row][col] = PLAYER_X
                score = self._minimax(game, depth + 1, True)
                game.board[row][col] = EMPTY
                best_score = min(score, best_score)
            return best_score
    
    def _get_valid_moves(self, game) -> List[Tuple[int, int]]:
        """Get all valid moves from current game state.
        
        Args:
            game: Current game instance
            
        Returns:
            List of valid (row, col) moves
        """
        valid_moves = []
        for row in range(BOARD_SIZE):
            for col in range(BOARD_SIZE):
                if game.board[row][col] == EMPTY:
                    valid_moves.append((row, col))
        return valid_moves
