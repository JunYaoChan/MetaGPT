    def _check_game_state(self) -> None:
        """Check if current move resulted in a win or draw."""
        winner = self._check_winner()
        if winner:
            self.winner = winner
            self.game_over = True
        elif self._is_draw():
            self.game_over = True

    def _check_winner(self) -> Optional[str]:
        """
        Check if the current player has won.
        
        Returns:
            Optional[str]: The winning player ('X' or 'O') or None if no winner
        """
        # Check rows
        for row in self.board:
            if row[0] == row[1] == row[2] and row[0] is not None:
                return row[0]

        # Check columns
        for col in range(3):
            if (self.board[0][col] == self.board[1][col] == self.board[2][col] and 
                self.board[0][col] is not None):
                return self.board[0][col]

        # Check diagonals
        if (self.board[0][0] == self.board[1][1] == self.board[2][2] and 
            self.board[0][0] is not None):
            return self.board[0][0]
        if (self.board[0][2] == self.board[1][1] == self.board[2][0] and 
            self.board[0][2] is not None):
            return self.board[0][2]

        return None

    def _is_draw(self) -> bool:
        """
        Check if the game is a draw (no winner and board full).
        
        Returns:
            bool: True if game is a draw, False otherwise
        """
        return all(cell is not None for row in self.board for cell in row)
