```import tkinter as tk
from tkinter import StringVar, messagebox

class Calculator:
    def __init__(self, master: tk.Tk) -> None:
        """Initialize the Calculator class with the main window and default values."""
        self.master = master
        self.master.title("Simple Calculator")  # Set the window title
        self.master.geometry("400x600")  # Set the window size

        # Initialize display variable and current state
        self.display_var = StringVar()
        self.current_input: str = ""
        self.current_operation: str = ""
        self.first_number: float = 0.0

        # Create the GUI components
        self.create_widgets()

    def create_widgets(self) -> None:
        """Create and layout the calculator widgets."""
        # Create display label
        display_label = tk.Label(self.master, textvariable=self.display_var, font=("Arial", 24), bg="white", anchor='e')
        display_label.pack(expand=True, fill='both')

        # Create buttons for numbers and operations
        button_frame = tk.Frame(self.master)
        button_frame.pack(expand=True, fill='both')

        # Define button layout
        buttons = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            '0', 'C', '=', '+'
        ]

        # Create buttons dynamically
        for i, button in enumerate(buttons):
            btn = tk.Button(button_frame, text=button, font=("Arial", 18), command=lambda b=button: self.button_click(b))
            btn.grid(row=i // 4, column=i % 4, sticky="nsew")

        # Configure grid weights for responsive design
        for i in range(4):
            button_frame.grid_columnconfigure(i, weight=1)
            button_frame.grid_rowconfigure(i, weight=1)

        # Update display to show initial state
        self.update_display()

    def update_display(self) -> None:
        """Update the display with the current input."""
        self.display_var.set(self.current_input)

# Main execution
if __name__ == "__main__":
    root = tk.Tk()  # Create the main window
    calculator = Calculator(root)  # Initialize the Calculator
    root.mainloop()  # Start the GUI event loop
```

```## Implement operation button functionality (operation_click)
        """Handle number button clicks and update the display."""
        if number == 'C':
            self.clear_click()  # Clear the input if 'C' is pressed
        else:
            self.current_input += number  # Append the clicked number to current input
            self.update_display()  # Update the display with the new input

    def operation_click(self, operation: str) -> None:
        """Handle operation button clicks and store the first number and operation."""
        if self.current_input:  # Check if there is a current input
            self.first_number = float(self.current_input)  # Convert current input to float
            self.current_operation = operation  # Store the operation
            self.current_input = ""  # Clear current input for the next number
            self.update_display()  # Update the display to reflect the operation

    def update_display(self) -> None:
        """Update the display with the current input."""
        self.display_var.set(self.current_input)

    def clear_click(self) -> None:
        """Clear the current input and reset the calculator state."""
        self.current_input = ""  # Reset current input
```

```"""A simple calculator application using tkinter."""
        """Initialize calculator with main window and default values.

        Args:
            master: The root tkinter window.
        """

        master.title("Calculator")
        master.resizable(False, False)

        # Calculator state variables
        self.display_var = tk.StringVar()
        self.current_input = "0"
        self.current_operation = None
        self.first_number = 0.0
        self.reset_input = False

        # Configure styles
        self.button_font = tkfont.Font(family="Arial", size=12, weight="bold")
        self.display_font = tkfont.Font(family="Arial", size=24)

        self.create_widgets()
        self.update_display()

    def create_widgets(self) -> None:
        """Create and arrange all GUI components."""
        # Display
        display = tk.Entry(
            self.master,
            textvariable=self.display_var,
            font=self.display_font,
            bd=10,
            insertwidth=1,
            width=14,
            borderwidth=4,
            justify="right",
            state="readonly",

        display.grid(row=0, column=0, columnspan=4, padx=5, pady=5)

        buttons = [
            ("7", 1, 0), ("8", 1, 1), ("9", 1, 2), ("/", 1, 3),
            ("4", 2, 0), ("5", 2, 1), ("6", 2, 2), ("*", 2, 3),
            ("1", 3, 0), ("2", 3, 1), ("3", 3, 2), ("-", 3, 3),
            ("0", 4, 0), ("C", 4, 1), ("=", 4, 2), ("+", 4, 3),
        ]

        # Create buttons
        for (text, row, col) in buttons:
            button = tk.Button(
                self.master,
                text=text,
                font=self.button_font,
                padx=20,
                pady=20,
                command=lambda t=text: self.on_button_click(t),
            )
            button.grid(row=row, column=col, sticky="nsew")

            # Configure button colors
            if text in {"+", "-", "*", "/", "="}:
                button.config(bg="#FF9500", fg="white")
            elif text == "C":
                button.config(bg="#A5A5A5", fg="white")
            else:
                button.config(bg="#333333", fg="white")

        # Configure grid weights
        for i in range(5):
            self.master.grid_rowconfigure(i, weight=1)
        for i in range(4):
            self.master.grid_columnconfigure(i, weight=1)

    def on_button_click(self, button_text: str) -> None:
        """Handle all button clicks.

        Args:
            button_text: The text of the button that was clicked.
        """
        if button_text.isdigit() or button_text == ".":
```
```
            self.button_click(button_text)
        elif button_text in {"+", "-", "*", "/"}:
            self.operation_click(button_text)
        elif button_text == "=":
            self.equals_click()
        elif button_text == "C":
            self.clear_click()

    def button_click(self, number: str) -> None:
        """Handle number button clicks.

        Args:
            number: The number or decimal point that was clicked.
        """
        if self.reset_input:
            self.current_input = "0"
            self.reset_input = False

        if number == ".":
            if "." not in self.current_input:
                self.current_input += number
        else:
            if self.current_input == "0":
                self.current_input = number
            else:
                self.current_input += number

        self.update_display()

    def operation_click(self, operation: str) -> None:
        """Handle operation button clicks.

        Args:
            operation: The operation that was clicked (+, -, *, /).
        """
        try:
            if self.current_operation and not self.reset_input:
                self.equals_click()

            self.first_number = float(self.current_input)
            self.current_operation = operation
            self.reset_input = True
        except ValueError:
            self.display_error()

    def equals_click(self) -> None:
        """Handle equals button click and perform calculation."""
        if not self.current_operation:
            return

        try:
            second_number = float(self.current_input)
            if self.current_operation == "+":
                result = self.first_number + second_number
            elif self.current_operation == "-":
                result = self.first_number - second_number
            elif self.current_operation == "*":
                result = self.first_number * second_number
            elif self.current_operation == "/":
                if second_number == 0:
                    raise ZeroDivisionError
                result = self.first_number / second_number

            self.current_input = str(result)
            self.current_operation = None
            self.reset_input = True
            self.update_display()
        except (ValueError, ZeroDivisionError):
            self.display_error()

    def clear_click(self) -> None:
        """Handle clear button click and reset calculator state."""
        self.current_input = "0"
        self.current_operation = None
        self.first_number = 0.0
        self.reset_input = False
        self.update_display()

    def update_display(self) -> None:
        """Update the display with current input value."""
        self.display_var.set(self.current_input)

    def display_error(self) -> None:
        """Display error message and reset calculator state."""
        self.display_var.set("Error")
        self.current_input = "0"
        self.current_operation = None
        self.first_number = 0.0
        self.reset_input = True
```

```import tkinter as tk
        else:
            self.current_input += number
        
        self.update_display()

            self.current_operation = operation
```

```"""A simple calculator application using tkinter."""
            self.update_display()

            ("1", 3, 0), ("2", 3, 1), ("3", 3, 2), ("-", 3, 3),
            ("0", 4, 0), ("C", 4, 1), ("=", 4, 2), ("+", 4, 3),
        ]

        # Create buttons
        for (text, row, col) in buttons:
            button = tk.Button(
                self.master,
                text=text,
                font=self.button_font,

                pady=20,

                button.config(bg="#A5A5A5", fg="white")

        for i in range(5):
```

```## Implement equals button functionality (equals_click)
        for i in range(4):

                self.current_input += number
        else:

                self.current_input = number
            else:
                self.current_input += number

        self.update_display()
```

```"""A simple calculator application using tkinter."""
import tkinter as tk
from tkinter import font as tkfont


class Calculator:
    """Main calculator class implementing GUI and logic."""

    def __init__(self, master: tk.Tk) -> None:
        """Initialize calculator with main window and default values.

        Args:
            master: The root tkinter window.
        """
        self.master = master
        master.title("Calculator")
        master.resizable(False, False)

        # Calculator state variables
        self.display_var = tk.StringVar()
        self.current_input = "0"
        self.current_operation = None
        self.first_number = 0.0
        self.reset_input = False

        # Configure styles
        self.button_font = tkfont.Font(family="Arial", size=12, weight="bold")
        self.display_font = tkfont.Font(family="Arial", size=24)

        self.create_widgets()
        self.update_display()

    def create_widgets(self) -> None:
        """Create and arrange all GUI components."""
        # Display
        display = tk.Entry(
            self.master,
            textvariable=self.display_var,
            font=self.display_font,
            bd=10,
            insertwidth=1,
            width=14,
            borderwidth=4,
            justify="right",
            state="readonly",
        )
        display.grid(row=0, column=0, columnspan=4, padx=5, pady=5)

        # Button layout
        buttons = [
            ("7", 1, 0), ("8", 1, 1), ("9", 1, 2), ("/", 1, 3),
            ("4", 2, 0), ("5", 2, 1), ("6", 2, 2), ("*", 2, 3),
            ("1", 3, 0), ("2", 3, 1), ("3", 3, 2), ("-", 3, 3),
            ("0", 4, 0), ("C", 4, 1), ("=", 4, 2), ("+", 4, 3),
        ]

        # Create buttons
        for (text, row, col) in buttons:
            button = tk.Button(
                self.master,
                text=text,
                font=self.button_font,
                padx=20,
                pady=20,
                command=lambda t=text: self.on_button_click(t),
            )
            button.grid(row=row, column=col, sticky="nsew")

            # Configure button colors
            if text in {"+", "-", "*", "/", "="}:
                button.config(bg="#FF9500", fg="white")
            elif text == "C":
                button.config(bg="#A5A5A5", fg="white")
            else:
                button.config(bg="#333333", fg="white")

        # Configure grid weights
        for i in range(5):
            self.master.grid_rowconfigure(i, weight=1)
        for i in range(4):
            self.master.grid_columnconfigure(i, weight=1)
```

    def on_button_click(self, button_text: str) -> None:
        """Handle all button clicks.

        Args:
            button_text: The text of the button that was clicked.
        """
        if button_text.isdigit() or button_text == ".":
            self.button_click(button_text)
        elif button_text in {"+", "-", "*", "/"}:
            self.operation_click(button_text)
        elif button_text == "=":
            self.equals_click()
        elif button_text == "C":
            self.clear_click()

    def button_click(self, number: str) -> None:
        """Handle number button clicks.

        Args:
            number: The number or decimal point that was clicked.
        """
        if self.reset_input:
            self.current_input = "0"
            self.reset_input = False

        if number == ".":
            if "." not in self.current_input:
                self.current_input += number
        else:
            if self.current_input == "0":
                self.current_input = number
            else:
                self.current_input += number

        self.update_display()

    def operation_click(self, operation: str) -> None:
        """Handle operation button clicks.

        Args:
            operation: The operation that was clicked (+, -, *, /).
        """
        try:
            if self.current_operation and not self.reset_input:
                self.equals_click()

            self.first_number = float(self.current_input)
            self.current_operation = operation
            self.reset_input = True
        except ValueError:
            self.display_error()

    def equals_click(self) -> None:
        """Handle equals button click and perform calculation."""
        if not self.current_operation:
            return

        try:
            second_number = float(self.current_input)
            if self.current_operation == "+":
                result = self.first_number + second_number
            elif self.current_operation == "-":
                result = self.first_number - second_number
            elif self.current_operation == "*":
                result = self.first_number * second_number
            elif self.current_operation == "/":
                if second_number == 0:
                    raise ZeroDivisionError
```python
                result = self.first_number / second_number

            self.current_input = str(result)
```
```python
            self.current_operation = None
            self.reset_input = True
            self.update_display()
        except (ValueError, ZeroDivisionError):
            self.display_error()

    def clear_click(self) -> None:
        """Handle clear button click and reset calculator state."""
        self.current_input = "0"
        self.current_operation = None
        self.first_number = 0.0
        self.reset_input = False
        self.update_display()

    def update_display(self) -> None:
        """Update the display with current input value."""

    def display_error(self) -> None:
        """Display error message and reset calculator state."""
        self.display_var.set("Error")

        self.first_number = 0.0
```

```python
"""A complete calculator implementation with error handling for division by zero and invalid inputs."""

import tkinter as tk
from tkinter import messagebox
from tkinter import font as tkfont
from typing import Optional


class Calculator:
    """A simple calculator GUI application with basic arithmetic operations."""

    def __init__(self, master: tk.Tk) -> None:
        """Initialize the calculator with main window and default values.
        
        Args:
            master: The root tkinter window.
        """
        self.master = master
        master.title("Calculator")
        master.resizable(False, False)

        # Calculator state variables
        self.display_var = tk.StringVar()
        self.current_input: str = "0"
        self.current_operation: Optional[str] = None
        self.first_number: float = 0.0
        self.reset_input: bool = False

        # Configure styles
        self.button_font = tkfont.Font(family="Arial", size=12, weight="bold")
        self.display_font = tkfont.Font(family="Arial", size=24)

        self.create_widgets()
        self.update_display()

    def create_widgets(self) -> None:
        """Create and arrange all GUI components."""
        # Display
        display = tk.Entry(
            self.master,
            textvariable=self.display_var,
            font=self.display_font,
            bd=10,
            insertwidth=1,
            width=14,
            borderwidth=4,
            justify="right",
            state="readonly",
        )
        display.grid(row=0, column=0, columnspan=4, padx=5, pady=5)

        # Button layout
        buttons = [
            ("7", 1, 0), ("8", 1, 1), ("9", 1, 2), ("/", 1, 3),
            ("4", 2, 0), ("5", 2, 1), ("6", 2, 2), ("*", 2, 3),
            ("1", 3, 0), ("2", 3, 1), ("3", 3, 2), ("-", 3, 3),
            ("0", 4, 0), ("C", 4, 1), ("=", 4, 2), ("+", 4, 3),
        ]

        # Create buttons
        for (text, row, col) in buttons:
            button = tk.Button(
                self.master,
                text=text,
                font=self.button_font,
                padx=20,
                pady=20,
                command=lambda t=text: self.on_button_click(t),
            )
            button.grid(row=row, column=col, sticky="nsew")

            # Configure button colors
            if text in {"+", "-", "*", "/", "="}:
                button.config(bg="#FF9500", fg="white")
            elif text == "C":
                button.config(bg="#A5A5A5", fg="white")
            else:
                button.config(bg="#333333", fg="white")

        # Configure grid weights
        for i in range(5):
            self.master.grid_rowconfigure(i, weight=1)
        for i in range(4):
            self.master.grid_columnconfigure(i, weight=1)

    def on_button_click(self, button_text: str) -> None:
        """Route button clicks to appropriate handlers.
        
        Args:
            button_text: The text of the button that was clicked.
        """
        if button_text.isdigit() or button_text == ".":
            self.button_click(button_text)
        elif button_text in {"+", "-", "*", "/"}:
            self.operation_click(button_text)
        elif button_text == "=":
            self.equals_click()
        elif button_text == "C":
            self.clear_click()

    def button_click(self, number: str) -> None:
        """Handle number button clicks.
        
        Args:
            number: The number or decimal point that was clicked.
        """
        if self.reset_input:
            self.current_input = "0"
            self.reset_input = False

        if number == ".":
            if "." not in self.current_input:
                self.current_input += number
        else:
            if self.current_input == "0":
                self.current_input = number
            else:
                self.current_input += number

        self.update_display()

    def operation_click(self, operation: str) -> None:
        """Handle operation button clicks with error checking.
        
        Args:
            operation: The operation that was clicked (+, -, *, /).
        """
        try:
            if self.current_operation and not self.reset_input:
                self.equals_click()

            self.first_number = float(self.current_input)
            self.current_operation = operation
            self.reset_input = True
        except ValueError:
            self.handle_error("Invalid input")

    def equals_click(self) -> None:
        """Perform calculation with error handling for division by zero and invalid inputs."""
        if not self.current_operation:
            return

        try:
            second_number = float(self.current_input)
            
            if self.current_operation == "+":
                result = self.first_number + second_number
            elif self.current_operation == "-":
                result = self.first_number - second_number
            elif self.current_operation == "*":
                result = self.first_number * second_number
            elif self.current_operation == "/":
                if second_number == 0:
                    raise ZeroDivisionError("Cannot divide by zero")
                result = self.first_number / second_number

            self.current_input = str(result)
            self.current_operation = None
            self.reset_input = True
            self.update_display()
        except ValueError:
            self.handle_error("Invalid input")
        except ZeroDivisionError as e:
            self.handle_error(str(e))

    def clear_click(self) -> None:
        """Reset calculator to initial state."""
        self.current_input = "0"
        self.current_operation = None
        self.first_number = 0.0
        self.reset_input = False
        self.update_display()

    def update_display(self) -> None:
        """Update the display with current input value."""
        self.display_var.set(self.current_input)

    def handle_error(self, message: str) -> None:
        """Handle errors by displaying message and resetting calculator.
        
        Args:
            message: The error message to display.
        """
        messagebox.showerror("Error", message)
        self.display_var.set("Error")
        self.current_input = "0"
        self.current_operation = None
        self.first_number = 0.0
        self.reset_input = True


def main() -> None:
    """Create and run the calculator application."""
    root = tk.Tk()
    calculator = Calculator(root)
    root.mainloop()


if __name__ == "__main__":
    main()
```
