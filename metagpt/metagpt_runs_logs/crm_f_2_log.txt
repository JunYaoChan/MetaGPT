## main.py
import sqlite3
import streamlit as st

class CustomerCRUD:
    """Class to handle CRUD operations for customer data in SQLite database."""

    def __init__(self) -> None:
        """Initialize the database connection and cursor."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()

    def create_table(self) -> None:
        """Create the customer table if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                birthday TEXT NOT NULL,
                age INTEGER NOT NULL,
                sex TEXT NOT NULL,
                phone TEXT NOT NULL
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Add a new customer to the database."""
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone)
            VALUES (?, ?, ?, ?, ?)
        ''', (name, birthday, age, sex, phone))
        self.conn.commit()

    def get_customer(self, name: str) -> dict:
        """Retrieve customer data by name."""
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        if row:
            return {
                'id': row[0],
                'name': row[1],
                'birthday': row[2],
                'age': row[3],
                'sex': row[4],
                'phone': row[5]
            }
        return {}

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Update existing customer data."""
        self.cursor.execute('''
            UPDATE customers
            SET birthday = ?, age = ?, sex = ?, phone = ?
            WHERE name = ?
        ''', (birthday, age, sex, phone, name))
        self.conn.commit()

    def delete_customer(self, name: str) -> None:
        """Delete a customer from the database."""
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()

    def close_connection(self) -> None:
        """Close the database connection."""
        self.conn.close()


class StreamlitUI:
    """Class to render the user interface using Streamlit."""

    def __init__(self, crud: CustomerCRUD) -> None:
        """Initialize the Streamlit UI with a CustomerCRUD instance."""
        self.crud: CustomerCRUD = crud

    def render(self) -> None:
        """Render the Streamlit user interface."""
        st.title("Customer Relationship Management (CRM) System")

        # Tab selection for different operations
        operation = st.sidebar.selectbox("Select Operation", ["Add", "View", "Update", "Delete"])

        if operation == "Add":
            self.add_customer_form()
        elif operation == "View":
            self.view_customer_form()
        elif operation == "Update":
            self.update_customer_form()
        elif operation == "Delete":
            self.delete_customer_form()

    def add_customer_form(self) -> None:
        """Render the form to add a new customer."""
        st.subheader("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")

        if st.button("Add Customer"):
            self.crud.add_customer(name, birthday, age, sex, phone)
            st.success("Customer added successfully!")

    def view_customer_form(self) -> None:
        """Render the form to view a customer's details."""
        st.subheader("View Customer")
        name = st.text_input("Enter Customer Name")

        if st.button("View Customer"):
            customer = self.crud.get_customer(name)
            if customer:
                st.write(customer)
            else:
                st.error("Customer not found.")

    def update_customer_form(self) -> None:
        """Render the form to update an existing customer's details."""
        st.subheader("Update Customer")
        name = st.text_input("Enter Customer Name")

        if st.button("Fetch Customer"):
            customer = self.crud.get_customer(name)
            if customer:
                birthday = st.text_input("Birthday (YYYY-MM-DD)", value=customer['birthday'])
                age = st.number_input("Age", min_value=0, value=customer['age'])
                sex = st.selectbox("Sex", ["Male", "Female", "Other"], index=["Male", "Female", "Other"].index(customer['sex']))
                phone = st.text_input("Phone", value=customer['phone'])

                if st.button("Update Customer"):
                    self.crud.update_customer(name, birthday, age, sex, phone)
                    st.success("Customer updated successfully!")
            else:
                st.error("Customer not found.")

    def delete_customer_form(self) -> None:
        """Render the form to delete a customer."""
        st.subheader("Delete Customer")
        name = st.text_input("Enter Customer Name")

        if st.button("Delete Customer"):
            self.crud.delete_customer(name)
            st.success("Customer deleted successfully!")


if __name__ == "__main__":
    # Initialize the CustomerCRUD and StreamlitUI classes
    crud = CustomerCRUD()
    ui = StreamlitUI(crud)
    ui.render()
    crud.close_connection()
