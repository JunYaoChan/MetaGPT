```"""
Tic Tac Toe game logic implementation.
Handles board state, move validation, win/draw checking, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game rules."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.reset_game()

    def reset_game(self) -> None:
        """Reset the game state to initial conditions."""
        self.board: List[List[Optional[str]]] = [
            [None, None, None],
            [None, None, None],
            [None, None, None]
        ]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.

        Args:
            row: Row index (0-2)
            col: Column index (0-2)

        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not self._is_valid_position(row, col):
            return False

        if self.board[row][col] is not None:
            return False

        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self._switch_player()
        return True

    def _is_valid_position(self, row: int, col: int) -> bool:
        """Check if position indices are within valid range."""
        return 0 <= row < 3 and 0 <= col < 3

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if current player has won or if game is drawn.
        Updates game_over and winner accordingly.
        """
        if self._check_winner():
            self.winner = self.current_player
            self.game_over = True
        elif self.is_draw():
            self.game_over = True

    def _check_winner(self) -> bool:
        """Check if current player has won."""
        # Check rows
        for row in range(3):
            if all(self.board[row][col] == self.current_player for col in range(3)):
                return True

        # Check columns
        for col in range(3):
            if all(self.board[row][col] == self.current_player for row in range(3)):
                return True

        # Check diagonals
        if (self.board[0][0] == self.board[1][1] == self.board[2][2] == self.current_player or
                self.board[0][2] == self.board[1][1] == self.board[2][0] == self.current_player):
            return True

        return False

    def is_draw(self) -> bool:
        """Check if the game is a draw (no winner and board full)."""
        return all(self.board[row][col] is not None
                   for row in range(3) for col in range(3))

    def get_available_moves(self) -> List[Tuple[int, int]]:
        """Get list of available (row, col) moves."""
        return [(row, col) for row in range(3) for col in range(3)
                if self.board[row][col] is None]

    def get_board_state(self) -> List[List[Optional[str]]]:
        """Get current board state."""
        return [row.copy() for row in self.board]
```

```python
Handles board state, move validation, win/draw checking, and game reset.

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game rules."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.reset_game()

    def reset_game(self) -> None:
        """Reset the game state to initial conditions."""
        self.board: List[List[Optional[str]]] = [
            [None, None, None],
            [None, None, None],
            [None, None, None]
        ]
        self.current_player: str = 'X'  # X always starts first
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.

        Args:
            row: Row index (0-2)
            col: Column index (0-2)

        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if self.game_over or not self._is_valid_position(row, col):
            return False

        if self.board[row][col] is not None:
            return False

        self.board[row][col] = self.current_player
        self._check_game_state()
        if not self.game_over:
            self._switch_player()
        return True

    def _is_valid_position(self, row: int, col: int) -> bool:
        """Check if position indices are within valid range."""
        return 0 <= row < 3 and 0 <= col < 3

    def _switch_player(self) -> None:
        """Switch the current player between X and O."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _check_game_state(self) -> None:
        """
        Check if current player has won or if game is drawn.
        Updates game_over and winner accordingly.
        """
        if self._check_winner():
            self.winner = self.current_player
            self.game_over = True
        elif self.is_draw():
            self.game_over = True

    def _check_winner(self) -> bool:
        """Check if current player has won."""
        # Check rows
        for row in range(3):
            if all(self.board[row][col] == self.current_player for col in range(3)):
                return True

        # Check columns
        for col in range(3):
            if all(self.board[row][col] == self.current_player for row in range(3)):
                return True

        # Check diagonals
        if (self.board[0][0] == self.board[1][1] == self.board[2][2] == self.current_player or
                self.board[0][2] == self.board[1][1] == self.board[2][0] == self.current_player):
            return True

        return False

    def is_draw(self) -> bool:
        """Check if the game is a draw (no winner and board full)."""
        return all(self.board[row][col] is not None
                   for row in range(3) for col in range(3))

    def get_available_moves(self) -> List[Tuple[int, int]]:
        """Get list of available (row, col) moves."""
        return [(row, col) for row in range(3) for col in range(3)
                if self.board[row][col] is None]

    def get_board_state(self) -> List[List[Optional[str]]]:
        """Get current board state."""
        return [row.copy() for row in self.board]
```

    def handle_click(self, pos: Tuple[int, int]) -> None:
        """
        Handle mouse click events.
        
        Args:
            pos: (x, y) position of mouse click
        """
        # Check restart button click
        if 620 <= pos[1] <= 670 and 200 <= pos[0] <= 400:
            self.game.reset_game()
            return
            
        # Ignore clicks outside board or when game is over
        if pos[1] < 100 or pos[1] > 600 or self.game.game_over:
            return
            
        # Convert click position to board coordinates
        row = (pos[1] - 100) // 200
        col = pos[0] // 200
        
        # Make move if it's human player's turn
        if self.game.current_player == 'X':
            self.game.make_move(row, col)

```python
        """Test game initialization sets up correct default state."""
        expected_board: List[List[Optional[str]]] = [
            [None, None, None],
            [None, None, None],
            [None, None, None]
        ]
        self.assertEqual(self.game.board, expected_board)

    def test_reset_game(self) -> None:
        """Test reset_game returns game to initial state."""
        # Make some moves and change state
        self.game.make_move(0, 0)
        self.game.make_move(1, 1)
        self.game.reset_game()

        expected_board: List[List[Optional[str]]] = [
            [None, None, None],
            [None, None, None],
            [None, None, None]
        ]
        self.assertEqual(self.game.board, expected_board)
        self.assertEqual(self.game.current_player, 'X')
        self.assertFalse(self.game.game_over)
        self.assertIsNone(self.game.winner)

    def test_make_valid_move(self) -> None:
        """Test making a valid move updates the board."""
        result: bool = self.game.make_move(0, 0)
        self.assertTrue(result)
        self.assertEqual(self.game.board[0][0], 'X')
        self.assertEqual(self.game.current_player, 'O')

        self.assertFalse(result)
        self.assertEqual(self.game.board[0][0], 'X')  # Should remain X
```
