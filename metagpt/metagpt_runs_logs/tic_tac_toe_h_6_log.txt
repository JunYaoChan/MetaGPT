BOARD_SIZE = 3
GRID_LINE_WIDTH = 4
WINDOW_SIZE = 600
CELL_SIZE = WINDOW_SIZE // BOARD_SIZE
PLAYER_X = 'X'
PLAYER_O = 'O'
EMPTY = ''
LINE_COLOR = (23, 145, 135)
X_COLOR = (84, 84, 84)
O_COLOR = (242, 235, 211)
BG_COLOR = (28, 170, 156)
TEXT_COLOR = (255, 255, 255)


class Game:
    """Main game class handling Tic Tac Toe logic and rendering."""

    def __init__(self) -> None:
        """Initialize game state with empty board and starting player."""
        self.board: List[List[str]] = [[EMPTY for _ in range(BOARD_SIZE)] 
                                      for _ in range(BOARD_SIZE)]
        self.current_player: str = PLAYER_X
        self.game_over: bool = False
        self.winner: Optional[str] = None

    def handle_click(self, pos: Tuple[int, int]) -> None:
        """Handle player click on the game board.
        
        Args:
            pos: Tuple of (x, y) coordinates where player clicked
        """
        if self.game_over:
            return

        x, y = pos[0] // CELL_SIZE, pos[1] // CELL_SIZE

        if 0 <= x < BOARD_SIZE and 0 <= y < BOARD_SIZE and self.board[y][x] == EMPTY:
            self.board[y][x] = self.current_player
            if self.check_win():
                self.game_over = True
                self.winner = self.current_player
            elif self.is_board_full():
                self.game_over = True
            else:
                self.current_player = PLAYER_O if self.current_player == PLAYER_X else PLAYER_X

    def check_win(self) -> bool:
        """Check if current player has won the game.
        
        Returns:
            bool: True if current player has won, False otherwise
        """
        # Check rows
        for row in self.board:
            if all(cell == self.current_player for cell in row):
                return True

        # Check columns
        for col in range(BOARD_SIZE):
            if all(self.board[row][col] == self.current_player for row in range(BOARD_SIZE)):
                return True

        # Check diagonals
        if all(self.board[i][i] == self.current_player for i in range(BOARD_SIZE)):
            return True
        if all(self.board[i][BOARD_SIZE-1-i] == self.current_player for i in range(BOARD_SIZE)):
            return True

        return False

    def is_board_full(self) -> bool:
        """Check if the board is completely filled.
        
        Returns:
            bool: True if board is full, False otherwise
        """
        return all(cell != EMPTY for row in self.board for cell in row)

    def reset(self) -> None:
        """Reset the game to initial state."""
        self.board = [[EMPTY for _ in range(BOARD_SIZE)] 
                     for _ in range(BOARD_SIZE)]
        self.current_player = PLAYER_X
        self.game_over = False
        self.winner = None

    def draw(self, screen: pygame.Surface) -> None:
        """Draw the game board and current state.
        
        Args:
            screen: Pygame surface to draw on
        """
        screen.fill(BG_COLOR)
        self._draw_grid(screen)
        self._draw_marks(screen)
        self._draw_status(screen)

    def _draw_grid(self, screen: pygame.Surface) -> None:
        """Draw the Tic Tac Toe grid lines.
        
        Args:
            screen: Pygame surface to draw on
        """
        for i in range(1, BOARD_SIZE):
            # Vertical lines
            pygame.draw.line(
                screen, 
                LINE_COLOR, 
                (i * CELL_SIZE, 0), 
                (i * CELL_SIZE, WINDOW_SIZE), 
                GRID_LINE_WIDTH
            )
            # Horizontal lines
            pygame.draw.line(
                screen, 
                LINE_COLOR, 
                (0, i * CELL_SIZE), 
                (WINDOW_SIZE, i * CELL_SIZE), 
                GRID_LINE_WIDTH
            )

    def _draw_marks(self, screen: pygame.Surface) -> None:
        """Draw X and O marks on the board.
        
        Args:
            screen: Pygame surface to draw on
        """
        for y in range(BOARD_SIZE):
            for x in range(BOARD_SIZE):
                if self.board[y][x] == PLAYER_X:
                    self._draw_x(screen, x, y)
                elif self.board[y][x] == PLAYER_O:
                    self._draw_o(screen, x, y)

    def _draw_x(self, screen: pygame.Surface, x: int, y: int) -> None:
        """Draw an X mark at the specified cell.
        
        Args:
            screen: Pygame surface to draw on
            x: Column index (0-2)
            y: Row index (0-2)
        """
        padding = CELL_SIZE // 4
        pygame.draw.line(
            screen, 
            X_COLOR, 
            (x * CELL_SIZE + padding, y * CELL_SIZE + padding),
            ((x + 1) * CELL_SIZE - padding, (y + 1) * CELL_SIZE - padding),
            GRID_LINE_WIDTH * 2
        )
        pygame.draw.line(
            screen, 
            X_COLOR, 
            ((x + 1) * CELL_SIZE - padding, y * CELL_SIZE + padding),
            (x * CELL_SIZE + padding, (y + 1) * CELL_SIZE - padding),
            GRID_LINE_WIDTH * 2
        )

    def _draw_o(self, screen: pygame.Surface, x: int, y: int) -> None:
        """Draw an O mark at the specified cell.
        
        Args:
            screen: Pygame surface to draw on
            x: Column index (0-2)
            y: Row index (0-2)
        """
        radius = CELL_SIZE // 3
        pygame.draw.circle(
            screen,
            O_COLOR,
            (x * CELL_SIZE + CELL_SIZE // 2, y * CELL_SIZE + CELL_SIZE // 2),
            radius,
            GRID_LINE_WIDTH
        )

    def _draw_status(self, screen: pygame.Surface) -> None:
        """Draw game status (current player or winner).
        
        Args:
            screen: Pygame surface to draw on
        """
        font = pygame.font.Font(None, 36)
        
        if self.game_over:
            if self.winner:
                text = f"Player {self.winner} wins!"
            else:
                text = "Game ended in a tie!"
        else:
            text = f"Player {self.current_player}'s turn"
            
        text_surface = font.render(text, True, TEXT_COLOR)
        text_rect = text_surface.get_rect(center=(WINDOW_SIZE // 2, WINDOW_SIZE - 30))
        screen.blit(text_surface, text_rect)

"""
Test cases for Tic Tac Toe game implementation using unittest.
"""

import unittest
from unittest.mock import MagicMock
import pygame
from tic_tac_toe_h_6.game import Game
from tic_tac_toe_h_6.constants import (
    GRID_SIZE,
    CELL_SIZE,
    PLAYER_X,
    PLAYER_O,
    EMPTY,
    GAME_STATE_RUNNING,
    GAME_STATE_OVER
)


class TestGame(unittest.TestCase):
    """Test cases for the Game class."""

    def setUp(self) -> None:
        """Set up test fixtures."""
        self.game = Game()
        self.mock_screen = MagicMock()

    def tearDown(self) -> None:
        """Clean up test fixtures."""
        del self.game

    ## INITIALIZATION TESTS
    def test_initialization(self) -> None:
        """Test game initialization with empty board and correct starting player."""
        self.assertEqual(self.game.current_player, PLAYER_X)
        self.assertFalse(self.game.game_over)
        self.assertIsNone(self.game.winner)
        for row in self.game.board:
            self.assertTrue(all(cell == EMPTY for cell in row))

    ## BOARD HANDLING TESTS
    def test_handle_click_valid_empty_cell(self) -> None:
        """Test handling click on valid empty cell."""
        pos = (CELL_SIZE // 2, CELL_SIZE // 2)  # Center of first cell
        self.game.handle_click(pos)
        self.assertEqual(self.game.board[0][0], PLAYER_X)
        self.assertEqual(self.game.current_player, PLAYER_O)

    def test_handle_click_out_of_bounds(self) -> None:
        """Test handling click outside game board boundaries."""
        initial_board = [row[:] for row in self.game.board]
        # Click outside grid (negative)
        self.game.handle_click((-10, -10))
        # Click outside grid (positive)
        self.game.handle_click((GRID_SIZE * CELL_SIZE + 10, GRID_SIZE * CELL_SIZE + 10))
        self.assertEqual(self.game.board, initial_board)

    def test_handle_click_on_occupied_cell(self) -> None:
        """Test handling click on already occupied cell."""
        self.game.board[0][0] = PLAYER_X
        pos = (CELL_SIZE // 2, CELL_SIZE // 2)  # Center of first cell
        self.game.handle_click(pos)
        self.assertEqual(self.game.board[0][0], PLAYER_X)  # Should not change
        self.assertEqual(self.game.current_player, PLAYER_X)  # Should not switch

        """Test handling click when game is already over."""
        self.game.game_over = True
        initial_board = [row[:] for row in self.game.board]
        pos = (CELL_SIZE // 2, CELL_SIZE // 2)
        self.game.handle_click(pos)
        self.assertEqual(self.game.board, initial_board)

        """Test horizontal win condition detection."""
        # Set up horizontal win for X in first row
        for i in range(GRID_SIZE):
            self.game.board[0][i] = PLAYER_X
        self.game.current_player = PLAYER_X
        self.assertTrue(self.game.check_win())

    def test_check_win_vertical(self) -> None:
        """Test vertical win condition detection."""
        # Set up vertical win for O in first column
        for i in range(GRID_SIZE):
            self.game.board[i][0] = PLAYER_O
        self.game.current_player = PLAYER_O
        self.assertTrue(self.game.check_win())

    def test_check_win_diagonal(self) -> None:
        """Test diagonal win condition detection."""
        # Set up main diagonal win for X
        for i in range(GRID_SIZE):
            self.game.board[i][i] = PLAYER_X
        self.game.current_player = PLAYER_X
        self.assertTrue(self.game.check_win())

        # Set up anti-diagonal win for O
        self.game.reset()
        for i in range(GRID_SIZE):
            self.game.board[i][GRID_SIZE-1-i] = PLAYER_O
        self.game.current_player = PLAYER_O
        self.assertTrue(self.game.check_win())

        """Test no win condition detected."""

    def test_is_board_full_true(self) -> None:
        """Test full board detection."""
        # Fill board with alternating players
        for y in range(GRID_SIZE):
            for x in range(GRID_SIZE):
                self.game.board[y][x] = PLAYER_X if (x + y) % 2 == 0 else PLAYER_O
        self.assertTrue(self.game.is_board_full())

        """Test not full board detection."""
        self.assertFalse(self.game.is_board_full())
        # Partially filled board

        self.assertFalse(self.game.is_board_full())
