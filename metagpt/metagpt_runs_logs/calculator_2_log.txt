## main.py
import tkinter as tk

class Calculator:
    """A simple calculator class using tkinter for GUI."""

    def __init__(self) -> None:
        """Initialize the calculator and create the main window."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='ridge')
        self.display.grid(row=0, column=0, columnspan=4)
        self.buttons = []
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.delete(0, tk.END)  # Clear the display

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            '0', 'C', '=', '+'
        ]
        
        # Create buttons and place them in the grid
        for i, text in enumerate(button_texts):
            button = tk.Button(self.root, text=text, width=5, height=2, command=lambda t=text: self.add_to_display(t))
            button.grid(row=(i // 4) + 1, column=i % 4)
            self.buttons.append(button)

    def add_to_display(self, value: str) -> None:
        """Add a value to the display."""
        if value == 'C':
            self.clear_display()
        elif value == '=':
            self.calculate()
        else:
            self.display.insert(tk.END, value)

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Calculate the result of the expression in the display."""
        try:
            result = eval(self.display.get())  # Evaluate the expression
            self.clear_display()  # Clear the display before showing result
            self.display.insert(tk.END, str(result))  # Show result
        except Exception as e:
            self.clear_display()  # Clear display on error
            self.display.insert(tk.END, "Error")  # Show error message

    def run(self) -> None:
        """Run the main loop of the calculator."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()  # Create a Calculator instance
    calculator.run()  # Run the calculator
