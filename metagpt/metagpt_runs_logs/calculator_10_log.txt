import tkinter as tk
from tkinter import messagebox

class Calculator:
    """A simple calculator class using tkinter for GUI."""

    def __init__(self):
        """Initialize the calculator with default values."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='ridge')
        self.buttons = []
        self.current_input = ""
        self.operation = ""
        self.first_number = 0.0

        self.create_display()
        self.create_buttons()

    def create_display(self):
        """Create the display area for the calculator."""
        self.display.grid(row=0, column=0, columnspan=4)
        self.display.insert(0, "0")

    def create_buttons(self):
        """Create buttons for digits and operations."""
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        row_val = 1
        col_val = 0
        
        for text in button_texts:
            button = tk.Button(self.root, text=text, width=5, height=2, font=('Arial', 18),
                               command=lambda t=text: self.button_click(t))
            button.grid(row=row_val, column=col_val)
            self.buttons.append(button)
            col_val += 1
            if col_val > 3:  # Move to the next row after 4 buttons
                col_val = 0
                row_val += 1

    def button_click(self, number: str):
        """Handle button click for numbers and operations."""
        if number == 'C':
            self.clear_click()
        elif number == '=':
            self.equals_click()
        else:
            self.current_input += number
            self.update_display()

    def operation_click(self, operation: str):
        """Handle operation button clicks."""
        if self.current_input:
            self.first_number = float(self.current_input)
            self.operation = operation
            self.current_input = ""
            self.update_display()

    def equals_click(self):
        """Calculate the result of the current operation."""
        if self.current_input and self.operation:
            second_number = float(self.current_input)
            try:
                if self.operation == '+':
                    result = self.first_number + second_number
                elif self.operation == '-':
                    result = self.first_number - second_number
                elif self.operation == '*':
                    result = self.first_number * second_number
                elif self.operation == '/':
                    if second_number == 0:
                        raise ZeroDivisionError("Cannot divide by zero.")
                    result = self.first_number / second_number
                else:
                    result = second_number  # No operation, just return the second number

                self.current_input = str(result)
                self.operation = ""
                self.update_display()
            except ZeroDivisionError as e:
                messagebox.showerror("Error", str(e))
                self.clear_click()

    def clear_click(self):
        """Clear the current input and reset the calculator."""
        self.current_input = ""
        self.operation = ""
        self.first_number = 0.0
        self.update_display()

    def update_display(self):
        """Update the display with the current input."""
        self.display.delete(0, tk.END)
        self.display.insert(0, self.current_input if self.current_input else "0")

    def run(self):
        """Run the main loop of the calculator."""
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
