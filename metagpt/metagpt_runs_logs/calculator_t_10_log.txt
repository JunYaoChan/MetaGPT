## main.py
import tkinter as tk  # Importing the Tkinter library for GUI

class Calculator:
    """A simple calculator class to perform basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator and create the main window."""
        self.root = tk.Tk()  # Create the main window
        self.root.title("Simple Calculator")  # Set the window title
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='ridge')  # Create display
        self.display.grid(row=0, column=0, columnspan=4)  # Place display in the grid
        self.buttons = {}  # Dictionary to hold button references
        self.create_buttons()  # Create buttons for the calculator
        self.create_display()  # Create the display for the calculator

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.insert(0, "0")  # Set default display value

    def create_buttons(self) -> None:
        """Create buttons for numbers and operations."""
        # Define button layout
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        # Create buttons and place them in the grid
        row, col = 1, 0
        for text in button_texts:
            self.buttons[text] = tk.Button(self.root, text=text, width=5, height=2,
                                            command=lambda t=text: self.button_click(t))
            self.buttons[text].grid(row=row, column=col)
            col += 1
            if col > 3:  # Move to the next row after 4 buttons
                col = 0
                row += 1

    def button_click(self, char: str) -> None:
        """Handle button click events."""
        if char == 'C':
            self.clear_display()  # Clear display if 'C' is pressed
        elif char == '=':
            self.calculate()  # Calculate result if '=' is pressed
        else:
            current_text = self.display.get()  # Get current display text
            if current_text == "0":  # Replace default '0' if it's the first input
                self.display.delete(0, tk.END)  # Clear display
            self.display.insert(tk.END, char)  # Append clicked button text to display

    def clear_display(self) -> None:
        """Clear the calculator display."""
        self.display.delete(0, tk.END)  # Clear the display
        self.display.insert(0, "0")  # Reset to default value

    def calculate(self) -> None:
        """Evaluate the expression in the display and update the display with the result."""
        try:
            expression = self.display.get()  # Get the expression from the display
            result = eval(expression)  # Evaluate the expression
            self.clear_display()  # Clear the display
            self.display.insert(0, str(result))  # Show the result
        except Exception as e:
            self.clear_display()  # Clear display on error
            self.display.insert(0, "Error")  # Show error message

    def run(self) -> None:
        """Run the calculator application."""
        self.root.mainloop()  # Start the Tkinter event loop

if __name__ == "__main__":
    calculator = Calculator()  # Create an instance of the Calculator
    calculator.run()  # Run the calculator application
