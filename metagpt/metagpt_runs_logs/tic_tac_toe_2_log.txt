import numpy as np

class Game:
    """Class to manage the state and rules of the Tic Tac Toe game."""

    def __init__(self) -> None:
        """Initialize the game board, current player, and game state."""
        self.board: np.ndarray = np.full((3, 3), '', dtype=str)  # 3x3 board initialized with empty strings
        self.current_player: str = 'X'  # Default starting player
        self.game_over: bool = False  # Game state flag
        self.winner: str = ''  # To store the winner

    def make_move(self, row: int, col: int) -> bool:
        """Make a move on the board.

        Args:
            row (int): The row index for the move.
            col (int): The column index for the move.

        Returns:
            bool: True if the move was successful, False otherwise.
        """
        if self.board[row, col] == '' and not self.game_over:
            self.board[row, col] = self.current_player  # Place the current player's mark
            return True
        return False

    def check_winner(self) -> str:
        """Check for a winner or a draw.

        Returns:
            str: The winner ('X' or 'O'), 'Draw' if it's a draw, or '' if the game is still ongoing.
        """
        # Check rows, columns, and diagonals for a win
        for i in range(3):
            if self.board[i, :].tolist().count(self.current_player) == 3:
                self.winner = self.current_player
                self.game_over = True
                return self.winner
            if self.board[:, i].tolist().count(self.current_player) == 3:
                self.winner = self.current_player
                self.game_over = True
                return self.winner

        if (self.board[0, 0] == self.current_player and
            self.board[1, 1] == self.current_player and
            self.board[2, 2] == self.current_player):
            self.winner = self.current_player
            self.game_over = True
            return self.winner

        if (self.board[0, 2] == self.current_player and
            self.board[1, 1] == self.current_player and
            self.board[2, 0] == self.current_player):
            self.winner = self.current_player
            self.game_over = True
            return self.winner

        if np.all(self.board != ''):  # Check for a draw
            self.game_over = True
            return 'Draw'

        return ''  # Game is still ongoing

    def reset(self) -> None:
        """Reset the game to the initial state."""
        self.board.fill('')  # Clear the board
        self.current_player = 'X'  # Reset to starting player
        self.game_over = False  # Reset game over flag
        self.winner = ''  # Clear winner

    def is_draw(self) -> bool:
        """Check if the game is a draw.

        Returns:
            bool: True if the game is a draw, False otherwise.
        """
        return np.all(self.board != '') and self.winner == ''

        """

        """Convert screen coordinates to grid cell indices.
        
        Args:
            pos: Tuple of (x, y) screen coordinates

            Tuple of (row, col) grid indices
        """
        x, y = pos

        background.fill((0, 0, 0, 180))

"""

    """Test cases for the Main class"""

    def setUp(self) -> None:
        """Set up test fixtures"""
        self.main = Main()

    def test_initialization(self) -> None:
        """Test that Main initializes with correct components"""
        self.assertIsInstance(self.main.game, Game)
        self.assertIsInstance(self.main.ai, AI)
        self.assertIsInstance(self.main.ui, UI)
        self.assertEqual(self.main.ai.difficulty, 'easy')

    @patch('pygame.init')
    @patch('pygame.display.set_mode')
    @patch('pygame.display.set_caption')
    @patch('pygame.time.Clock')
    @patch('pygame.quit')
    @patch.object(UI, 'handle_events')
    @patch.object(UI, 'update')
    @patch.object(UI, 'draw_board')
    @patch.object(UI, 'show_message')
    @patch.object(Game, 'check_winner')
    @patch.object(Game, 'reset')
    def test_run_game_loop(
        self,
        mock_reset: MagicMock,
        mock_check_winner: MagicMock,
        mock_show_message: MagicMock,
        mock_draw_board: MagicMock,
        mock_update: MagicMock,
        mock_handle_events: MagicMock,
        mock_clock: MagicMock,
        mock_quit: MagicMock,
        mock_set_caption: MagicMock,
        mock_set_mode: MagicMock,
        mock_init: MagicMock
    ) -> None:
        """Test the main game loop execution"""
        # Setup mocks
        mock_screen = MagicMock()
        mock_set_mode.return_value = mock_screen
        mock_clock_instance = MagicMock()
        mock_clock.return_value = mock_clock_instance

        # Simulate game loop running then stopping
        mock_handle_events.side_effect = [None, None, pygame.event.Event(pygame.QUIT)]

        # Test with no winner
        mock_check_winner.return_value = None
        self.main.run()

        # Verify pygame initialization
        mock_init.assert_called_once()
        mock_set_mode.assert_called_once_with((300, 300))
        mock_set_caption.assert_called_once_with("Tic Tac Toe")

        # Verify game loop operations
        self.assertEqual(mock_handle_events.call_count, 3)
        mock_update.assert_called()
        mock_screen.fill.assert_called_with((255, 255, 255))
        mock_draw_board.assert_called()
        mock_clock_instance.tick.assert_called_with(30)
        mock_quit.assert_called_once()

        # Test with winner
        mock_check_winner.return_value = 'X'
        mock_handle_events.side_effect = [None, pygame.event.Event(pygame.QUIT)]
        self.main.run()
        mock_show_message.assert_called_with("Winner: X")
        mock_reset.assert_called()

        # Test with draw
        mock_check_winner.return_value = 'Draw'
        mock_handle_events.side_effect = [None, pygame.event.Event(pygame.QUIT)]
        self.main.run()
        mock_show_message.assert_called_with("It's a Draw!")

    def tearDown(self) -> None:
        """Clean up after tests"""
        pygame.quit()

    unittest.main()

## game.py
import numpy as np

class Game:
    """Class to manage the state and rules of the Tic Tac Toe game."""

    def __init__(self) -> None:
        """Initialize the game board, current player, and game state."""
        self.board: np.ndarray = np.full((3, 3), '', dtype=str)  # 3x3 board initialized with empty strings
        self.current_player: str = 'X'  # Default starting player
        self.game_over: bool = False  # Game state flag
        self.winner: str = ''  # To store the winner

    def make_move(self, row: int, col: int) -> bool:
        """Make a move on the board.

        Args:
            row (int): The row index for the move.
            col (int): The column index for the move.

        Returns:
            bool: True if the move was successful, False otherwise.
        """
        if self.board[row, col] == '' and not self.game_over:
            self.board[row, col] = self.current_player  # Place the current player's mark
            self._switch_player()
            return True
        return False

    def _switch_player(self) -> None:
        """Switch the current player."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def check_winner(self) -> str:
        """Check for a winner or a draw.

        Returns:
            str: The winner ('X' or 'O'), 'Draw' if it's a draw, or '' if the game is still ongoing.
        """
        # Check rows, columns, and diagonals for a win
        for i in range(3):
            if self.board[i, :].tolist().count('X') == 3:
                self.winner = 'X'
                self.game_over = True
                return self.winner
            if self.board[i, :].tolist().count('O') == 3:
                self.winner = 'O'
                self.game_over = True
                return self.winner
            if self.board[:, i].tolist().count('X') == 3:
                self.winner = 'X'
                self.game_over = True
                return self.winner
            if self.board[:, i].tolist().count('O') == 3:
                self.winner = 'O'
                self.game_over = True
                return self.winner

        # Check diagonals
        if (self.board[0, 0] == self.board[1, 1] == self.board[2, 2] != ''):
            self.winner = self.board[0, 0]
            self.game_over = True
            return self.winner

        if (self.board[0, 2] == self.board[1, 1] == self.board[2, 0] != ''):
            self.winner = self.board[0, 2]
            self.game_over = True
            return self.winner

        # Check for draw
        if np.all(self.board != '') and self.winner == '':
            self.game_over = True
            return 'Draw'

        return ''  # Game is still ongoing

    def reset(self) -> None:
        """Reset the game to the initial state."""
        self.board.fill('')  # Clear the board
        self.current_player = 'X'  # Reset to starting player
        self.game_over = False  # Reset game over flag
        self.winner = ''  # Clear winner

    def is_draw(self) -> bool:
        """Check if the game is a draw.

        Returns:
            bool: True if the game is a draw, False otherwise.
        """
        return np.all(self.board != '') and self.winner == ''
