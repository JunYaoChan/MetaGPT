```"""
Tic Tac Toe game logic implementation.
Handles board state, move validation, winner checking, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game logic."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.reset_game()

    def reset_game(self) -> None:
        """Reset the game state to initial conditions."""
        self.board: List[List[Optional[str]]] = [
            [None, None, None],
            [None, None, None],
            [None, None, None]
        ]
        self.current_player: str = 'X'  # X always starts first
        self.game_state: str = 'ongoing'  # ongoing, X_won, O_won, draw

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.

        Args:
            row: Row index (0-2)
            col: Column index (0-2)

        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if (self.game_state != 'ongoing' or
                not (0 <= row < 3 and 0 <= col < 3) or
                self.board[row][col] is not None):
            return False

        self.board[row][col] = self.current_player
        self._update_game_state()
        self._switch_player()
        return True

    def _switch_player(self) -> None:
        """Switch the current player after a valid move."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _update_game_state(self) -> None:
        """Check if the current move resulted in a win or draw."""
        winner = self.check_winner()
        if winner:
            self.game_state = f'{winner}_won'
        elif self.is_board_full():
            self.game_state = 'draw'

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.

        Returns:
            Optional[str]: 'X' or 'O' if winner found, None otherwise
        """
        # Check rows and columns
        for i in range(3):
            if (self.board[i][0] == self.board[i][1] == self.board[i][2] and
                    self.board[i][0] is not None):
                return self.board[i][0]
            if (self.board[0][i] == self.board[1][i] == self.board[2][i] and
                    self.board[0][i] is not None):
                return self.board[0][i]

        # Check diagonals
        if (self.board[0][0] == self.board[1][1] == self.board[2][2] and
                self.board[0][0] is not None):
            return self.board[0][0]
        if (self.board[0][2] == self.board[1][1] == self.board[2][0] and
                self.board[0][2] is not None):
            return self.board[0][2]

        return None

    def is_board_full(self) -> bool:
        """
        Check if the board is completely filled.

        Returns:
            bool: True if board is full, False otherwise
        """
        return all(cell is not None for row in self.board for cell in row)

    def get_available_moves(self) -> List[Tuple[int, int]]:
        """
        Get all available moves on the current board.

        Returns:

```"""
Tic Tac Toe game logic implementation.
Handles board state, move validation, winner checking, and game reset.
"""

from typing import List, Optional, Tuple


class Game:
    """Main game class managing the board state and game logic."""

    def __init__(self) -> None:
        """Initialize a new game with empty board and starting player."""
        self.reset_game()

    def reset_game(self) -> None:
        """Reset the game state to initial conditions."""
        self.board: List[List[Optional[str]]] = [
            [None, None, None],
            [None, None, None],
            [None, None, None]
        ]
        self.current_player: str = 'X'  # X always starts first
        self.game_state: str = 'ongoing'  # ongoing, X_won, O_won, draw

    def make_move(self, row: int, col: int) -> bool:
        """
        Attempt to make a move at the specified position.

        Args:
            row: Row index (0-2)
            col: Column index (0-2)

        Returns:
            bool: True if move was valid and made, False otherwise
        """
        if (self.game_state != 'ongoing' or
                not (0 <= row < 3 and 0 <= col < 3) or
                self.board[row][col] is not None):
            return False

        self.board[row][col] = self.current_player
        self._update_game_state()
        self._switch_player()
        return True

    def _switch_player(self) -> None:
        """Switch the current player after a valid move."""
        self.current_player = 'O' if self.current_player == 'X' else 'X'

    def _update_game_state(self) -> None:
        """Check if the current move resulted in a win or draw."""
        winner = self.check_winner()
        if winner:
            self.game_state = f'{winner}_won'
        elif self.is_board_full():
            self.game_state = 'draw'

    def check_winner(self) -> Optional[str]:
        """
        Check if there's a winner on the current board.

        Returns:
            Optional[str]: 'X' or 'O' if winner found, None otherwise
        """
        # Check rows and columns
        for i in range(3):
            if (self.board[i][0] == self.board[i][1] == self.board[i][2] and
                    self.board[i][0] is not None):
                return self.board[i][0]
            if (self.board[0][i] == self.board[1][i] == self.board[2][i] and
                    self.board[0][i] is not None):
                return self.board[0][i]

        # Check diagonals
        if (self.board[0][0] == self.board[1][1] == self.board[2][2] and
                self.board[0][0] is not None):
            return self.board[0][0]
        if (self.board[0][2] == self.board[1][1] == self.board[2][0] and
                self.board[0][2] is not None):
            return self.board[0][2]

        return None

    def is_board_full(self) -> bool:
        """
        Check if the board is completely filled.

        Returns:
            bool: True if board is full, False otherwise
        """
        return all(cell is not None for row in self.board for cell in row)

    def get_available_moves(self) -> List[Tuple[int, int]]:
        """
        Get all available moves on the current board.

        Returns:
            List[Tuple[int, int]]: List of (row, col) tuples of empty cells
        """
        return [(i, j) for i in range(3) for j in range(3)
                if self.board[i][j] is None]

    def get_board_copy(self) -> List[List[Optional[str]]]:
        """
        Create a deep copy of the current board state.

        Returns:
            List[List[Optional[str]]]: Copy of the board
        """
        return [row.copy() for row in self.board]

```"""
AI module for Tic Tac Toe game implementing different difficulty levels.
"""

import random
from typing import Tuple
from game import Game

class AI:
    """
    AI player for Tic Tac Toe with three difficulty levels:
    - Easy: Random moves
    - Medium: Basic blocking strategy
    - Hard: Minimax algorithm
    """
    
    def __init__(self, difficulty: str = 'medium') -> None:
        """
        Initialize AI with specified difficulty level.
        
        Args:
            difficulty: AI difficulty level ('easy', 'medium', or 'hard')
        """
        self._difficulty = difficulty.lower()
        self._valid_difficulties = ['easy', 'medium', 'hard']
        
        if self._difficulty not in self._valid_difficulties:
            raise ValueError(f"Invalid difficulty. Must be one of {self._valid_difficulties}")

    def set_difficulty(self, difficulty: str) -> None:
        """
        Set the AI difficulty level.
        
        Args:
            difficulty: New difficulty level ('easy', 'medium', or 'hard')
        """
        if difficulty.lower() not in self._valid_difficulties:
            raise ValueError(f"Invalid difficulty. Must be one of {self._valid_difficulties}")
        self._difficulty = difficulty.lower()

    def get_move(self, game: Game) -> Tuple[int, int]:
        """
        Get the AI's next move based on current game state and difficulty level.
        
        Args:
            game: Current game instance
            
        Returns:
            Tuple of (row, col) coordinates for the next move
        """
        if self._difficulty == 'easy':
            return self._get_random_move(game)
        elif self._difficulty == 'medium':
            return self._get_medium_move(game)
        else:  # hard
            return self._get_hard_move(game)

    def _get_random_move(self, game: Game) -> Tuple[int, int]:
        """
        Get a random valid move from available spaces.
        
        Args:
            game: Current game instance
            
        Returns:
            Random valid (row, col) move
        """
        empty_cells = [
            (i, j) 
            for i in range(3) 
            for j in range(3) 
            if game.board[i][j] == ' '
        ]
        return random.choice(empty_cells)

    def _get_medium_move(self, game: Game) -> Tuple[int, int]:
        """
        Get a move using basic strategy (win if possible, block if needed, else random).
        
        Args:
            game: Current game instance
            
        Returns:
            (row, col) coordinates for the move
        """
        # First check for winning move
        for i in range(3):
            for j in range(3):
                if game.board[i][j] == ' ':
                    # Simulate move
                    game.board[i][j] = 'O'
                    if game.check_winner() == 'O':
                        game.board[i][j] = ' '  # Undo simulation
                        return (i, j)
                    game.board[i][j] = ' '  # Undo simulation

        # Then check for blocking move
        for i in range(3):
            for j in range(3):
                if game.board[i][j] == ' ':
                    # Simulate opponent move
                    game.board[i][j] = 'X'
                    if game.check_winner() == 'X':
                        game.board[i][j] = ' '  # Undo simulation
                        return (i, j)
                    game.board[i][j] = ' '  # Undo simulation

        # If no immediate win/block, return random move
        return self._get_random_move(game)

    def _get_hard_move(self, game: Game) -> Tuple[int, int]:
        """
        Get the best move using minimax algorithm.
        
        Args:
            game: Current game instance
            
        Returns:
            (row, col) coordinates for the best move
        """
        best_score = -float('inf')
        best_move = (-1, -1)
        
        for i in range(3):
            for j in range(3):
                if game.board[i][j] == ' ':
                    game.board[i][j] = 'O'
                    score = self._minimax(game, 0, False)
                    game.board[i][j] = ' '
                    if score > best_score:
                        best_score = score
                        best_move = (i, j)
        
        return best_move

    def _minimax(self, game: Game, depth: int, is_maximizing: bool) -> int:
        """
        Minimax algorithm implementation for determining optimal moves.
        
        Args:
            game: Current game state
            depth: Current depth in the game tree
            is_maximizing: Whether current player is maximizing (AI) or minimizing (human)
            
        Returns:
            Score for the current board state
        """
        result = game.check_winner()
        if result == 'O':
            return 10 - depth
        elif result == 'X':
            return depth - 10
        elif game.is_board_full():
            return 0

        if is_maximizing:
            best_score = -float('inf')
            for i in range(3):
                for j in range(3):
                    if game.board[i][j] == ' ':
                        game.board[i][j] = 'O'
                        score = self._minimax(game, depth + 1, False)
                        game.board[i][j] = ' '
                        best_score = max(score, best_score)
            return best_score
        else:
            best_score = float('inf')
            for i in range(3):
                for j in range(3):
                    if game.board[i][j] == ' ':
                        game.board[i][j] = 'X'
                        score = self._minimax(game, depth + 1, True)
                        game.board[i][j] = ' '
                        best_score = min(score, best_score)
            return best_score

        """Draw the Tic Tac Toe grid lines."""
        # Vertical lines
        for i in range(1, 3):

        # Horizontal lines

    def _draw_symbols(self) -> None:

                    self._draw_x(row, col)
                elif self.game.board[row][col] == 'O':
                    self._draw_o(row, col)
                    
    def _draw_x(self, row: int, col: int) -> None:
        """Draw X symbol at specified cell."""
        center_x = col * self.CELL_SIZE + self.CELL_SIZE // 2

        offset = self.CELL_SIZE // 3

            self.SYMBOL_WIDTH
        )
        
    def _draw_o(self, row: int, col: int) -> None:
        """Draw O symbol at specified cell."""
        center_x = col * self.CELL_SIZE + self.CELL_SIZE // 2

            self.BLUE,
            (center_x, center_y),

            self.SYMBOL_WIDTH
        )
