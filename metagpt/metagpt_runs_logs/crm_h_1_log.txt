import sqlite3  # Importing SQLite for database operations
import streamlit as st  # Importing Streamlit for the user interface

class CustomerManager:
    """Class to manage customer data in the SQLite database."""

    def __init__(self, db_name: str = "customers.db") -> None:
        """Initialize the CustomerManager with a database name.

        Args:
            db_name (str): The name of the SQLite database file.
        """
        self.db_name: str = db_name
        self.conn: sqlite3.Connection = sqlite3.connect(self.db_name)
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()  # Create the customer table if it doesn't exist

    def create_table(self) -> None:
        """Create the customers table in the database if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                birthday TEXT NOT NULL,
                age INTEGER NOT NULL,
                sex TEXT NOT NULL,
                phone TEXT NOT NULL
            )
        ''')
        self.conn.commit()  # Commit the changes to the database

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Add a new customer to the database.

        Args:
            name (str): The name of the customer.
            birthday (str): The birthday of the customer.
            age (int): The age of the customer.
            sex (str): The sex of the customer.
            phone (str): The phone number of the customer.
        """
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone)
            VALUES (?, ?, ?, ?, ?)
        ''', (name, birthday, age, sex, phone))
        self.conn.commit()  # Commit the changes to the database

    def query_customer(self, name: str) -> dict:
        """Query a customer by name.

        Args:
            name (str): The name of the customer to query.

        Returns:
            dict: A dictionary containing customer data if found, else empty.
        """
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()  # Fetch the first matching row
        if row:
            return {
                "id": row[0],
                "name": row[1],
                "birthday": row[2],
                "age": row[3],
                "sex": row[4],
                "phone": row[5]
            }
        return {}  # Return an empty dictionary if no customer found

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Update an existing customer's information.

        Args:
            name (str): The name of the customer to update.
            birthday (str): The new birthday of the customer.
            age (int): The new age of the customer.
            sex (str): The new sex of the customer.
            phone (str): The new phone number of the customer.
        """
        self.cursor.execute('''
            UPDATE customers
            SET birthday = ?, age = ?, sex = ?, phone = ?
            WHERE name = ?
        ''', (birthday, age, sex, phone, name))
        self.conn.commit()  # Commit the changes to the database

    def delete_customer(self, name: str) -> None:
        """Delete a customer from the database.

        Args:
            name (str): The name of the customer to delete.
        """
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()  # Commit the changes to the database

    def close_connection(self) -> None:
        """Close the database connection."""
        self.conn.close()  # Close the connection to the database


class StreamlitUI:
    """Class to manage the Streamlit user interface."""

    def __init__(self, customer_manager: CustomerManager) -> None:
        """Initialize the Streamlit UI with a CustomerManager instance.

        Args:
            customer_manager (CustomerManager): An instance of CustomerManager.
        """
        self.customer_manager: CustomerManager = customer_manager

    def render(self) -> None:
        """Render the Streamlit user interface."""
        st.title("Customer Relationship Management System")  # Set the title of the app

        # User selection for operations
        operation = st.selectbox("Select Operation", ["Add Customer", "Query Customer", "Update Customer", "Delete Customer"])

        if operation == "Add Customer":
            self.add_customer_ui()
        elif operation == "Query Customer":
            self.query_customer_ui()
        elif operation == "Update Customer":
            self.update_customer_ui()
        elif operation == "Delete Customer":
            self.delete_customer_ui()

    def add_customer_ui(self) -> None:
        """User interface for adding a customer."""
        st.subheader("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")

        if st.button("Add Customer"):
            self.customer_manager.add_customer(name, birthday, age, sex, phone)
            st.success("Customer added successfully!")

    def query_customer_ui(self) -> None:
        """User interface for querying a customer."""
        st.subheader("Query Customer")
        name = st.text_input("Name")

        if st.button("Query Customer"):
            customer_data = self.customer_manager.query_customer(name)
            if customer_data:
                st.write(customer_data)
            else:
                st.warning("Customer not found.")

    def update_customer_ui(self) -> None:
        """User interface for updating a customer."""
        st.subheader("Update Customer")
        name = st.text_input("Name")

        if st.button("Fetch Customer"):
            customer_data = self.customer_manager.query_customer(name)
            if customer_data:
                birthday = st.text_input("Birthday (YYYY-MM-DD)", value=customer_data["birthday"])
                age = st.number_input("Age", min_value=0, value=customer_data["age"])
                sex = st.selectbox("Sex", ["Male", "Female", "Other"], index=["Male", "Female", "Other"].index(customer_data["sex"]))
                phone = st.text_input("Phone", value=customer_data["phone"])

                if st.button("Update Customer"):
                    self.customer_manager.update_customer(name, birthday, age, sex, phone)
                    st.success("Customer updated successfully!")
            else:
                st.warning("Customer not found.")

    def delete_customer_ui(self) -> None:
        """User interface for deleting a customer."""
        st.subheader("Delete Customer")
        name = st.text_input("Name")

        if st.button("Delete Customer"):
            self.customer_manager.delete_customer(name)
            st.success("Customer deleted successfully!")


def main() -> None:
    """Main function to run the Streamlit application."""
    customer_manager = CustomerManager()  # Initialize the CustomerManager
    ui = StreamlitUI(customer_manager)  # Initialize the Streamlit UI
    ui.render()  # Render the UI


if __name__ == "__main__":
    main()  # Run the main function
