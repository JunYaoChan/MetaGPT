def _draw_board(screen: pygame.Surface) -> None:
    """Draw the Tic Tac Toe board grid.
    
    Args:
        screen: The pygame surface to draw on.
    """
    from constants import (
        LINE_COLOR,
        LINE_WIDTH,
        BOARD_ROWS,
        BOARD_COLS,
        CELL_SIZE
    )
    
    # Draw horizontal lines
    for row in range(1, BOARD_ROWS):
        pygame.draw.line(
            screen,
            LINE_COLOR,
            (0, row * CELL_SIZE),
            (SCREEN_WIDTH, row * CELL_SIZE),
            LINE_WIDTH
        )
    
    # Draw vertical lines
    for col in range(1, BOARD_COLS):
        pygame.draw.line(
            screen,
            LINE_COLOR,
            (col * CELL_SIZE, 0),
            (col * CELL_SIZE, SCREEN_HEIGHT),
            LINE_WIDTH
        )


def _draw_markers(screen: pygame.Surface, board: list[list[str]]) -> None:
    """Draw X's and O's on the board based on game state.
    
    Args:
        screen: The pygame surface to draw on.
        board: The current game board state.
    """
    from constants import (
        X_COLOR,
        O_COLOR,
        CELL_SIZE,
        MARKER_SIZE,
        BOARD_ROWS,
        BOARD_COLS
    )
    
    for row in range(BOARD_ROWS):
        for col in range(BOARD_COLS):
            marker = board[row][col]
            if marker == "X":
                # Draw X
                x_pos = col * CELL_SIZE + CELL_SIZE // 2
                y_pos = row * CELL_SIZE + CELL_SIZE // 2
                # Draw first diagonal
                pygame.draw.line(
                    screen,
                    X_COLOR,
                    (x_pos - MARKER_SIZE, y_pos - MARKER_SIZE),
                    (x_pos + MARKER_SIZE, y_pos + MARKER_SIZE),
                    MARKER_SIZE // 4
                )
                # Draw second diagonal
                pygame.draw.line(
                    screen,
                    X_COLOR,
                    (x_pos - MARKER_SIZE, y_pos + MARKER_SIZE),
                    (x_pos + MARKER_SIZE, y_pos - MARKER_SIZE),
                    MARKER_SIZE // 4

def _draw_status(screen: pygame.Surface, game: Game) -> None:
    """Draw the current game status (current player or winner).
    
    Args:
        screen: The pygame surface to draw on.
        game: The game instance.
    """
    from constants import (
        STATUS_FONT,
        STATUS_COLOR,
        STATUS_POSITION
    )
    
    font = pygame.font.SysFont(*STATUS_FONT)
    
    if game.is_game_over():
        if game.get_winner():
            text = f"Player {game.get_winner()} wins!"
        else:
            text = "Game ended in a draw!"
    else:
        text = f"Player {game.get_current_player()}'s turn"
    
    text_surface = font.render(text, True, STATUS_COLOR)
    screen.blit(text_surface, STATUS_POSITION)


def _draw_restart_button(screen: pygame.Surface) -> None:
    """Draw the restart button.
    
    Args:
        screen: The pygame surface to draw on.
    """
    from constants import (
        RESTART_BUTTON_TEXT,
        RESTART_BUTTON_FONT,
        RESTART_BUTTON_COLOR,
        RESTART_BUTTON_POSITION,
        RESTART_BUTTON_BG_COLOR,
        RESTART_BUTTON_SIZE
    )
    
    font = pygame.font.SysFont(*RESTART_BUTTON_FONT)
    text_surface = font.render(RESTART_BUTTON_TEXT, True, RESTART_BUTTON_COLOR)
    text_rect = text_surface.get_rect(center=RESTART_BUTTON_POSITION)
    
    # Draw button background
    button_rect = pygame.Rect(
        RESTART_BUTTON_POSITION[0] - RESTART_BUTTON_SIZE[0] // 2,
        RESTART_BUTTON_POSITION[1] - RESTART_BUTTON_SIZE[1] // 2,
        *RESTART_BUTTON_SIZE
    )
    pygame.draw.rect(screen, RESTART_BUTTON_BG_COLOR, button_rect)
    pygame.draw.rect(screen, RESTART_BUTTON_COLOR, button_rect, 2)
    
    # Draw button text
    screen.blit(text_surface, text_rect)

        pygame.init()
        self.screen = MagicMock()
        self.game = MagicMock()

                LINE_WIDTH
