"""
Tic Tac Toe game implementation using Pygame.
Handles game logic, board state, and rendering.
"""

import pygame
from typing import List, Optional, Tuple

# Constants would normally be imported from constants.py
# Defined here to make the file self-contained
BOARD_SIZE = 3
CELL_SIZE = 100
MARGIN = 20
LINE_WIDTH = 5
BOARD_WIDTH = BOARD_SIZE * CELL_SIZE + (BOARD_SIZE + 1) * MARGIN
BOARD_HEIGHT = BOARD_SIZE * CELL_SIZE + (BOARD_SIZE + 1) * MARGIN

# Colors
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
GRAY = (200, 200, 200)
RED = (255, 0, 0)
BLUE = (0, 0, 255)
GREEN = (0, 255, 0)

# Players
PLAYER_X = 'X'
PLAYER_O = 'O'
EMPTY = ''


class Game:
    """Main game class handling state, logic, and rendering."""

    def __init__(self, screen: pygame.Surface) -> None:
        """Initialize game with empty board and starting player."""
        self.screen = screen
        self.board: List[List[str]] = [[EMPTY for _ in range(BOARD_SIZE)] 
                                     for _ in range(BOARD_SIZE)]
        self.current_player = PLAYER_X
        self.game_over = False
        self.winner: Optional[str] = None

    def handle_click(self, pos: Tuple[int, int]) -> None:
        """
        Handle player click on the board.
        
        Args:
            pos: Tuple of (x, y) mouse coordinates
        """
        if self.game_over:
            return

        # Convert screen coordinates to board indices
        col = pos[0] // (CELL_SIZE + MARGIN)
        row = pos[1] // (CELL_SIZE + MARGIN)

        # Validate click is within board and cell is empty
        if (0 <= row < BOARD_SIZE and 0 <= col < BOARD_SIZE and 
            self.board[row][col] == EMPTY):
            self.board[row][col] = self.current_player
            
            # Check for win or draw
            self.winner = self.check_winner()
            if self.winner or self.is_draw():
                self.game_over = True
            else:
                # Switch player
                self.current_player = PLAYER_O if self.current_player == PLAYER_X else PLAYER_X

    def check_winner(self) -> Optional[str]:
        """Check if current player has won. Returns winner symbol or None."""
        # Check rows
        for row in self.board:
            if all(cell == self.current_player for cell in row):
                return self.current_player

        # Check columns
        for col in range(BOARD_SIZE):
            if all(self.board[row][col] == self.current_player for row in range(BOARD_SIZE)):
                return self.current_player

        # Check diagonals
        if all(self.board[i][i] == self.current_player for i in range(BOARD_SIZE)):
            return self.current_player
        if all(self.board[i][BOARD_SIZE-1-i] == self.current_player for i in range(BOARD_SIZE)):
            return self.current_player

        return None

    def is_draw(self) -> bool:
        """Check if the game is a draw (no empty cells and no winner)."""
        return (not any(EMPTY in row for row in self.board) and 
                self.winner is None)

    def reset(self) -> None:
        """Reset the game state to start a new game."""
        self.board = [[EMPTY for _ in range(BOARD_SIZE)] 
                      for _ in range(BOARD_SIZE)]
        self.current_player = PLAYER_X
        self.game_over = False
        self.winner = None

    def draw(self) -> None:
        """Draw the game board and current state."""
        self.screen.fill(WHITE)
        
        # Draw board grid
        for row in range(BOARD_SIZE):
            for col in range(BOARD_SIZE):
                # Calculate cell position
                x = col * (CELL_SIZE + MARGIN) + MARGIN
                y = row * (CELL_SIZE + MARGIN) + MARGIN
                
                # Draw cell background
                pygame.draw.rect(self.screen, GRAY, (x, y, CELL_SIZE, CELL_SIZE))
                
                # Draw player symbol if cell is occupied
                if self.board[row][col] == PLAYER_X:
                    # Draw X (two crossed lines)
                    pygame.draw.line(self.screen, RED, 
                                    (x + 10, y + 10), 
                                    (x + CELL_SIZE - 10, y + CELL_SIZE - 10), 
                                    LINE_WIDTH)
                    pygame.draw.line(self.screen, RED, 
                                    (x + CELL_SIZE - 10, y + 10), 
                                    (x + 10, y + CELL_SIZE - 10), 
                                    LINE_WIDTH)
                elif self.board[row][col] == PLAYER_O:
                    # Draw O (circle)
                    pygame.draw.circle(self.screen, BLUE, 
                                     (x + CELL_SIZE // 2, y + CELL_SIZE // 2), 
                                     CELL_SIZE // 2 - 10, 
                                     LINE_WIDTH)
        
        # Draw game status
        font = pygame.font.SysFont(None, 36)
        if self.game_over:
            if self.winner:
                status_text = f"Player {self.winner} wins! Click to restart."
            else:
                status_text = "Game ended in a draw! Click to restart."
        else:
            status_text = f"Player {self.current_player}'s turn"
        
        text_surface = font.render(status_text, True, BLACK)
        text_rect = text_surface.get_rect(center=(BOARD_WIDTH // 2, BOARD_HEIGHT + 30))
        self.screen.blit(text_surface, text_rect)
