## main.py

import sqlite3
import streamlit as st

class CustomerCRM:
    """A class to manage customer data in a lightweight CRM system."""

    def __init__(self) -> None:
        """Initialize the database connection and cursor."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.initialize_db()

    def initialize_db(self) -> None:
        """Create the customer table if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                name TEXT PRIMARY KEY,
                birthday TEXT,
                age INTEGER,
                sex TEXT,
                phone TEXT
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Add a new customer to the database."""
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone)
            VALUES (?, ?, ?, ?, ?)
        ''', (name, birthday, age, sex, phone))
        self.conn.commit()

    def get_customer(self, name: str) -> dict:
        """Retrieve customer data by name."""
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        if row:
            return {
                'name': row[0],
                'birthday': row[1],
                'age': row[2],
                'sex': row[3],
                'phone': row[4]
            }
        return {}

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Update existing customer data."""
        self.cursor.execute('''
            UPDATE customers
            SET birthday = ?, age = ?, sex = ?, phone = ?
            WHERE name = ?
        ''', (birthday, age, sex, phone, name))
        self.conn.commit()

    def delete_customer(self, name: str) -> None:
        """Delete a customer from the database."""
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()

    def close_connection(self) -> None:
        """Close the database connection."""
        self.conn.close()


class StreamlitUI:
    """A class to handle the Streamlit user interface for the CRM system."""

    def __init__(self, crm: CustomerCRM) -> None:
        """Initialize the Streamlit UI with a CustomerCRM instance."""
        self.crm = crm

    def display_add_form(self) -> None:
        """Display the form to add a new customer."""
        st.header("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0, value=0)
        sex = st.selectbox("Sex", options=["Male", "Female", "Other"])
        phone = st.text_input("Phone (digits only)")

        if st.button("Add Customer"):
            self.crm.add_customer(name, birthday, age, sex, phone)
            st.success(f"Customer {name} added successfully!")

    def display_view_form(self) -> None:
        """Display the form to view a customer's details."""
        st.header("View Customer")
        name = st.text_input("Enter name to search")
        
        if st.button("View Customer"):
            customer = self.crm.get_customer(name)
            if customer:
                st.write(customer)
            else:
                st.error("Customer not found.")

    def display_update_form(self) -> None:
        """Display the form to update an existing customer's details."""
        st.header("Update Customer")
        name = st.text_input("Enter name to update")
        birthday = st.text_input("New Birthday (YYYY-MM-DD)")
        age = st.number_input("New Age", min_value=0, value=0)
        sex = st.selectbox("New Sex", options=["Male", "Female", "Other"])
        phone = st.text_input("New Phone (digits only)")

        if st.button("Update Customer"):
            self.crm.update_customer(name, birthday, age, sex, phone)
            st.success(f"Customer {name} updated successfully!")

    def display_delete_form(self) -> None:
        """Display the form to delete a customer."""
        st.header("Delete Customer")
        name = st.text_input("Enter name to delete")

        if st.button("Delete Customer"):
            self.crm.delete_customer(name)
            st.success(f"Customer {name} deleted successfully!")


def main() -> None:
    """Run the Streamlit application."""
    crm = CustomerCRM()
    ui = StreamlitUI(crm)

    # Display the main options for the user
    st.title("Customer Relationship Management System")
    operation = st.selectbox("Select operation", options=["Add", "View", "Update", "Delete"])

    if operation == "Add":
        ui.display_add_form()
    elif operation == "View":
        ui.display_view_form()
    elif operation == "Update":
        ui.display_update_form()
    elif operation == "Delete":
        ui.display_delete_form()

    # Close the database connection when the app ends
    crm.close_connection()


if __name__ == "__main__":
    main()
