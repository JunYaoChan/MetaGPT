## main.py

import sqlite3
import streamlit as st

class CustomerCRUD:
    """Class to handle CRUD operations for customer data in SQLite database."""

    def __init__(self) -> None:
        """Initialize the database connection and cursor."""
        self.conn: sqlite3.Connection = sqlite3.connect('client.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.check_table_exists()

    def check_table_exists(self) -> bool:
        """Check if the customer table exists in the database.

        Returns:
            bool: True if the table exists, False otherwise.
        """
        self.cursor.execute("SELECT name FROM sqlite_master WHERE type='table' AND name='customers';")
        return self.cursor.fetchone() is not None

    def create_table(self) -> None:
        """Create the customer table in the database."""
        self.cursor.execute('''
            CREATE TABLE customers (
                name TEXT PRIMARY KEY,
                birthday TEXT,
                age INTEGER,
                sex TEXT,
                phone TEXT
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> bool:
        """Add a new customer to the database.

        Args:
            name (str): The name of the customer.
            birthday (str): The birthday of the customer.
            age (int): The age of the customer.
            sex (str): The sex of the customer.
            phone (str): The phone number of the customer.

        Returns:
            bool: True if the customer was added successfully, False otherwise.
        """
        try:
            self.cursor.execute('''
                INSERT INTO customers (name, birthday, age, sex, phone)
                VALUES (?, ?, ?, ?, ?)
            ''', (name, birthday, age, sex, phone))
            self.conn.commit()
            return True
        except sqlite3.IntegrityError:
            return False

    def query_customer(self, name: str) -> dict:
        """Query a customer by name.

        Args:
            name (str): The name of the customer to query.

        Returns:
            dict: A dictionary containing customer data if found, otherwise an empty dict.
        """
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        row = self.cursor.fetchone()
        return {
            'name': row[0],
            'birthday': row[1],
            'age': row[2],
            'sex': row[3],
            'phone': row[4]
        } if row else {}

    def update_customer(self, name: str, field: str, new_value: str) -> bool:
        """Update a customer's information.

        Args:
            name (str): The name of the customer to update.
            field (str): The field to update (birthday, age, sex, phone).
            new_value (str): The new value for the specified field.

        Returns:
            bool: True if the update was successful, False otherwise.
        """
        if field not in ['birthday', 'age', 'sex', 'phone']:
            return False
        self.cursor.execute(f'UPDATE customers SET {field} = ? WHERE name = ?', (new_value, name))
        self.conn.commit()
        return self.cursor.rowcount > 0

    def delete_customer(self, name: str) -> bool:
        """Delete a customer from the database.

        Args:
            name (str): The name of the customer to delete.

        Returns:
            bool: True if the customer was deleted successfully, False otherwise.
        """
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()
        return self.cursor.rowcount > 0


class StreamlitUI:
    """Class to render the Streamlit user interface for the CRM system."""

    def __init__(self, crud: CustomerCRUD) -> None:
        """Initialize the Streamlit UI with a CustomerCRUD instance.

        Args:
            crud (CustomerCRUD): An instance of the CustomerCRUD class.
        """
        self.crud: CustomerCRUD = crud

    def render_add_tab(self) -> None:
        """Render the Add Customer tab in the Streamlit UI."""
        st.header("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=0, max_value=120)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")

        if st.button("Add Customer"):
            if self.crud.add_customer(name, birthday, age, sex, phone):
                st.success("Customer added successfully!")
            else:
                st.error("Error: Customer with this name already exists.")

    def render_query_tab(self) -> None:
        """Render the Query Customer tab in the Streamlit UI."""
        st.header("Query Customer")
        name = st.text_input("Enter Customer Name")

        if st.button("Query"):
            customer = self.crud.query_customer(name)
            if customer:
                st.write(customer)
            else:
                st.error("Customer not found.")

    def render_update_tab(self) -> None:
        """Render the Update Customer tab in the Streamlit UI."""
        st.header("Update Customer")
        name = st.text_input("Enter Customer Name to Update")
        field = st.selectbox("Select Field to Update", ["birthday", "age", "sex", "phone"])
        new_value = st.text_input("New Value")

        if st.button("Update Customer"):
            if self.crud.update_customer(name, field, new_value):
                st.success("Customer updated successfully!")
            else:
                st.error("Error: Update failed. Check if the customer exists.")

    def render_delete_tab(self) -> None:
        """Render the Delete Customer tab in the Streamlit UI."""
        st.header("Delete Customer")
        name = st.text_input("Enter Customer Name to Delete")

        if st.button("Delete Customer"):
            if self.crud.delete_customer(name):
                st.success("Customer deleted successfully!")
            else:
                st.error("Error: Customer not found.")

    def run(self) -> None:
        """Run the Streamlit application."""
        st.title("Customer Relationship Management System")
        tab1, tab2, tab3, tab4 = st.tabs(["Add", "Query", "Update", "Delete"])
        
        with tab1:
            self.render_add_tab()
        with tab2:
            self.render_query_tab()
        with tab3:
            self.render_update_tab()
        with tab4:
            self.render_delete_tab()


if __name__ == "__main__":
    # Create an instance of CustomerCRUD and StreamlitUI, then run the app.
    crud = CustomerCRUD()
    ui = StreamlitUI(crud)
    ui.run()

## instruction:
The errors occur in the test file `test_main.py` when testing the StreamlitUI class methods `render_add_tab` and `render_update_tab`. The issue is with the mock setup for `text_input` - it's using `side_effect` with a list of values but the methods call `text_input` more times than the number of values provided, causing a `StopIteration` error when the mock runs out of values to return.
