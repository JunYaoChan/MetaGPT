import sqlite3
import streamlit as st

class CustomerCRUD:
    """Class to handle CRUD operations for customers in the SQLite database."""

    def __init__(self) -> None:
        """Initialize the database connection and create the customer table if it doesn't exist."""
        self.conn: sqlite3.Connection = sqlite3.connect('customers.db')
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()

    def create_table(self) -> None:
        """Create the customer table if it does not exist."""
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS customers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                birthday TEXT NOT NULL,
                age INTEGER NOT NULL CHECK(age > 0),
                sex TEXT NOT NULL,
                phone TEXT NOT NULL
            )
        ''')
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> bool:
        """Add a new customer to the database.

        Args:
            name (str): The name of the customer.
            birthday (str): The birthday of the customer.
            age (int): The age of the customer.
            sex (str): The sex of the customer.
            phone (str): The phone number of the customer.

        Returns:
            bool: True if the customer was added successfully, False otherwise.
        """
        if not self.validate_customer_data(name, birthday, age, sex, phone):
            return False
        
        self.cursor.execute('''
            INSERT INTO customers (name, birthday, age, sex, phone)
            VALUES (?, ?, ?, ?, ?)
        ''', (name, birthday, age, sex, phone))
        self.conn.commit()
        return True

    def search_customer(self, name: str) -> dict:
        """Search for a customer by name.

        Args:
            name (str): The name of the customer to search for.

        Returns:
            dict: A dictionary containing customer data if found, otherwise an empty dictionary.
        """
        self.cursor.execute('SELECT * FROM customers WHERE name = ?', (name,))
        customer = self.cursor.fetchone()
        if customer:
            return {
                'id': customer[0],
                'name': customer[1],
                'birthday': customer[2],
                'age': customer[3],
                'sex': customer[4],
                'phone': customer[5]
            }
        return {}

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> bool:
        """Update an existing customer's information.

        Args:
            name (str): The name of the customer to update.
            birthday (str): The new birthday of the customer.
            age (int): The new age of the customer.
            sex (str): The new sex of the customer.
            phone (str): The new phone number of the customer.

        Returns:
            bool: True if the customer was updated successfully, False otherwise.
        """
        if not self.validate_customer_data(name, birthday, age, sex, phone):
            return False
        
        self.cursor.execute('''
            UPDATE customers
            SET birthday = ?, age = ?, sex = ?, phone = ?
            WHERE name = ?
        ''', (birthday, age, sex, phone, name))
        self.conn.commit()
        return True

    def delete_customer(self, name: str) -> bool:
        """Delete a customer from the database.

        Args:
            name (str): The name of the customer to delete.

        Returns:
            bool: True if the customer was deleted successfully, False otherwise.
        """
        self.cursor.execute('DELETE FROM customers WHERE name = ?', (name,))
        self.conn.commit()
        return self.cursor.rowcount > 0

    def validate_customer_data(self, name: str, birthday: str, age: int, sex: str, phone: str) -> bool:
        """Validate customer data before adding or updating.

        Args:
            name (str): The name of the customer.
            birthday (str): The birthday of the customer.
            age (int): The age of the customer.
            sex (str): The sex of the customer.
            phone (str): The phone number of the customer.

        Returns:
            bool: True if data is valid, False otherwise.
        """
        if not name or not birthday or age <= 0 or not sex or not phone:
            return False
        return True


class StreamlitUI:
    """Class to manage the Streamlit user interface for the CRM system."""

    def __init__(self, crud: CustomerCRUD) -> None:
        """Initialize the Streamlit UI with the CustomerCRUD instance.

        Args:
            crud (CustomerCRUD): An instance of the CustomerCRUD class.
        """
        self.crud: CustomerCRUD = crud

    def render(self) -> None:
        """Render the main interface for the application."""
        st.title("Customer Relationship Management System")
        operation = st.selectbox("Select Operation", ["Add", "Search", "Update", "Delete"])

        if operation == "Add":
            self.render_add_form()
        elif operation == "Search":
            self.render_search_form()
        elif operation == "Update":
            self.render_update_form()
        elif operation == "Delete":
            self.render_delete_form()

    def render_add_form(self) -> None:
        """Render the form for adding a new customer."""
        st.subheader("Add Customer")
        name = st.text_input("Name")
        birthday = st.text_input("Birthday (YYYY-MM-DD)")
        age = st.number_input("Age", min_value=1)
        sex = st.selectbox("Sex", ["Male", "Female", "Other"])
        phone = st.text_input("Phone")

        if st.button("Add Customer"):
            if self.crud.add_customer(name, birthday, age, sex, phone):
                st.success("Customer added successfully!")
            else:
                st.error("Failed to add customer. Please check your input.")

    def render_search_form(self) -> None:
        """Render the form for searching a customer."""
        st.subheader("Search Customer")
        name = st.text_input("Enter Name")

        if st.button("Search"):
            customer = self.crud.search_customer(name)
            if customer:
                st.write(customer)
            else:
                st.error("Customer not found.")

    def render_update_form(self) -> None:
        """Render the form for updating a customer's information."""
        st.subheader("Update Customer")
        name = st.text_input("Enter Name to Update")
        birthday = st.text_input("New Birthday (YYYY-MM-DD)")
        age = st.number_input("New Age", min_value=1)
        sex = st.selectbox("New Sex", ["Male", "Female", "Other"])
        phone = st.text_input("New Phone")

        if st.button("Update Customer"):
            if self.crud.update_customer(name, birthday, age, sex, phone):
                st.success("Customer updated successfully!")
            else:
                st.error("Failed to update customer. Please check your input.")

    def render_delete_form(self) -> None:
        """Render the form for deleting a customer."""
        st.subheader("Delete Customer")
        name = st.text_input("Enter Name to Delete")

        if st.button("Delete Customer"):
            if self.crud.delete_customer(name):
                st.success("Customer deleted successfully!")
            else:
                st.error("Failed to delete customer. Please check the name.")

if __name__ == "__main__":
    # Create an instance of CustomerCRUD and StreamlitUI, then render the UI
    crud = CustomerCRUD()
    ui = StreamlitUI(crud)
    ui.render()
