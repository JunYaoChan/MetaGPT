import sqlite3
import streamlit as st

class CustomerManager:
    """Handles all database operations related to customer data."""

    def __init__(self, db_name: str = "customers.db") -> None:
        """Initializes the CustomerManager with a database name and creates a connection."""
        self.db_name: str = db_name
        self.conn: sqlite3.Connection = sqlite3.connect(self.db_name)
        self.cursor: sqlite3.Cursor = self.conn.cursor()
        self.create_table()

    def create_table(self) -> None:
        """Creates the customer table if it does not exist."""
        create_table_query: str = """
        CREATE TABLE IF NOT EXISTS customers (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            birthday TEXT NOT NULL,
            age INTEGER NOT NULL,
            sex TEXT NOT NULL,
            phone TEXT NOT NULL
        );
        """
        self.cursor.execute(create_table_query)
        self.conn.commit()

    def add_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Adds a new customer to the database."""
        insert_query: str = """
        INSERT INTO customers (name, birthday, age, sex, phone)
        VALUES (?, ?, ?, ?, ?);
        """
        self.cursor.execute(insert_query, (name, birthday, age, sex, phone))
        self.conn.commit()

    def query_customer(self, name: str) -> dict:
        """Queries a customer by name and returns their data."""
        select_query: str = "SELECT * FROM customers WHERE name = ?;"
        self.cursor.execute(select_query, (name,))
        customer_data: tuple = self.cursor.fetchone()
        if customer_data:
            return {
                "id": customer_data[0],
                "name": customer_data[1],
                "birthday": customer_data[2],
                "age": customer_data[3],
                "sex": customer_data[4],
                "phone": customer_data[5]
            }
        return {}

    def update_customer(self, name: str, birthday: str, age: int, sex: str, phone: str) -> None:
        """Updates an existing customer's information."""
        update_query: str = """
        UPDATE customers
        SET birthday = ?, age = ?, sex = ?, phone = ?
        WHERE name = ?;
        """
        self.cursor.execute(update_query, (birthday, age, sex, phone, name))
        self.conn.commit()

        """Deletes a customer from the database."""
        delete_query: str = "DELETE FROM customers WHERE name = ?;"

        self.conn.commit()

        """Closes the database connection."""
        self.conn.close()

        st.title("Customer Relationship Management System")

        # Add Customer
        st.header("Add Customer")
        with st.form(key='add_customer_form'):
            name = st.text_input("Name")
            birthday = st.text_input("Birthday (YYYY-MM-DD)")
            age = st.number_input("Age", min_value=0)
            sex = st.selectbox("Sex", ["Male", "Female", "Other"])
            phone = st.text_input("Phone")
            submit_button = st.form_submit_button("Add Customer")
            if submit_button:
                self.manager.add_customer(name, birthday, age, sex, phone)
                st.success("Customer added successfully!")

        # Query Customer
        st.header("Query Customer")
        query_name = st.text_input("Enter name to query")
        if st.button("Query"):
            customer = self.manager.query_customer(query_name)
            if customer:
                st.write(customer)
            else:
                st.warning("Customer not found.")

        # Update Customer
        st.header("Update Customer")
        with st.form(key='update_customer_form'):
            update_name = st.text_input("Name to update")
