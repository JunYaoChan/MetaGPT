## main.py
import tkinter as tk

class Calculator:
    """A simple calculator application using Tkinter for basic arithmetic operations."""

    def __init__(self) -> None:
        """Initialize the calculator application."""
        self.root = tk.Tk()
        self.root.title("Simple Calculator")
        self.display = tk.Entry(self.root, width=16, font=('Arial', 24), borderwidth=2, relief='solid')
        self.display.grid(row=0, column=0, columnspan=4)
        self.buttons = {}
        self.create_buttons()

    def create_display(self) -> None:
        """Create the display for the calculator."""
        self.display.delete(0, tk.END)  # Clear the display initially

    def create_buttons(self) -> None:
        """Create buttons for the calculator."""
        # Define button layout
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            'C', '0', '=', '+'
        ]
        
        # Create buttons and place them in the grid
        for index, text in enumerate(button_texts):
            button = tk.Button(self.root, text=text, width=5, height=2, command=lambda t=text: self.add_to_display(t))
            button.grid(row=(index // 4) + 1, column=index % 4)
            self.buttons[text] = button

    def add_to_display(self, value: str) -> None:
        """Add a value to the display."""
        if value == 'C':
            self.clear_display()
        elif value == '=':
            self.calculate()
        else:
            current_text = self.display.get()
            new_text = current_text + value
            self.display.delete(0, tk.END)
            self.display.insert(0, new_text)

    def clear_display(self) -> None:
        """Clear the display."""
        self.display.delete(0, tk.END)

    def calculate(self) -> None:
        """Calculate the result of the expression in the display."""
        try:
            expression = self.display.get()
            result = eval(expression)  # Evaluate the arithmetic expression
            self.clear_display()
            self.display.insert(0, str(result))
        except Exception as e:
            self.clear_display()
            self.display.insert(0, "Error")

    def run(self) -> None:
        """Run the calculator application."""
        self.create_display()
        self.root.mainloop()

if __name__ == "__main__":
    calculator = Calculator()
    calculator.run()
